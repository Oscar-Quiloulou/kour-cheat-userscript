    // Public builds contain only the verification key. The PRIVATE owner build

    const HAZY_OWNER_PUBLIC_JWK = {"kty":"EC","crv":"P-256","x":"LdvyZSUL6r8-w52qu-m40_FqtekjSxlemO22BkbaI4Q","y":"PEke8HNQkAgQahZDiHr6t6G95KU1L4jA9adSN0DuVAI","ext":true};
    const HAZY_OWNER_PRIVATE_JWK = null;

    const HAZY_BADGE_MAX_PEERS = 64;

    // Built in, not a mod: active from startup. This flag only pauses the

    // Entries and signatures read from the user-supplied Kour 4.24 archive.
    // Allocations are managed objects, rooted with GCHandle while JS holds them.
    // SetCustomProperties gets a NEW one-entry Hashtable, never the game's
    // entire property dictionary and never a made-up object in raw WASM memory.
    const HAZY_BADGE_NATIVE = {
        inRoom:             [0x6c9cdc, 27689, 1],
        currentRoom:        [0x6c9c7c, 27666, 1],
        localPlayer:        [0x6c9c80, 27667, 1],

        cloudRegion:        [0x6c9c50, 27655, 1],
        gameVersion:        [0x6c9c38, 27649, 1],

    };
    let hazyBadgeNative = null;
    let hazyBadgeSession = null;
    let hazyBadgeEpoch = 0;
    let hazyBadgeBusy = false;
    let hazyBadgeRetryAt = 0;
    let hazyBadgeWarned = false;
    let hazyBadgeVerifyKey = null;
    let hazyBadgeSignKey = null;
    const hazyBadgeStatus = { state: 'waiting for room', peers: 0, owners: 0, lastError: '' };
    Object.defineProperty(window, '__hazyBadgeStatus', {
        configurable: true, get: () => ({ ...hazyBadgeStatus })
    });

    function badgeFreshNonce() {
        const bytes = new Uint8Array(16);
        crypto.getRandomValues(bytes);
        return Array.from(bytes, n => n.toString(16).padStart(2, '0')).join('');
    }
    function badgeProofBytes(scope, owner, ownerNonce, viewer, viewerNonce) {
        // The viewer chooses its own fresh nonce; an old signature copied to
        // another actor, room, region, or new session is not an owner proof.
        return new TextEncoder().encode(JSON.stringify([
            'hazy-owner-proof-v1', scope, owner, ownerNonce, viewer, viewerNonce
        ]));
    }
    function badgeBase64(bytes) {
        return btoa(String.fromCharCode(...new Uint8Array(bytes)))
            .replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/g, '');
    }
    function badgeSignatureBytes(text) {
        if (typeof text !== 'string' || !/^[A-Za-z0-9_-]{86}$/.test(text)) return null;
        try {
            const raw = atob(text.replace(/-/g, '+').replace(/_/g, '/') + '==');
            return raw.length === 64 ? Uint8Array.from(raw, c => c.charCodeAt(0)) : null;
        } catch (_) { return null; }
    }
    function parseHazyBadge(text, actor) {
        if (typeof text !== 'string' || text.length > HAZY_BADGE_MAX_TEXT) return null;
        try {
            const data = JSON.parse(text);
            if (!data || data.v !== 1 || data.a !== actor ||
                !Number.isInteger(actor) || actor < 1 || actor > 1000000 ||
                typeof data.n !== 'string' || !/^[a-f0-9]{32}$/.test(data.n) ||
                !['u', 'o'].includes(data.r) || !Array.isArray(data.p) ||
                data.p.length > HAZY_BADGE_MAX_PEERS) return null;
            const proofs = [];
            for (const entry of data.p) {
                if (!Array.isArray(entry) || entry.length !== 3 ||
                    !Number.isInteger(entry[0]) || entry[0] < 1 || entry[0] > 1000000 ||
                    typeof entry[1] !== 'string' || !/^[a-f0-9]{32}$/.test(entry[1]) ||
                    !badgeSignatureBytes(entry[2])) continue;
                proofs.push(entry);
            }
            return { v: 1, a: actor, n: data.n, r: data.r, p: proofs };
        } catch (_) { return null; }
    }
    async function prepareBadgeKeys() {
        if (!crypto?.subtle) throw new Error('Owner verification needs HTTPS/Web Crypto.');
        if (!hazyBadgeVerifyKey) {
            hazyBadgeVerifyKey = await crypto.subtle.importKey('jwk',
                HAZY_OWNER_PUBLIC_JWK, { name: 'ECDSA', namedCurve: 'P-256' }, false, ['verify']);
        }
        if (HAZY_OWNER_PRIVATE_JWK && !hazyBadgeSignKey) {
            const key = await crypto.subtle.importKey('jwk', HAZY_OWNER_PRIVATE_JWK,
                { name: 'ECDSA', namedCurve: 'P-256' }, false, ['sign']);
            // Check the private build's key against the public trust anchor.
            const probe = new TextEncoder().encode('hazy-key-check:' + badgeFreshNonce());
            const signature = await crypto.subtle.sign({ name: 'ECDSA', hash: 'SHA-256' }, key, probe);
            if (!await crypto.subtle.verify({ name: 'ECDSA', hash: 'SHA-256' },
                hazyBadgeVerifyKey, signature, probe)) throw new Error('Owner key does not match this release.');
            hazyBadgeSignKey = key;
        }
    }
    function releaseBadgeNative() {
        const old = hazyBadgeNative;
        hazyBadgeNative = null;
        if (!old) return;
        try { if (old.keyHandle) old.freeObject(old.keyHandle, 0); } catch (_) {}
        try { old.freeScratch(); } catch (_) {}
    }
    function getBadgeNative() {
        const runtime = findSceneRuntime();
        if (!runtime) return null;
        const view = sceneView(runtime);
        if (!Object.values(HAZY_BADGE_NATIVE).every(([address, slot]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === slot)) {
            throw new Error('Badge networking layout differs from the supplied Kour build.');
        }
        if (hazyBadgeNative?.runtime.table === runtime.table &&
            infoValid(view, hazyBadgeNative.scratch, 16384)) {
            hazyBadgeNative.runtime = runtime;
            return hazyBadgeNative;
        }
        releaseBadgeNative();
        const api = { runtime, keyHandle: 0, key: 0, scratch: 0, freeScratch: () => {} };
        for (const [name, [, slot, arity]] of Object.entries(HAZY_BADGE_NATIVE)) {
            const fn = sceneFunction(runtime, slot);
            if (fn.length !== arity) throw new Error(`Badge method ${name} has an unexpected signature.`);
            api[name] = fn;
        }
        for (const owner of [runtime.module, runtime.exports, runtime.module?.asm, runtime.module?.wasmExports]) {
            const alloc = owner?._malloc || owner?.malloc;
            const free = owner?._free || owner?.free;
            if (typeof alloc !== 'function' || typeof free !== 'function') continue;
            const pointer = Reflect.apply(alloc, owner, [16384]) >>> 0;
            if (!infoValid(sceneView(runtime), pointer, 16384)) {
                if (pointer) Reflect.apply(free, owner, [pointer]);
                throw new Error('Badge scratch allocation failed.');
            }
            api.scratch = pointer;
            api.freeScratch = () => Reflect.apply(free, owner, [pointer]);
            break;
        }
        if (!api.scratch) throw new Error('Badge allocator unavailable. Save and refresh Kour.');
        try {
            api.key = createBadgeString(api, HAZY_BADGE_KEY);
            api.keyHandle = api.keepObject(api.key, 0);
            if (!api.keyHandle) throw new Error('Badge key could not be retained.');
        } catch (error) { api.freeScratch(); throw error; }
        hazyBadgeNative = api;
        return api;
    }
    function createBadgeString(api, text) {
        if (typeof text !== 'string' || text.length > HAZY_BADGE_MAX_TEXT || /[^\x20-\x7e]/.test(text)) {
            throw new Error('Badge payload must be bounded ASCII JSON.');
        }
        const view = sceneView(api.runtime);
        for (let i = 0; i < text.length; i++) view.setUint8(api.scratch + i, text.charCodeAt(i));
        view.setUint8(api.scratch + text.length, 0);
        const pointer = api.utf8String(api.scratch, text.length, 0);
        const fresh = sceneView(api.runtime);
        if (!infoValid(fresh, pointer, 12 + text.length * 2) ||
            !infoClassNamed(fresh, infoUint(fresh, pointer), 'String') ||
            fresh.getInt32(pointer + 8, true) !== text.length) throw new Error('Badge string creation failed.');
        return pointer;
    }
    function badgeContext(api) {
        if (api.inRoom(0) !== 1) return null;
        const room = api.currentRoom(0), player = api.localPlayer(0);
        let view = sceneView(api.runtime);
        if (!infoValid(view, room, 12) || !infoClassNamed(view, infoUint(view, room), 'Room') ||
            !infoValid(view, player, 40) || !infoClassNamed(view, infoUint(view, player), 'Player') ||
            view.getUint8(player + 0x10) !== 1) return null;
        const actor = view.getInt32(player + 0x0c, true);
        if (actor < 1 || actor > 1000000) return null;
        const read = pointer => {
            // Room identity must be exact, not display-name sanitized/truncated.
            const fresh = sceneView(api.runtime);
            if (!infoValid(fresh, pointer, 12) ||
                !infoClassNamed(fresh, infoUint(fresh, pointer), 'String')) return '';
            const length = fresh.getInt32(pointer + 8, true);
            if (length < 1 || length > 512 || !infoValid(fresh, pointer, 12 + length * 2)) return '';
            let text = '';
            for (let i = 0; i < length; i++) text += String.fromCharCode(fresh.getUint16(pointer + 12 + 2 * i, true));
            return text;
        };
        const roomName = read(api.roomName(room, 0));
        const region = read(api.cloudRegion(0));
        const gameVersion = read(api.gameVersion(0));
        if (!roomName || !region || !gameVersion) return null;
        return { api, room, player, actor, scope: JSON.stringify([region, gameVersion, roomName]) };
    }
    function sameBadgeContext(a, b) {
        return !!(a && b && a.api.runtime.table === b.api.runtime.table &&
            a.room === b.room && a.player === b.player && a.actor === b.actor && a.scope === b.scope);
    }
    function readBadgeProperty(api, player) {
        const props = api.customProperties(player, 0);
        const view = sceneView(api.runtime);
        if (!infoValid(view, props, 12) || !infoClassNamed(view, infoUint(view, props), 'Hashtable')) return null;
        const value = api.getItem(props, api.key, 0);
        const fresh = sceneView(api.runtime);
        if (!value || !infoValid(fresh, value, 12) ||
            !infoClassNamed(fresh, infoUint(fresh, value), 'String')) return null;
        const length = fresh.getInt32(value + 8, true);
        if (length < 1 || length > HAZY_BADGE_MAX_TEXT ||
            !infoValid(fresh, value, 12 + length * 2)) return null;
        // The general name helper intentionally caps strings at 4096 chars;
        // an owner proof list can be longer. This dedicated reader is bounded.
        let text = '';
        for (let i = 0; i < length; i++) {
            const c = fresh.getUint16(value + 12 + i * 2, true);
            if (c < 0x20 || c > 0x7e) return null;
            text += String.fromCharCode(c);
        }
        return text;
    }
    function publishBadge(context, value) {
        const { api, player } = context;
        if (!sameBadgeContext(context, badgeContext(api))) return false;
        const handles = [];
        const hold = object => {
            if (!object) throw new Error('Badge managed allocation is unavailable.');
            const handle = api.keepObject(object, 0);
            if (!handle) throw new Error('Badge managed object could not be retained.');
            handles.push(handle); return object;
        };
        try {
            const existing = api.customProperties(player, 0);
            let view = sceneView(api.runtime);
            if (!infoValid(view, existing, 12) ||
                !infoClassNamed(view, infoUint(view, existing), 'Hashtable')) return false;
            const type = hold(api.objectType(existing, 0));
            const freshTable = hold(api.createInstance(type, 0));
            view = sceneView(api.runtime);
            if (!infoValid(view, freshTable, 12) ||
                infoUint(view, freshTable) !== infoUint(view, existing)) {
                throw new Error('Badge Hashtable creation failed.');
            }
            const str = value === null ? 0 : hold(createBadgeString(api, value));
            api.setItem(freshTable, api.key, str, 0);
            // ONLY our own actor and single namespaced field. Other properties
            // and player names are neither copied into the update nor edited.
            return api.setProperties(player, freshTable, 0, 0, 0) === 1;
        } finally {
            for (let i = handles.length - 1; i >= 0; i--) {
                try { api.freeObject(handles[i], 0); } catch (_) {}
            }
        }
    }
    function readBadgePeers(context) {
        const { api } = context;
        const array = api.playerList(0);
        let view = sceneView(api.runtime);
        if (!infoValid(view, array, 16)) return [];
        const count = view.getInt32(array + 12, true);
        if (count < 0 || count > 256 || !infoValid(view, array, 16 + count * 4)) {
            throw new Error('Badge player list is outside the expected bounds.');
        }
        const root = api.keepObject(array, 0);
        if (!root) throw new Error('Badge player list could not be retained.');
        const peers = [];
        try {
            for (let i = 0; i < count && peers.length < HAZY_BADGE_MAX_PEERS; i++) {
                view = sceneView(api.runtime);
                const player = infoUint(view, array + 16 + i * 4);
                if (!infoValid(view, player, 40) ||
                    !infoClassNamed(view, infoUint(view, player), 'Player') ||
                    view.getUint8(player + 0x1c) !== 0) continue; // IsInactive
                const actor = view.getInt32(player + 0x0c, true);
                if (actor === context.actor) continue;
                const marker = parseHazyBadge(readBadgeProperty(api, player), actor);
                if (marker) peers.push(marker);
            }
            return peers.sort((a, b) => a.a - b.a);
        } finally { api.freeObject(root, 0); }
    }
    function clearBadgeSession() {
        hazyBadgeSession = null;
        hazyBadgeEpoch++;
        hazyBadgeRecords.clear();
        hazyBadgeStatus.peers = 0; hazyBadgeStatus.owners = 0;
    }
    function badgeRoleForActor(actor) {
        if (!hazyBadges || !hazyBadgeSession) return '';
        const record = hazyBadgeRecords.get(actor);
        return record && performance.now() - record.t <= 4000 ? record.role : '';
    }
    async function updateHazyBadges() {
        if (!hazyBadges || hazyBadgeBusy || performance.now() < hazyBadgeRetryAt) return;
        hazyBadgeBusy = true;
        try {
            const api = getBadgeNative();
            const context = api && badgeContext(api);
            if (!context) {
                clearBadgeSession(); hazyBadgeStatus.state = 'waiting for room'; return;
            }
            const now = performance.now();
            if (!sameBadgeContext(hazyBadgeSession, context) ||
                now - hazyBadgeSession.started >= HAZY_BADGE_NONCE_MS) {
                clearBadgeSession();
                hazyBadgeSession = { ...context, epoch: hazyBadgeEpoch, n: badgeFreshNonce(),
                    started: now, sent: null, sentAt: -Infinity, signatures: new Map(), verified: new Map() };
            }
            const session = hazyBadgeSession;
            await prepareBadgeKeys();
            if (!hazyBadges || hazyBadgeSession !== session) return;
            const peers = readBadgePeers(context);
            const proofs = [];
            if (hazyBadgeSignKey) {
                for (const peer of peers) {
                    const token = `${peer.a}:${peer.n}`;
                    let signature = session.signatures.get(token);
                    if (!signature) {
                        const data = badgeProofBytes(session.scope, session.actor, session.n, peer.a, peer.n);
                        signature = badgeBase64(await crypto.subtle.sign(
                            { name: 'ECDSA', hash: 'SHA-256' }, hazyBadgeSignKey, data));
                        session.signatures.set(token, signature);
                    }
                    proofs.push([peer.a, peer.n, signature]);
                }
                if (session.signatures.size > 128) session.signatures.clear();
            }
            const next = new Map();
            for (const peer of peers) {
                let owner = false;
                const proof = peer.r === 'o' && peer.p.find(p => p[0] === session.actor && p[1] === session.n);
                if (proof) {
                    const token = `${peer.a}:${peer.n}:${proof[2]}`;
                    if (session.verified.has(token)) owner = session.verified.get(token);
                    else {
                        owner = await crypto.subtle.verify({ name: 'ECDSA', hash: 'SHA-256' },
                            hazyBadgeVerifyKey, badgeSignatureBytes(proof[2]),
                            badgeProofBytes(session.scope, peer.a, peer.n, session.actor, session.n));
                        session.verified.set(token, owner);
                    }
                }
                next.set(peer.a, { role: owner ? 'owner' : 'user', t: performance.now() });
            }
            if (session.verified.size > 256) session.verified.clear();
            // Native references never survive an await without rechecking the
            // room and local actor. Leaving a room cancels pending badge writes.
            if (!hazyBadges || hazyBadgeSession !== session ||
                !sameBadgeContext(session, badgeContext(api))) return;
            const marker = JSON.stringify({ v: 1, a: session.actor, n: session.n,
                r: hazyBadgeSignKey ? 'o' : 'u', p: proofs });
            if (marker !== session.sent && performance.now() - session.sentAt >= 1000) {
                session.sentAt = performance.now();
                if (publishBadge(context, marker)) session.sent = marker;
                else hazyBadgeRetryAt = performance.now() + 5000;
            }
            hazyBadgeRecords.clear();
            for (const [actor, record] of next) hazyBadgeRecords.set(actor, record);
            hazyBadgeRecords.set(session.actor, {
                role: hazyBadgeSignKey ? 'owner' : 'user', t: performance.now()
            });
            hazyBadgeStatus.state = session.sent ? 'sharing in current room' : 'waiting to announce';
            hazyBadgeStatus.peers = next.size;
            hazyBadgeStatus.owners = [...next.values()].filter(p => p.role === 'owner').length;
            hazyBadgeStatus.lastError = ''; hazyBadgeWarned = false;
        } catch (error) {
            hazyBadgeRecords.clear();
            hazyBadgeStatus.state = 'unavailable';
            hazyBadgeStatus.lastError = error.message || String(error);
            hazyBadgeRetryAt = performance.now() + 10000;
            if (!hazyBadgeWarned) {
                hazyBadgeWarned = true;
                reportSceneError('Hazy Badges', error);
            }
        } finally { hazyBadgeBusy = false; }
    }
    // Automatic startup/retry: joining a room needs no menu interaction.
    window.setInterval(() => { void updateHazyBadges(); }, HAZY_BADGE_REFRESH_MS);
    window.addEventListener('pageshow', () => {
        // Resume after Back/Forward cache restoration as well as a fresh load.
        hazyBadges = true;
        hazyBadgeRetryAt = 0; hazyBadgeWarned = false;
        void updateHazyBadges();
    });
    window.addEventListener('pagehide', () => {
        const session = hazyBadgeSession;
        hazyBadges = false; clearBadgeSession();
        // Best effort: remove ONLY our own namespaced presence property.
        // The session check prevents writes into a different room.
        try { if (session) publishBadge(session, null); } catch (_) {}
        hazyBadgeStatus.state = 'page closed';
        releaseBadgeNative();
    });

    // ============================================================
    // SOFT MENU-OPENING DING — NO SOUND FILE OR NETWORK REQUEST
    // ============================================================
    // A low digital gain; actual loudness also depends on device volume.
    // Set to 0 to mute. Keep low when using headphones.
    const MENU_DING_VOLUME = 0.035;

    const menuDing = (() => {
        let audioContext = null;
        let requestId = 0;
        let lastPlayedAt = -Infinity;

        function getContext() {
            if (!(MENU_DING_VOLUME > 0)) return null;
            if (!audioContext || audioContext.state === 'closed') {
                const AudioContextClass =
                    window.AudioContext || window.webkitAudioContext;
                if (!AudioContextClass) return null;
                audioContext = new AudioContextClass({ latencyHint: 'interactive' });
            }
            return audioContext;
        }

        function prime() {
            // A trusted click can unlock audio, but never plays a sound itself.
            try {
                const ctx = getContext();
                if (ctx && ctx.state !== 'running') {
                    Promise.resolve(ctx.resume()).catch(() => {});
                }
            } catch (_) {
                // Sound must never prevent the menu or game from working.
            }
        }

        function tone(ctx, start, frequency, level, duration) {
            const oscillator = ctx.createOscillator();
            const envelope = ctx.createGain();
            oscillator.type = 'sine';
            oscillator.frequency.setValueAtTime(frequency, start);

            // Gentle attack and short fading tail, rather than an abrupt beep.
            envelope.gain.setValueAtTime(0, start);
            envelope.gain.linearRampToValueAtTime(level, start + 0.012);
            envelope.gain.exponentialRampToValueAtTime(0.00001, start + duration - 0.025);
            envelope.gain.linearRampToValueAtTime(0, start + duration);

            oscillator.connect(envelope);
            envelope.connect(ctx.destination);
            oscillator.onended = () => {
                oscillator.disconnect();
                envelope.disconnect();
            };
            oscillator.start(start);
            oscillator.stop(start + duration + 0.01);
        }

        async function play() {
            const thisRequest = ++requestId;
            const requestedAt = performance.now();
            if (!menuOpen || document.hidden || !(MENU_DING_VOLUME > 0)) return;

            try {
                const ctx = getContext();
                if (!ctx) return;
                if (ctx.state !== 'running') await ctx.resume();

                const now = performance.now();
                // Drop blocked/stale requests: no surprise ding on a later click.
                // Also avoid stacking sounds when Right Shift is tapped rapidly.
                if (thisRequest !== requestId || !menuOpen || document.hidden ||
                    ctx.state !== 'running' || now - requestedAt > 250 ||
                    now - lastPlayedAt < 180) return;

                lastPlayedAt = now;
                const start = ctx.currentTime + 0.005;
                const volume = Math.min(0.1, Math.max(0, MENU_DING_VOLUME));
                tone(ctx, start, 880, volume, 0.28);
                tone(ctx, start, 1760, volume * 0.16, 0.17);
            } catch (_) {
                // Audio failure does not affect menu visibility or mod toggles.
            }
        }

        function cancelPending() {
            requestId++;
        }

        function dispose() {
            cancelPending();
            const ctx = audioContext;
            audioContext = null;
            try {
                if (ctx && ctx.state !== 'closed') {
                    Promise.resolve(ctx.close()).catch(() => {});
                }
            } catch (_) {}
        }

        window.addEventListener('pointerdown', event => {
            if (event.isTrusted && !document.hidden) prime();
        }, { capture: true, passive: true });
        window.addEventListener('blur', cancelPending);
        document.addEventListener('visibilitychange', () => {
            if (document.hidden) cancelPending();
        });
        window.addEventListener('pagehide', dispose);

        return { play, cancelPending };
    })();

    const SPEED_MULTIPLIER = 1.60;
    const SPEED_CHECK_MS = 500;
    // Exact-build speed controller. No object scans, timer replacement, or position writes.
    function createHazySpeedController(getModule, notify, multiplier = 1.6) {
        const OFF = Object.freeze({
            shooterClassSlot: 0x6972ec,
            className: 0x08,
            classSelf: 0x3c,
            classParent: 0x2c,
            classStaticFields: 0x5c,
            objectNative: 0x08,
            photonView: 0x10,
            photonIsMine: 0x40,
            controller: 0x74,
            currentWeapon: 0xe0,
            controllerShooter: 0x4c,
            rigidbody: 0xb0,
            groundAcceleration: 0x6c,
            groundLimit: 0x78,
            crouchGroundLimit: 0x7c,
            initialGroundLimit: 0x80,
            weaponIsLocal: 0x6c,
            weaponShooter: 0x70,
            weaponGroundMultiplier: 0x80
        });
        // Read-only registration/field-layout words from the uploaded WASM.
        const FINGERPRINT = [
            [0x47a768, 0x2a8ff], [0x47a76c, 0x12e1], [0x47a770, 0x6be9a0],
            [0x6805cc, 0x3c6750], [0x680e54, 0x3c9da0],
            [0x3c67d8, 0x78], [0x3c67e0, 0x80], [0x3c9e14, 0x74]
        ];
        const MANAGED = ['groundAcceleration', 'crouchGroundLimit', 'initialGroundLimit'];
        let wanted = false;
        let active = false;
        let saved = null;
        let lastMessage = '';
        let lastState = { wanted: false, active: false, code: 'off', message: 'Speed: Normal' };
        multiplier = Number(multiplier);
        if (!Number.isFinite(multiplier) || multiplier <= 1 || multiplier > 3) {
            throw new RangeError('Speed multiplier must be above 1 and at most 3.');
        }
        const close = (a, b) => Number.isFinite(a) && Number.isFinite(b) &&
            Math.abs(a - b) <= Math.max(0.00001, Math.abs(b) * 0.00001);

        function publish(code, message, extra = {}) {
            lastState = { wanted, active, code, message, ...extra };
            const signature = `${wanted}|${active}|${code}|${message}`;
            if (signature !== lastMessage) {
                lastMessage = signature;
                if (typeof notify === 'function') notify({ ...lastState });
            }
        }
        function viewOf(module) {
            const bytes = module && (module.HEAPU8 || module.HEAP8);
            if (bytes && bytes.buffer && bytes.byteLength >= 0x700000) {
                return new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
            }
            const buffer = module && module.wasmMemory && module.wasmMemory.buffer;
            return buffer && buffer.byteLength >= 0x700000 ? new DataView(buffer) : null;
        }
        function valid(view, address, size = 4) {
            return Number.isInteger(address) && address >= 1024 && address % 4 === 0 &&
                address + size <= view.byteLength;
        }
        const uint = (view, address) => valid(view, address) ? view.getUint32(address, true) : 0;
        function asciiEquals(view, address, text) {
            if (!Number.isInteger(address) || address < 1024 ||
                address + text.length >= view.byteLength) return false;
            for (let i = 0; i < text.length; i++) {
                if (view.getUint8(address + i) !== text.charCodeAt(i)) return false;
            }
            return view.getUint8(address + text.length) === 0;
        }
        function classNamed(view, klass, name) {
            return valid(view, klass, 0x60) && uint(view, klass + OFF.classSelf) === klass &&
                asciiEquals(view, uint(view, klass + OFF.className), name);
        }
        function objectNamed(view, object, name, size) {
            return valid(view, object, size) && uint(view, object + OFF.objectNative) !== 0 &&
                classNamed(view, uint(view, object), name);
        }
        function derivesFrom(view, klass, name) {
            for (let depth = 0; depth < 12 && valid(view, klass, 0x60); depth++) {
                if (classNamed(view, klass, name)) return true;
                const next = uint(view, klass + OFF.classParent);
                if (next === klass) return false;
                klass = next;
            }
            return false;
        }
        function matchingBuild(module, view) {
            if (typeof module.productVersion === 'string' && module.productVersion !== '4.24') {
                return false;
            }
            return FINGERPRINT.every(([address, value]) => uint(view, address) === value);
        }
        function weaponMultiplier(view, shooter) {
            const weapon = uint(view, shooter + OFF.currentWeapon);
            if (!valid(view, weapon, 0x88) || !uint(view, weapon + OFF.objectNative) ||
                !derivesFrom(view, uint(view, weapon), 'Weapon') ||
                view.getUint8(weapon + OFF.weaponIsLocal) !== 1 ||
                uint(view, weapon + OFF.weaponShooter) !== shooter) return null;
            const value = view.getFloat32(weapon + OFF.weaponGroundMultiplier, true);
            return Number.isFinite(value) && value > 0 && value <= 5 ? value : null;
        }
        function resolve(module, view, needWeapon = true) {
            const shooterClass = uint(view, OFF.shooterClassSlot);
            if (!classNamed(view, shooterClass, 'Shooter')) return null;
            const statics = uint(view, shooterClass + OFF.classStaticFields);
            const shooter = uint(view, statics);
            if (!objectNamed(view, shooter, 'Shooter', 0x208) ||
                uint(view, shooter) !== shooterClass) return null;
            const photon = uint(view, shooter + OFF.photonView);
            if (!objectNamed(view, photon, 'PhotonView', 0x44) ||
                view.getUint8(photon + OFF.photonIsMine) !== 1) return null;
            const controller = uint(view, shooter + OFF.controller);
            if (!objectNamed(view, controller, 'PlayerController', 0x12c) ||
                uint(view, controller + OFF.controllerShooter) !== shooter) return null;
            const body = uint(view, controller + OFF.rigidbody);
            if (!objectNamed(view, body, 'Rigidbody', 0x0c)) return null;
            const weaponScale = weaponMultiplier(view, shooter);
            if (weaponScale === null && needWeapon) return null;
            return {
                module, view, shooter, controller, weaponScale,
                shooterNative: uint(view, shooter + OFF.objectNative),
                controllerNative: uint(view, controller + OFF.objectNative)
            };
        }
        function samePlayer(a, b) {
            return a && b && a.module === b.module && a.shooter === b.shooter &&
                a.controller === b.controller && a.shooterNative === b.shooterNative &&
                a.controllerNative === b.controllerNative;
        }
        function read(ctx, field) {
            return ctx.view.getFloat32(ctx.controller + OFF[field], true);
        }
        function write(ctx, field, value) {
            ctx.view.setFloat32(ctx.controller + OFF[field], Math.fround(value), true);
            if (!close(read(ctx, field), Math.fround(value))) {
                throw new Error('Speed value did not read back correctly.');
            }
        }
        function plausible(value, field) {
            const maximum = field === 'groundAcceleration' ? 5000 : 200;
            return Number.isFinite(value) && value >= 0 && value <= maximum &&
                (field === 'crouchGroundLimit' || value > 0);
        }
        function capture(ctx) {
            const original = {};
            for (const field of [...MANAGED, 'groundLimit']) {
                original[field] = read(ctx, field);
                if (!plausible(original[field], field)) {
                    throw new Error('Movement values are outside the expected range.');
                }
            }
            // SetWeapon / SetSpeedMultis establish this relationship in this build.
            if (!close(original.groundLimit, original.initialGroundLimit * ctx.weaponScale)) {
                return null; // Let a spawn or weapon-switch transition finish first.
            }
            return { ...ctx, original };
        }
        function restore(record) {
            if (!record) return true;
            const view = viewOf(record.module);
            if (!view || !matchingBuild(record.module, view)) return false;
            const now = resolve(record.module, view, false);
            if (!samePlayer(record, now)) return false;
            const base = read(now, 'initialGroundLimit');
            const ourBase = Math.fround(record.original.initialGroundLimit * multiplier);
            // Only restore values that still equal values written by this controller.
            for (const field of MANAGED) {
                if (close(read(now, field), Math.fround(record.original[field] * multiplier))) {
                    write(now, field, record.original[field]);
                }
            }
            if (now.weaponScale !== null) {
                if (close(base, ourBase) || close(base, record.original.initialGroundLimit)) {
                    const expectedBoosted = Math.fround(ourBase * now.weaponScale);
                    if (close(read(now, 'groundLimit'), expectedBoosted)) {
                        write(now, 'groundLimit', record.original.initialGroundLimit * now.weaponScale);
                    }
                }
            } else if (close(base, ourBase)) {
                // The weapon can briefly be absent during a switch. Undo our scale
                // without retaining a stale weapon pointer or preventing deactivation.
                const current = read(now, 'groundLimit');
                if (Number.isFinite(current) && current >= 0 && current <= 600) {
                    write(now, 'groundLimit', current / multiplier);
                }
            }
            return true;
        }
        function apply(ctx) {
            // Never multiply an already-boosted value again.
            for (const field of MANAGED) {
                const current = read(ctx, field);
                const original = saved.original[field];
                const target = Math.fround(original * multiplier);
                if (!close(current, original) && !close(current, target)) {
                    throw new Error('Movement settings changed; switch Speed Boost on again.');
                }
            }
            for (const field of MANAGED) {
                const target = Math.fround(saved.original[field] * multiplier);
                if (!close(read(ctx, field), target)) write(ctx, field, target);
            }
            const normalLimit = saved.original.initialGroundLimit * ctx.weaponScale;
            const targetLimit = Math.fround(Math.fround(saved.original.initialGroundLimit * multiplier) * ctx.weaponScale);
            if (!close(read(ctx, 'groundLimit'), targetLimit)) write(ctx, 'groundLimit', targetLimit);
            active = true;
            publish('written', `Speed: limit ${normalLimit.toFixed(2)} → ${targetLimit.toFixed(2)}`, {
                normalLimit, writtenLimit: read(ctx, 'groundLimit'), multiplier
            });
        }
        function tick() {
            if (!wanted) return;
            try {
                const module = getModule();
                const view = viewOf(module);
                if (!view) {
                    active = false;
                    publish('waiting-runtime', 'Speed: waiting for game');
                    return;
                }
                if (!matchingBuild(module, view)) {
                    wanted = false; active = false; saved = null;
                    publish('different-build', 'Speed: different build — not applied');
                    return;
                }
                const ctx = resolve(module, view);
                if (!ctx) {
                    // Keep the saved record until a player is resolved again; do not
                    // snapshot boosted values as a new baseline during weapon switches.
                    active = false;
                    publish('waiting-player', 'Speed: waiting for your player');
                    return;
                }
                if (!samePlayer(saved, ctx)) {
                    saved = capture(ctx);
                    if (!saved) {
                        active = false;
                        publish('waiting-values', 'Speed: waiting for movement setup');
                        return;
                    }
                }
                apply(ctx);
            } catch (error) {
                try { restore(saved); } catch (_) { /* Do not continue unsafe writes. */ }
                saved = null; wanted = false; active = false;
                publish('error', `Speed: ${error instanceof Error ? error.message : String(error)}`);
            }
        }
        function setEnabled(enabled) {
            enabled = Boolean(enabled);
            if (enabled === wanted) return;
            if (enabled) {
                wanted = true;
                publish('enabled', 'Speed: Enabled');
                tick();
            } else {
                wanted = false; active = false;
                let restored = true;
                try { restored = restore(saved); } catch (_) { restored = false; }
                saved = null;
                publish('off', restored ? 'Speed: Normal' : 'Speed: Off — old player unavailable');
            }
        }
        return {
            tick, setEnabled,
            getPlayer() {
                const module = getModule();
                const view = viewOf(module);
                return view && matchingBuild(module, view)
                    ? resolve(module, view, false) : null;
            },
            get wanted() { return wanted; },
            get active() { return active; },
            get state() { return { ...lastState }; }
        };
    }

    const speedEngine = createHazySpeedController(
        () => window.unityInstance?.Module || window.Module || null,
        state => {
            const before = speedBoost || speedBoostWanted;
            speedBoost = state.active;
            speedBoostWanted = state.wanted && !state.active;
            if (before !== (speedBoost || speedBoostWanted)) renderActiveMods();
        },
        SPEED_MULTIPLIER
    );

        speedEngine.setEnabled(!speedEngine.wanted);

    // Cheap pointer/field checks only. Unchanged values are NOT rewritten.
    // There is no speed work in the requestAnimationFrame loop.
    window.setInterval(() => speedEngine.tick(), SPEED_CHECK_MS);
    window.addEventListener('pagehide', () => speedEngine.setEnabled(false));

    const HIGH_JUMP_MULTIPLIER = 2.50;

    function createHighJumpController(getPlayer, notify, multiplier = 2.5) {
        // PlayerController.jumpHeight (float32) in the uploaded 4.24 build.
        // The shared player resolver already checks the build and ownership.
        const JUMP_HEIGHT_OFFSET = 0x9c;
        const JUMP_HEIGHT_LAYOUT_WORD = 0x3c67fc;
        // Keep the boosted value within the existing Air Jump validation range.
        const MAX_JUMP_STRENGTH = 100;

        multiplier = Number(multiplier);
        if (!Number.isFinite(multiplier) || multiplier <= 1 || multiplier > 5) {
            throw new RangeError('High Jump multiplier must be above 1 and at most 5.');
        }

        let wanted = false;
        let saved = null;
        let errorReported = false;

        const close = (a, b) => Number.isFinite(a) && Number.isFinite(b) &&
            Math.abs(a - b) <= Math.max(0.00001, Math.abs(b) * 0.00001);

        function samePlayer(a, b) {
            return Boolean(a && b && a.module === b.module &&
                a.shooter === b.shooter && a.controller === b.controller &&
                a.shooterNative === b.shooterNative &&
                a.controllerNative === b.controllerNative);
        }

        function resolve() {
            const ctx = getPlayer();
            if (!ctx) return null;
            const view = ctx.view;
            const address = ctx.controller + JUMP_HEIGHT_OFFSET;
            if (!view || view.byteLength < JUMP_HEIGHT_LAYOUT_WORD + 4 ||
                !Number.isInteger(address) || ctx.controller < 1024 ||
                address % 4 !== 0 || address + 4 > view.byteLength ||
                view.getUint32(JUMP_HEIGHT_LAYOUT_WORD, true) !== JUMP_HEIGHT_OFFSET) {
                throw new Error('High Jump: different jump layout; not applied.');
            }
            return ctx;
        }

        function read(ctx) {
            return ctx.view.getFloat32(ctx.controller + JUMP_HEIGHT_OFFSET, true);
        }

        function write(ctx, value) {
            value = Math.fround(value);
            if (!Number.isFinite(value) || value <= 0 || value > MAX_JUMP_STRENGTH) {
                throw new Error('High Jump: unexpected jump strength; not applied.');
            }
            // No write when the game already has the desired value.
            if (close(read(ctx), value)) return;
            ctx.view.setFloat32(ctx.controller + JUMP_HEIGHT_OFFSET, value, true);
            if (!close(read(ctx), value)) {
                throw new Error('High Jump: jump strength did not read back correctly.');
            }
        }

        function snapshot(ctx, original) {
            // Save identity, not a DataView: the game's heap can grow later.
            return {
                module: ctx.module,
                shooter: ctx.shooter,
                controller: ctx.controller,
                shooterNative: ctx.shooterNative,
                controllerNative: ctx.controllerNative,
                original,
                boosted: Math.fround(Math.min(original * multiplier, MAX_JUMP_STRENGTH)),
                wrote: false
            };
        }

        function restore(ctx) {
            if (!saved) return;
            if (samePlayer(saved, ctx) && saved.wrote && close(read(ctx), saved.boosted)) {
                // Do not overwrite a newer value set by the game or another mod.
                write(ctx, saved.original);
            }
            saved = null;
        }

        function tick() {
            if (!wanted && !saved) return;
            try {
                const ctx = resolve();
                if (!ctx) {
                    // Lobby, respawn, or missing player: no guessed memory access.
                    // Keep a pending restore until the same player is reachable again.
                    return;
                }
                if (saved && !samePlayer(saved, ctx)) {
                    saved = null;
                }
                if (!wanted) {
                    restore(ctx);
                    return;
                }

                const current = read(ctx);
                if (current === 0) return; // Let player initialization finish.
                if (!Number.isFinite(current) || current < 0 || current > MAX_JUMP_STRENGTH) {
                    throw new Error('High Jump: unexpected jump strength; not applied.');
                }

                if (!saved) {
                    saved = snapshot(ctx, current);
                } else if (!close(current, saved.boosted) && !close(current, saved.original)) {
                    // A class/round change can set a new normal jump strength.
                    // Use that fresh baseline instead of restoring stale settings.
                    saved = snapshot(ctx, current);
                }

                write(ctx, saved.boosted);
                saved.wrote = true;
            } catch (error) {
                wanted = false;
                try {
                    const ctx = resolve();
                    if (ctx) restore(ctx);
                } catch (_) {
                    // An incompatible layout must never be used for a restore write.
                    saved = null;
                }
                if (typeof notify === 'function') notify(false);
                if (!errorReported) {
                    errorReported = true;
                    console.warn('[Hazy Client] High Jump stopped:', error);
                }
            }
        }

        function setEnabled(enabled) {
            enabled = Boolean(enabled);
            if (enabled === wanted) return;
            wanted = enabled;
            errorReported = false;
            // Button/Active Mods respond immediately, including in the lobby.
            if (typeof notify === 'function') notify(wanted);
            tick();
        }

        return {
            tick,
            setEnabled,
            get wanted() { return wanted; }
        };
    }

    function readExtraMovementVelocity(ctx) {
        const p = ctx.controller + 0xb4;
        ctx.getVelocity(p, ctx.body, 0);
        const v = currentMemoryView(ctx.module);
        const result = { x: v.getFloat32(p, true), y: v.getFloat32(p + 4, true),
            z: v.getFloat32(p + 8, true) };
        if (!Object.values(result).every(x => Number.isFinite(x) && Math.abs(x) <= 200)) {
            throw new Error('Unexpected movement velocity; no extra force applied.');

    function extraMovementDirection(ctx, view, velocity) {
        let x = view.getFloat32(ctx.controller + 0xc0, true);
        let z = view.getFloat32(ctx.controller + 0xc8, true);
        if (![x, z].every(n => Number.isFinite(n) && Math.abs(n) <= 2.5)) {
            throw new Error('Unexpected movement direction.');
        }
        let length = Math.hypot(x, z);
        if (length < 0.05) {
            // If coasting, dash along that movement rather than an unrelated axis.
            x = velocity.x; z = velocity.z; length = Math.hypot(x, z);
            if (length < 0.25) {
                const runtime = findSceneRuntime();
                if (!runtime || runtime.table !== ctx.table) return null;
                const camera = infoUint(view, ctx.controller + 0x64);
                if (!infoObjectNamed(view, camera, 'Camera', 12)) return null;
                const transform = sceneFunction(runtime, 2964)(camera, 0);
                if (!infoObjectNamed(sceneView(runtime), transform, 'Transform', 12)) return null;
                const scratch = sceneScratchPointer(runtime);
                sceneFunction(runtime, 4579)(scratch, transform, 0);
                const fresh = sceneView(runtime);
                x = fresh.getFloat32(scratch, true);
                z = fresh.getFloat32(scratch + 8, true);
                length = Math.hypot(x, z);
            }
        }
        if (!Number.isFinite(length) || length < 0.00001) return null;
        return { x: x / length, z: z / length };
    }
    function captureExtraMovement(ctx) {
        if (!extraMovementWanted() || fly || noClipFly || !extraMovementInputActive()) {
            dashRequested = false;
            if (extraMovementState) {
                extraMovementState.resumeBaseline = true;
                extraMovementState.dashRemaining = 0;
            }
            return null;
        }
        const view = currentMemoryView(ctx.module);
        if (!view || !EXTRA_MOVEMENT_CHECKS.every(([p, n]) =>
            p + 4 <= view.byteLength && view.getUint32(p, true) === n)) {
            throw new Error('Movement extras do not match this game build.');
        }
        const hp = infoUint(view, ctx.shooter + 0x6c);
        if (!infoObjectNamed(view, hp, 'Health', 0xb8) ||
            view.getUint8(hp + 0x80) || view.getUint8(hp + 0xa8)) {
            extraMovementState = null; dashRequested = false; return null;
        }
        const getStep = ctx.table.get(40956);
        if (typeof getStep !== 'function') throw new Error('Physics timestep unavailable.');
        const dt = getStep(0);
        if (!Number.isFinite(dt) || dt <= 0 || dt > 0.1) {
            throw new Error('Unexpected physics timestep.');
        }
        const gravity = view.getFloat32(ctx.controller + 0x94, true);
        const limit = view.getFloat32(ctx.controller + 0x78, true);
        const lastJump = view.getFloat32(ctx.controller + 0x120, true);
        if (!Number.isFinite(gravity) || gravity < 0 || gravity > 500 ||
            !Number.isFinite(limit) || limit <= 0 || limit > 200 ||
            !Number.isFinite(lastJump)) throw new Error('Unexpected movement settings.');
        if (!sameExtraMovementPlayer(extraMovementState, ctx)) {
            if (extraMovementState) dashRequested = false; // No dash queued before a respawn.
            extraMovementState = { module: ctx.module, table: ctx.table,
                controller: ctx.controller, controllerNative: ctx.controllerNative,
                body: ctx.body, bodyNative: ctx.bodyNative,
                lastJump, dashRemaining: 0, cooldownRemaining: 0, dashDirection: null,
                resumeBaseline: false };
        }
        const state = extraMovementState;
        if (state.resumeBaseline) { state.lastJump = lastJump; state.resumeBaseline = false; }
        const velocity = readExtraMovementVelocity(ctx);
        const direction = extraMovementDirection(ctx, currentMemoryView(ctx.module), velocity);
        const thrust = jetpack && extraMovementKeys.has('Space');
        if (thrust) currentMemoryView(ctx.module).setUint8(ctx.controller + 0xe5, 0);
        return { ctx, state, dt, gravity, limit, lastJump, direction, thrust };
    }

    // Pure velocity calculation: game inputs/physics remain outside this helper.
    function calculateExtraMovement(sample, postVelocity, postJumpTime) {
        const { state, dt, gravity, limit, direction } = sample;
        const next = { ...postVelocity };
        let changed = false;
        state.cooldownRemaining = Math.max(0, state.cooldownRemaining - dt);
        if (longJump && postJumpTime !== state.lastJump && !sample.thrust && direction) {
            next.x += direction.x * LONG_JUMP_FORWARD_BOOST;
            next.z += direction.z * LONG_JUMP_FORWARD_BOOST;
            const speed = Math.hypot(next.x, next.z);
            if (speed > LONG_JUMP_MAX_SPEED) {
                next.x *= LONG_JUMP_MAX_SPEED / speed;
                next.z *= LONG_JUMP_MAX_SPEED / speed;
            }
            changed = true;
        }
        // Native Jump updates lastJumpTime; walking off a ledge does not.
        state.lastJump = postJumpTime;
        if (dashRequested) {
            dashRequested = false;
            if (dash && state.cooldownRemaining === 0 && direction) {
                state.dashDirection = direction;
                state.dashRemaining = DASH_DURATION;
                state.cooldownRemaining = DASH_COOLDOWN;
            }
        }
        if (dash && state.dashRemaining > 0 && state.dashDirection) {
            next.x = state.dashDirection.x * DASH_SPEED;
            next.z = state.dashDirection.z * DASH_SPEED;
            state.dashRemaining = Math.max(0, state.dashRemaining - dt);
            state.finishDash = state.dashRemaining === 0;
            changed = true;
        } else if (state.finishDash) {
            // End the burst at ordinary horizontal speed, retaining Y/gravity.
            const speed = Math.hypot(next.x, next.z);
            if (speed > limit) { next.x *= limit / speed; next.z *= limit / speed; changed = true; }
            state.finishDash = false;
        }
        if (sample.thrust && next.y < JETPACK_MAX_UP_SPEED) {
            // Compensate Kour's downward acceleration, then add gentle net lift.
            // Neither native gravity nor collision is switched off.
            next.y = Math.min(JETPACK_MAX_UP_SPEED,
                next.y + (gravity + JETPACK_LIFT_ACCELERATION) * dt);
            changed = true;
        }
        return changed ? next : null;
    }
    function applyExtraMovement(ctx, sample) {
        if (!ctx || !sample || !sameExtraMovementPlayer(sample.state, ctx) ||
            ctx.isKinematic(ctx.body, 0) || !extraMovementInputActive() || fly || noClipFly) return;
        const postVelocity = readExtraMovementVelocity(ctx);
        const lastJump = currentMemoryView(ctx.module).getFloat32(ctx.controller + 0x120, true);
        if (!Number.isFinite(lastJump)) throw new Error('Invalid jump timestamp.');
        const next = calculateExtraMovement(sample, postVelocity, lastJump);
        if (!next) return;
        if (!Object.values(next).every(v => Number.isFinite(v) && Math.abs(v) <= 200)) {
            throw new Error('Extra movement produced an invalid velocity.');
        }
        const p = ctx.controller + 0xb4;
        const view = currentMemoryView(ctx.module);
        view.setFloat32(p, next.x, true); view.setFloat32(p + 4, next.y, true);
        view.setFloat32(p + 8, next.z, true);
        ctx.setVelocity(ctx.body, p, 0);
    }

    function createFlyController(getPlayer, getTable, getControls, notify) {
        // Exact signatures and table slots from the user's exported 4.24 build.

        // void(Rigidbody, Vector3*, MethodInfo).

        const GET_VELOCITY_SLOT = 45204;
        const SET_VELOCITY_SLOT = 45205;
        const GET_GRAVITY_SLOT = 45209;
        const SET_GRAVITY_SLOT = 45210;
        const IS_KINEMATIC_SLOT = 45211;

        const CHECKS = [
            [0x6bea80, FIXED_UPDATE_SLOT],
            [0x6dec38, GET_VELOCITY_SLOT],
            [0x6dec3c, SET_VELOCITY_SLOT],
            [0x6dec4c, GET_GRAVITY_SLOT],

        function samePlayer(a, b) {

                a.shooterNative === b.shooterNative &&
                a.controllerNative === b.controllerNative &&

        function resolve() {
            const ctx = getPlayer();
            if (!ctx) return null;
            const view = currentMemoryView(ctx.module);
            if (!view) return null;
            if (!CHECKS.every(([address, value]) => address + 4 <= view.byteLength &&
                view.getUint32(address, true) === value)) {
                throw new Error('Fly: different movement layout; not applied.');
            }
            const body = view.getUint32(ctx.controller + 0xb0, true);
            if (body < 1024 || body % 4 || body + 12 > view.byteLength) return null;
            const bodyNative = view.getUint32(body + 8, true);
            if (!bodyNative) return null;
            const table = getTable({ ...ctx, view });
            if (!table || table.length <= IS_KINEMATIC_SLOT) {
                throw new Error('Fly: save the script and refresh Kour to capture its functions.');
            }
            const getVelocity = table.get(GET_VELOCITY_SLOT);
            const setVelocity = table.get(SET_VELOCITY_SLOT);
            const getGravity = table.get(GET_GRAVITY_SLOT);
            const setGravity = table.get(SET_GRAVITY_SLOT);
            const isKinematic = table.get(IS_KINEMATIC_SLOT);
            if (![getVelocity, setVelocity, getGravity, setGravity, isKinematic]
                .every(fn => typeof fn === 'function')) {
                throw new Error('Fly: expected physics functions unavailable.');
            }
            return { ...ctx, view, body, bodyNative, table,
                getVelocity, setVelocity, getGravity, setGravity, isKinematic };
        }

        // NoClip uses the local controller's physical capsule only. These
        // fields and native wrappers were identified in the uploaded 4.24 files.
        // No global collision layers, map colliders, or other players are changed.
        const COLLIDER_OFFSET = 0x3c;
        const COLLIDER_GET_ENABLED = 3170;
        const COLLIDER_SET_ENABLED = 45052;
        const COLLIDER_GET_BODY = 45053;
        const COLLIDER_CHECKS = [
            [0x3c6780, COLLIDER_OFFSET],
            [0x6de9c0, COLLIDER_GET_ENABLED],
            [0x6de9c4, COLLIDER_SET_ENABLED],
            [0x6de9c8, COLLIDER_GET_BODY]
        ];

        function validPointer(view, pointer, bytes = 4) {
            return Number.isInteger(pointer) && pointer >= 1024 &&
                pointer % 4 === 0 && pointer + bytes <= view.byteLength;
        }

        function objectHasClass(view, object, name) {
            if (!validPointer(view, object, 12)) return false;
            const klass = view.getUint32(object, true);
            if (!validPointer(view, klass, 0x60) ||
                view.getUint32(klass + 0x3c, true) !== klass) return false;
            const address = view.getUint32(klass + 8, true);
            if (address < 1024 || address + name.length >= view.byteLength) return false;
            for (let i = 0; i < name.length; i++) {
                if (view.getUint8(address + i) !== name.charCodeAt(i)) return false;
            }
            return view.getUint8(address + name.length) === 0;
        }

        function resolveCollider(ctx, existing = null) {
            const view = currentMemoryView(ctx.module);
            if (!view) return null;
            if (!COLLIDER_CHECKS.every(([address, value]) => address + 4 <= view.byteLength &&
                view.getUint32(address, true) === value)) {
                throw new Error('NoClip Fly: different collider layout; not applied.');
            }
            const collider = existing ? existing.collider
                : view.getUint32(ctx.controller + COLLIDER_OFFSET, true);
            if (!objectHasClass(view, collider, 'CapsuleCollider')) return null;
            const colliderNative = view.getUint32(collider + 8, true);
            if (!colliderNative || (existing && colliderNative !== existing.colliderNative)) {
                return null;
            }
            const getEnabled = ctx.table.get(COLLIDER_GET_ENABLED);
            const setEnabled = ctx.table.get(COLLIDER_SET_ENABLED);
            const getBody = ctx.table.get(COLLIDER_GET_BODY);
            if (![getEnabled, setEnabled, getBody].every(fn => typeof fn === 'function')) {
                throw new Error('NoClip Fly: collider functions unavailable.');
            }
            // Verify the capsule belongs to this player's rigidbody, not a
            // similarly named collider on a different character or scene object.
            const attached = getBody(collider, 0);
            const ownerMatches = objectHasClass(view, attached, 'Rigidbody') &&
                view.getUint32(attached + 8, true) === ctx.bodyNative;
            // A disabled collider can be temporarily unavailable to native queries.
            // Reuse ONLY this exact capsule's previously verified owner association.
            // A non-null, different rigidbody is never accepted.
            const prior = existing || collisionRecord;
            const knownDisabled = !attached && getEnabled(collider, 0) === 0 &&
                prior && prior.attachmentVerified && samePlayer(prior, ctx) &&
                prior.collider === collider && prior.colliderNative === colliderNative;
            if (!ownerMatches && !knownDisabled) return null;
            return { ...ctx, collider, colliderNative, getEnabled, setEnabled,
                attachmentVerified: true };
        }

        function readColliderEnabled(ctx) {
            const value = ctx.getEnabled(ctx.collider, 0);
            if (value !== 0 && value !== 1) {
                throw new Error('NoClip Fly: unexpected collider state.');
            }
            return value;
        }

        function setColliderEnabled(ctx, enabled) {
            if (readColliderEnabled(ctx) === enabled) return;
            ctx.setEnabled(ctx.collider, enabled, 0);
            if (readColliderEnabled(ctx) !== enabled) {
                throw new Error('NoClip Fly: collider state did not read back correctly.');
            }
        }

        function restoreCollisions(ctx) {
            if (!collisionRecord) return;
            if (!samePlayer(collisionRecord, ctx)) {
                collisionRecord = null;
                return;
            }
            // Revalidate a saved capsule before touching it, even if the game
            // replaced the controller's current capsule during the same life.
            const savedCollider = resolveCollider(ctx, collisionRecord);
            if (savedCollider && readColliderEnabled(savedCollider) === 0) {
                setColliderEnabled(savedCollider, collisionRecord.originalEnabled);
            }
            collisionRecord = null;
        }

        function updateCollisionMode(ctx) {
            if (!noclipWanted) {
                restoreCollisions(ctx);
                return;
            }
            const current = resolveCollider(ctx);
            if (!current) {
                throw new Error('NoClip Fly: local physical capsule unavailable.');
            }
            if (collisionRecord && (!samePlayer(collisionRecord, current) ||
                collisionRecord.collider !== current.collider ||
                collisionRecord.colliderNative !== current.colliderNative)) {
                restoreCollisions(ctx);
            }
            if (!collisionRecord) {
                collisionRecord = { ...current,
                    originalEnabled: readColliderEnabled(current) };
            }
            setColliderEnabled(current, 0);
        }

        function readGravity(ctx) {
            const value = ctx.getGravity(ctx.body, 0);
            if (value !== 0 && value !== 1) {
                throw new Error('Fly: unexpected gravity setting.');
            }
            return value;
        }

        function setGravity(ctx, enabled) {
            if (readGravity(ctx) === enabled) return;
            ctx.setGravity(ctx.body, enabled, 0);
            if (readGravity(ctx) !== enabled) {
                throw new Error('Fly: gravity setting did not read back correctly.');
            }
        }

        function writeVelocity(ctx, x, y, z) {
            if (![x, y, z].every(value => Number.isFinite(value) && Math.abs(value) <= 200)) {
                throw new Error('Fly: invalid flight velocity; not applied.');
            }
            // Use the verified cached velocity as the native Vector3 argument.
            // Writing only this cache would NOT move the physics body.
            const view = currentMemoryView(ctx.module);
            const address = ctx.controller + 0xb4;
            view.setFloat32(address, x, true);
            view.setFloat32(address + 4, y, true);
            view.setFloat32(address + 8, z, true);
            ctx.setVelocity(ctx.body, address, 0);
        }

        function stopVerticalMotion(ctx) {
            if (ctx.isKinematic(ctx.body, 0)) return;
            const address = ctx.controller + 0xb4;
            ctx.getVelocity(address, ctx.body, 0);
            const view = currentMemoryView(ctx.module);
            const x = view.getFloat32(address, true);
            const z = view.getFloat32(address + 8, true);
            if ([x, z].every(value => Number.isFinite(value) && Math.abs(value) <= 200)) {
                writeVelocity(ctx, x, 0, z);
            }
        }

        function restorePlayer(ctx) {
            if (!ctx) return;
            // Restore collision state before returning to normal movement.
            // Still attempt gravity cleanup even if collider restoration fails.
            let collisionError = null;
            try { restoreCollisions(ctx); } catch (error) { collisionError = error; }
            if (!record) {
                if (collisionError) throw collisionError;
                return;
            }
            if (!samePlayer(record, ctx)) {
                // A replaced/destroyed player must never receive stale writes.
                record = null;
                return;
            }
            // Restore native gravity only if it is still in the state Fly set.
            if (readGravity(ctx) === 0) setGravity(ctx, record.originalGravity);
            // Do not restore a minutes-old falling/upward velocity.
            stopVerticalMotion(ctx);
            currentMemoryView(ctx.module).setUint8(ctx.controller + 0xe5, 0);
            record = null;
            if (collisionError) throw collisionError;
        }

        // Keep one forwarding wrapper installed for each game table. Some runtimes
        // cache table.get() results: changing the table after that first lookup does
        // not replace the cached function. Toggling now changes state, not the entry.
        function prepareRuntime(table, memory) {
            if (!table || typeof table.get !== 'function' || !memory?.buffer ||
                memory.buffer.byteLength < 0x700000) return false;
            const view = new DataView(memory.buffer);
            const build = [
                [0x47a768, 0x2a8ff], [0x47a76c, 0x12e1],
                [0x47a770, 0x6be9a0], [0x6805cc, 0x3c6750],
                [0x680e54, 0x3c9da0], [0x3c67d8, 0x78],
                [0x3c67e0, 0x80], [0x3c9e14, 0x74],
                [0x6beab0, SPIDER_COLLISION_STAY_SLOT], ...CHECKS
            ];
            if (!build.every(([address, value]) => address + 4 <= view.byteLength &&
                view.getUint32(address, true) === value)) return false;
            // Install both forwarding hooks before Unity starts caching callbacks.
            // They remain inert and forward directly to Kour while their mods are off.
            installSpiderCollisionHook({ table, view });
            installHook({ table });
            return true;
        }

        function fail(error) {
            wanted = false;
            noclipWanted = false;
            lastFlightAt = null;
            waitingSince = null;
            // The permanent wrapper becomes an inert call to the original update.
            try {
                const ctx = resolve();
                if (ctx) restorePlayer(ctx);
            } catch (_) {
                // A build mismatch is not a reason to guess a restore address.
                record = null;
            }
            notify(false, false);
            if (!errorReported) {
                errorReported = true;
                console.warn('[Hazy Client] Fly stopped:', error);
            }
        }

        function failSpider(error) {
            spider = false;
            spiderContactSince = null;
            spiderCollisionAt = -Infinity;
            renderActiveMods();
            if (!spiderErrorReported) {
                spiderErrorReported = true;
                console.warn('[Hazy Client] Spider stopped:', error);
            }
        }

        function installSpiderCollisionHook(ctx) {
            const table = ctx?.table;
            if (!table || typeof table.get !== 'function' ||
                table.length <= SPIDER_COLLISION_STAY_SLOT) {
                return false;
            }

            const view = ctx.view || currentMemoryView(ctx.module);
            if (!view || view.byteLength <= 0x6beab4 ||
                view.getUint32(0x6beab0, true) !== SPIDER_COLLISION_STAY_SLOT) {
                throw new Error('Spider: OnCollisionStay does not match this Kour build.');
            }

            const existing = spiderCollisionHooksByTable.get(table);
            if (existing) {
                if (table.get(SPIDER_COLLISION_STAY_SLOT) !== existing.wrapper) {
                    throw new Error('Spider: another script changed OnCollisionStay.');
                }
                return true;
            }

            const original = table.get(SPIDER_COLLISION_STAY_SLOT);
            if (typeof original !== 'function') {
                throw new Error('Spider: OnCollisionStay is unavailable.');
            }

            // Typed WASM trampoline: (controller, Collision, MethodInfo) -> void.
            const shimBytes = new Uint8Array([
                0,97,115,109,1,0,0,0,
                1,7,1,96,3,127,127,127,0,
                2,13,1,4,104,97,122,121,4,115,116,101,112,0,0,
                3,2,1,0,
                7,8,1,4,115,116,101,112,0,1,
                10,12,1,10,0,32,0,32,1,32,2,16,0,11
            ]);

            const collisionStep = (controller, collision, methodInfo) => {
                // Kour handles the collision first; Spider only observes it.
                original(controller, collision, methodInfo);

                if (!spider || wanted || !collision) return;

                try {
                    const local = getPlayer();
                    if (!local || local.controller !== controller) return;

                    const fresh = currentMemoryView(local.module);
                    if (!fresh || fresh.byteLength <= controller + 0xe5) return;
                    if (fresh.getUint32(0x3c6834, true) !== 0xe4) return;

                    // This is a real Unity collision callback on our own player.
                    // Ground collisions are filtered later by the sustained
                    // blocked-forward-motion test, so Spider can start from the floor.
                    spiderCollisionAt = clock();
                } catch (error) {
                    failSpider(error);
                }
            };

            const shimModule = new WebAssembly.Module(shimBytes);
            const shim = new WebAssembly.Instance(shimModule, {
                hazy: { step: collisionStep }
            });
            const wrapper = shim.exports.step;

            table.set(SPIDER_COLLISION_STAY_SLOT, wrapper);
            if (table.get(SPIDER_COLLISION_STAY_SLOT) !== wrapper) {
                throw new Error('Spider: collision hook installation failed.');
            }

            spiderCollisionHooksByTable.set(table, {
                table, original, wrapper, shim
            });
            return true;
        }

        function captureSpiderSample(ctx) {
            if (!spider || wanted || (jetpack && extraMovementKeys.has('Space')) ||
                !ctx || ctx.isKinematic(ctx.body, 0)) {
                spiderContactSince = null;
                return null;
            }

            const now = clock();
            if (now - spiderCollisionAt > SPIDER_COLLISION_WINDOW_MS) {
                spiderContactSince = null;
                return null;
            }

            const view = currentMemoryView(ctx.module);
            if (!view) {
                spiderContactSince = null;
                return null;
            }

            const layout = [
                [0x3c67d8, 0x78], // groundLimit
                [0x3c67e4, 0x84], // airLimit
                [0x3c6820, 0xb0], // Rigidbody
                [0x3c6824, 0xb4], // velocity cache
                [0x3c6828, 0xc0], // inputDir (world-space)
                [0x3c6834, 0xe4]  // onGround
            ];
            if (!layout.every(([address, value]) =>
                address + 4 <= view.byteLength &&
                view.getUint32(address, true) === value)) {
                throw new Error('Spider: movement layout does not match this Kour build.');
            }

            let inputX = view.getFloat32(ctx.controller + 0xc0, true);
            let inputZ = view.getFloat32(ctx.controller + 0xc8, true);
            if (![inputX, inputZ].every(value =>
                Number.isFinite(value) && Math.abs(value) <= 2.5)) {
                throw new Error('Spider: unexpected movement direction.');
            }

            const inputLength = Math.hypot(inputX, inputZ);
            if (inputLength < SPIDER_INPUT_MIN) {
                spiderContactSince = null;
                return null;
            }
            inputX /= inputLength;
            inputZ /= inputLength;

            // IMPORTANT: read the real Rigidbody velocity BEFORE Kour's current
            // FixedUpdate accelerates into the wall again. This is the collision-
            // resolved velocity from the previous physics step.
            const velocityAddress = ctx.controller + 0xb4;
            ctx.getVelocity(velocityAddress, ctx.body, 0);
            const fresh = currentMemoryView(ctx.module);
            const vx = fresh.getFloat32(velocityAddress, true);
            const vy = fresh.getFloat32(velocityAddress + 4, true);
            const vz = fresh.getFloat32(velocityAddress + 8, true);
            if (![vx, vy, vz].every(value =>
                Number.isFinite(value) && Math.abs(value) <= 250)) {
                throw new Error('Spider: unexpected player velocity.');
            }

            const grounded = fresh.getUint8(ctx.controller + 0xe4);
            if (grounded !== 0 && grounded !== 1) {
                throw new Error('Spider: unexpected grounded state.');
            }

            const speedLimit = fresh.getFloat32(
                ctx.controller + (grounded ? 0x78 : 0x84), true
            );
            if (!Number.isFinite(speedLimit) || speedLimit <= 0 || speedLimit > 200) {
                throw new Error('Spider: unexpected movement speed.');
            }

            const forwardSpeed = vx * inputX + vz * inputZ;
            const blockedThreshold = Math.max(
                0.65,
                Math.min(2.0, speedLimit * 0.22)
            );

            if (forwardSpeed >= blockedThreshold) {
                spiderContactSince = null;
                return null;
            }

            if (spiderContactSince === null) {
                spiderContactSince = now;
                return { ready: false, vx, vy, vz };
            }

            return {
                ready: now - spiderContactSince >= SPIDER_CONTACT_DELAY_MS,
                vx, vy, vz
            };
        }

        function applySpiderClimb(ctx, sample) {
            if (!sample?.ready || !spider || wanted ||
                !ctx || ctx.isKinematic(ctx.body, 0)) {
                return;
            }

            // Re-read Kour's post-FixedUpdate velocity so normal horizontal
            // acceleration/collision bookkeeping stays intact; change only Y.
            const address = ctx.controller + 0xb4;
            ctx.getVelocity(address, ctx.body, 0);
            const view = currentMemoryView(ctx.module);
            const x = view.getFloat32(address, true);
            const y = view.getFloat32(address + 4, true);
            const z = view.getFloat32(address + 8, true);

            if (![x, y, z].every(value =>
                Number.isFinite(value) && Math.abs(value) <= 250)) {
                throw new Error('Spider: invalid post-update velocity.');
            }

            // Keep gravity enabled. Reapply upward velocity each physics update
            // only while a real collision + sustained blocked movement continues.
            if (y < SPIDER_CLIMB_SPEED - 0.05) {
                writeVelocity(ctx, x, SPIDER_CLIMB_SPEED, z);
            }
        }

        function installHook(ctx) {
            const installed = hooksByTable.get(ctx.table);
            if (installed) {
                if (ctx.table.get(FIXED_UPDATE_SLOT) !== installed.wrapper) {
                    throw new Error('Fly: another script changed the movement hook.');
                }
                hook = installed;
                return;
            }
            const installedTable = ctx.table;
            const original = installedTable.get(FIXED_UPDATE_SLOT);
            if (typeof original !== 'function') {
                throw new Error('Fly: movement update function unavailable.');
            }

            // A tiny typed WASM trampoline: (i32, i32) -> void.
            // The exported WASM function is legal in the game's function table;
            // a plain JavaScript function would not be a valid table entry.
            const shimBytes = new Uint8Array([
                0,97,115,109,1,0,0,0,
                1,6,1,96,2,127,127,0,
                2,13,1,4,104,97,122,121,4,115,116,101,112,0,0,
                3,2,1,0,
                7,8,1,4,115,116,101,112,0,1,
                10,10,1,8,0,32,0,32,1,16,0,11
            ]);

            const step = (controller, methodInfo) => {
                if (insideHook || (!wanted && !spider && !extraMovementWanted())) {
                    return original(controller, methodInfo);
                }

                insideHook = true;
                let before = null;
                let motion = null;
                let spiderOnly = false;
                let spiderSample = null;
                let extraSample = null;

                try {
                    try {
                        const candidate = resolve();
                        if (candidate && candidate.table === installedTable &&
                            controller === candidate.controller &&
                            !candidate.isKinematic(candidate.body, 0)) {
                            if (!wanted && extraMovementWanted()) {
                                try { extraSample = captureExtraMovement(candidate); }
                                catch (error) { stopExtraMovement(error); }
                            }

                            if (wanted && samePlayer(record, candidate)) {
                                before = candidate;

                                // Collision changes wait until the flight velocity is applied.
                                const controls = getControls();

                                // GetMovementInput already rotates WASD into world space.
                                // Read BEFORE native FixedUpdate adjusts it for a slope.
                                const view = before.view;
                                let x = controls.enabled
                                    ? view.getFloat32(controller + 0xc0, true) : 0;
                                let z = controls.enabled
                                    ? view.getFloat32(controller + 0xc8, true) : 0;

                                if (![x, z].every(v =>
                                    Number.isFinite(v) && Math.abs(v) <= 2)) {
                                    throw new Error('Fly: unexpected movement direction.');
                                }

                                const length = Math.hypot(x, z);
                                if (length > 0.00001) {
                                    x /= length;
                                    z /= length;
                                } else {
                                    x = 0;
                                    z = 0;
                                }

                                // Uses the current movement limit, so Speed Boost also
                                // increases horizontal flight speed when active.
                                const limit = view.getFloat32(controller + 0x78, true);
                                if (!Number.isFinite(limit) ||
                                    limit <= 0 || limit > 200) {
                                    throw new Error('Fly: unexpected movement speed.');
                                }

                                const horizontalSpeed = Math.min(limit, 60);
                                const verticalInput = controls.enabled
                                    ? Number(controls.up) - Number(controls.down) : 0;

                                motion = [
                                    x * horizontalSpeed,
                                    verticalInput * FLY_VERTICAL_SPEED,
                                    z * horizontalSpeed
                                ];

                                // Space is flight input, not a queued normal/high/air jump.
                                view.setUint8(controller + 0xe5, 0);
                            } else if (!wanted && spider) {
                                // Capture collision-resolved velocity BEFORE Kour
                                // accelerates this frame. The climb itself is applied
                                // after Kour finishes its normal FixedUpdate.
                                before = candidate;
                                spiderOnly = true;
                                spiderSample = captureSpiderSample(candidate);
                            }
                        }
                    } catch (error) {
                        before = null;
                        if (wanted) fail(error);
                        else if (spider) failSpider(error);
                        else if (extraMovementWanted()) stopExtraMovement(error);
                    }

                    // Always preserve the game's own controller bookkeeping first.
                    try {
                        original(controller, methodInfo);
                    } catch (error) {
                        if (before) {
                            if (wanted) fail(error);
                            else if (spiderOnly) failSpider(error);
                        }
                        throw error;
                    }

                    if (extraSample) {
                        try { applyExtraMovement(resolve(), extraSample); }
                        catch (error) { stopExtraMovement(error); }
                    }
                    if (!before) return;

                    // Spider is deliberately a separate path from Fly. Gravity and
                    // the physical capsule remain enabled.
                    if (spiderOnly) {
                        try {
                            const after = resolve();
                            if (!samePlayer(before, after) ||
                                after.isKinematic(after.body, 0)) {
                                spiderContactSince = null;
                                return;
                            }
                            applySpiderClimb(after, spiderSample);
                        } catch (error) {
                            failSpider(error);
                        }
                        return;
                    }

                    if (!motion || !wanted) return;

                    try {
                        const after = resolve();
                        if (!samePlayer(before, after) ||
                            after.isKinematic(after.body, 0)) return;

                        setGravity(after, 0);

                        // Native FixedUpdate has applied its custom gravity/friction.
                        // Replace its final velocity before the physics step.
                        writeVelocity(after, ...motion);

                        if (lastFlightAt === null) {
                            const address = after.controller + 0xb4;
                            after.getVelocity(address, after.body, 0);
                            const fresh = currentMemoryView(after.module);

                            for (let axis = 0; axis < 3; axis++) {
                                const actual =
                                    fresh.getFloat32(address + axis * 4, true);
                                if (!Number.isFinite(actual) ||
                                    Math.abs(actual - motion[axis]) > 0.001) {
                                    throw new Error(
                                        'Fly: flight velocity did not read back correctly.'
                                    );
                                }
                            }
                        }

                        // NoClip is applied ONLY from a running flight update.
                        updateCollisionMode(after);
                        lastFlightAt = clock();
                        waitingSince = null;
                    } catch (error) {
                        fail(error);
                    }
                } finally {
                    insideHook = false;
                }
            };

            const shimModule = new WebAssembly.Module(shimBytes);
            const shim = new WebAssembly.Instance(shimModule, { hazy: { step } });
            const wrapper = shim.exports.step;
            ctx.table.set(FIXED_UPDATE_SLOT, wrapper);
            if (ctx.table.get(FIXED_UPDATE_SLOT) !== wrapper) {
                throw new Error('Fly: movement hook installation failed.');
            }
            hook = { table: ctx.table, original, wrapper, shim };
            hooksByTable.set(ctx.table, hook);
        }

        function tick() {
            if (!wanted && !spider && !extraMovementWanted() && !record && !collisionRecord) return;

            try {
                const ctx = resolve();
                if (!ctx) {
                    waitingSince = null;
                    spiderContactSince = null;
                    return; // Lobby/respawn: no guessed pointer writes.
                }

                if (record && !samePlayer(record, ctx)) {
                    record = null;
                    collisionRecord = null;
                    lastFlightAt = null;
                    waitingSince = null;
                }

                if (!wanted) {
                    // Clean up any prior Fly state first, then leave ordinary
                    // gravity/collision untouched for Spider.
                    if (record || collisionRecord) restorePlayer(ctx);

                    if (!spider && !extraMovementWanted()) {
                        spiderContactSince = null;
                        return;
                    }

                    if (ctx.isKinematic(ctx.body, 0)) {
                        spiderContactSince = null;
                        return;
                    }

                    if (spider) installSpiderCollisionHook(ctx);
                    installHook(ctx);
                    return;
                }

                if (ctx.isKinematic(ctx.body, 0)) {
                    restorePlayer(ctx);
                    return;
                }

                installHook(ctx);

                if (!record) {
                    record = {
                        module: ctx.module,
                        shooter: ctx.shooter,
                        controller: ctx.controller,
                        shooterNative: ctx.shooterNative,
                        controllerNative: ctx.controllerNative,
                        body: ctx.body,
                        bodyNative: ctx.bodyNative,
                        originalGravity: readGravity(ctx)
                    };
                    lastFlightAt = null;
                    waitingSince = null;
                }

                // Switching back to ordinary Fly restores the capsule immediately.
                if (!noclipWanted) restoreCollisions(ctx);

                const controls = getControls();
                if (!controls.enabled) {
                    waitingSince = null;
                    return;
                }

                const now = clock();
                if (waitingSince === null) waitingSince = now;
                const lastSeen = lastFlightAt === null
                    ? waitingSince
                    : Math.max(lastFlightAt, waitingSince);

                if (now - lastSeen > CALLBACK_TIMEOUT_MS) {
                    throw new Error(
                        'Fly: movement callback was not reached; save and refresh Kour.'
                    );
                }
            } catch (error) {
                if (wanted) fail(error);
                else if (spider) failSpider(error);
                else if (extraMovementWanted()) stopExtraMovement(error);
            }
        }

        function setMode(mode) {
            if (!['off', 'fly', 'noclip'].includes(mode)) {
                throw new RangeError('Flight mode must be off, fly, or noclip.');
            }
            const enabled = mode !== 'off';
            const clip = mode === 'noclip';
            if (enabled === wanted && clip === noclipWanted) return;
            resetExtraMovementInput();
            wanted = enabled;
            noclipWanted = clip;
            errorReported = false;
            notify(wanted, noclipWanted); // Immediate lobby/menu feedback.
            waitingSince = null;
            if (!wanted) lastFlightAt = null;
            // All modes share a single startup-installed movement wrapper.
            // It forwards directly to the game's function when both modes are off.
            tick();
        }

        function setEnabled(enabled) {
            setMode(enabled ? 'fly' : 'off');
        }

        function setSpiderEnabled(enabled) {
            spider = Boolean(enabled);
            spiderContactSince = null;
            spiderCollisionAt = -Infinity;
            spiderErrorReported = false;
            tick();
        }

        return {
            tick, setEnabled, setMode, setSpiderEnabled, prepareRuntime,
            get wanted() { return wanted; },
            get mode() { return !wanted ? 'off' : noclipWanted ? 'noclip' : 'fly'; }
        };
    }

    const flyEngine = createFlyController(
        () => speedEngine.getPlayer(),
        ctx => findJumpTable(ctx),
        () => ({
            enabled: Boolean(document.pointerLockElement) && !document.hidden &&
                (typeof document.hasFocus !== 'function' || document.hasFocus()) &&
                !editingText({ target: document.activeElement }),
            up: flyHeldKeys.has('Space'),
            down: flyHeldKeys.has('ControlLeft') || flyHeldKeys.has('ControlRight')
        }),
        (enabled, noclipEnabled = false) => {
            fly = enabled && !noclipEnabled;
            noClipFly = enabled && noclipEnabled;
            if (!enabled) releaseFlyKeys();
            renderActiveMods();
        }
    );

    window.setInterval(() => flyEngine.tick(), FLY_CHECK_MS);
    window.addEventListener('pagehide', () => {
        flyEngine.setSpiderEnabled(false);
        flyEngine.setEnabled(false);
    });

    // Registered before Air Jump: when Fly is on, Space means rise, not jump.
    window.addEventListener('keydown', event => {
        const flightKey = event.code === 'Space' ||
            event.code === 'ControlLeft' || event.code === 'ControlRight';
        if (!flightKey || !flyEngine.wanted || !document.pointerLockElement ||
            event.isComposing || editingText(event)) return;
        flyHeldKeys.add(event.code);
        event.preventDefault();
        event.stopImmediatePropagation();
    }, true);

    window.addEventListener('keyup', event => {
        // Always clear keys, including after a toggle, chat focus, or pointer unlock.
        flyHeldKeys.delete(event.code);
    }, true);

    window.addEventListener('blur', releaseFlyKeys);
    document.addEventListener('pointerlockchange', () => {
        if (!document.pointerLockElement) releaseFlyKeys();
    });
    document.addEventListener('visibilitychange', () => {
        if (document.hidden) releaseFlyKeys();
    });
    document.addEventListener('focusin', event => {
        if (editingText(event)) releaseFlyKeys();
    });

    // AIR JUMP — ONE NATIVE JUMP PER FRESH SPACE PRESS
    // ============================================================
    // Exact function-table entries from the uploaded 4.24 WASM.
    const AIR_JUMP_TABLE = 17495;       // PlayerController.Jump(bool force)
    const GET_VELOCITY_TABLE = 45204;   // Rigidbody.get_linearVelocity
    const SET_VELOCITY_TABLE = 45205;   // Rigidbody.set_linearVelocity

    let spaceHeld = false;
    let spaceConsumed = false;
    let airJumpErrorReported = false;
    const capturedRuntimes = [];

    // Remember the game's exports without changing its WASM or its clock.
    // Module.asm is also checked below, in case the game loaded earlier.
    function rememberRuntime(result) {
        const exports = result?.instance?.exports || result?.exports;
        if (!exports) return;
        const table = exports.__indirect_function_table;
        const memory = exports.memory;
        if (!table || !memory || typeof table.get !== 'function') return;
        if (capturedRuntimes.some(item => item.table === table)) return;
        capturedRuntimes.push({ table, memory, exports });
        // Install before this promise returns the instance to Unity's bootstrap,
        // so later cached movement callbacks refer to our stable forwarding entry.
        try { flyEngine.prepareRuntime(table, memory); }
        catch (error) { console.warn('[Hazy Client] Flight startup hook:', error); }
        try { prepareChickenRuntime(table, memory); }
        catch (error) { console.warn('[Hazy Client] Chicken ESP startup hook:', error); }

        catch (error) { console.warn('[Hazy Client] Weapon-settings startup hook:', error); }
        if (capturedRuntimes.length > 4) capturedRuntimes.shift();
    }

        try {
            const original = WebAssembly[method];
            if (typeof original !== 'function') continue;

                    try { rememberRuntime(result); } catch (_) {}

                });
            };

    }

    function currentMemoryView(module) {
        const heap = module?.HEAPU8 || module?.HEAP8;
        if (heap?.buffer) {
            return new DataView(heap.buffer, heap.byteOffset, heap.byteLength);
        }
        const buffer = module?.wasmMemory?.buffer;
        return buffer ? new DataView(buffer) : null;
    }

    function findJumpTable(ctx) {
        const m = ctx.module;
        const candidates = [
            m.asm?.__indirect_function_table,
            m.wasmExports?.__indirect_function_table,
            m.__indirect_function_table,
            m.wasmTable
        ];
        for (const record of capturedRuntimes) {
            if (record.memory.buffer === ctx.view.buffer) candidates.push(record.table);
        }
        return candidates.find(table => table && typeof table.get === 'function' &&
            table.length > SET_VELOCITY_TABLE) || null;
    }

    function infoValid(view, pointer, bytes = 4) {
        return Boolean(view) && Number.isInteger(pointer) && pointer >= 1024 &&
            pointer % 4 === 0 && pointer + bytes <= view.byteLength;
    }
    function infoUint(view, pointer) {
        return infoValid(view, pointer) ? view.getUint32(pointer, true) : 0;
    }
    function infoAsciiEquals(view, pointer, value) {
        if (!Number.isInteger(pointer) || pointer < 1024 ||
            pointer + value.length >= view.byteLength) return false;
        for (let i = 0; i < value.length; i++) {
            if (view.getUint8(pointer + i) !== value.charCodeAt(i)) return false;
        }
        return view.getUint8(pointer + value.length) === 0;
    }
    function infoClassNamed(view, klass, name) {
        return infoValid(view, klass, 0x60) &&
            infoUint(view, klass + 0x3c) === klass &&
            infoAsciiEquals(view, infoUint(view, klass + 8), name);
    }
    function infoObjectNamed(view, object, name, bytes = 12) {
        return infoValid(view, object, bytes) && infoUint(view, object + 8) !== 0 &&
            infoClassNamed(view, infoUint(view, object), name);
    }
    function infoDerivesFrom(view, klass, name) {
        for (let depth = 0; depth < 12 && infoValid(view, klass, 0x60); depth++) {
            if (infoClassNamed(view, klass, name)) return true;
            const parent = infoUint(view, klass + 0x2c);
            if (!parent || parent === klass) return false;
            klass = parent;
        }
        return false;
    }
    function readManagedString(view, pointer, maxChars = 64) {
        if (!infoValid(view, pointer, 12)) return '';
        const length = view.getInt32(pointer + 8, true);
        if (length < 0 || length > 4096 || pointer + 12 + length * 2 > view.byteLength) return '';
        let text = '';
        for (let i = 0; i < Math.min(length, maxChars); i++) {
            const code = view.getUint16(pointer + 12 + i * 2, true);
            if (!code) break;
            text += String.fromCharCode(code);
        }
        return text.replace(/[\u0000-\u001f\u007f]/g, '').trim();
    }
    function readObscuredInt(view, pointer) {
        // Embedded value type: no boxed-object header. Do not substitute a
        // fakeValue/debug copy, and do not invent health before initialization.
        if (!infoValid(view, pointer, 20) || view.getUint8(pointer + 8) !== 1) return null;
        return (view.getInt32(pointer + 4, true) ^ view.getInt32(pointer, true)) | 0;
    }
    function sceneView(runtime) {
        if (runtime?.memory?.buffer) return new DataView(runtime.memory.buffer);
        return runtime?.module ? currentMemoryView(runtime.module) : null;
    }
    function sceneBuildMatches(view) {
        return Boolean(view) && SCENE_CHECKS.every(([address, value]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === value);
    }
    function findSceneRuntime() {
        const module = window.unityInstance?.Module || window.Module || null;
        const moduleView = currentMemoryView(module);
        for (let i = capturedRuntimes.length - 1; i >= 0; i--) {
            const runtime = capturedRuntimes[i];
            const view = sceneView(runtime);
            if (sceneBuildMatches(view) && runtime.table?.length > 41054) {
                if (moduleView?.buffer === view.buffer) runtime.module = module;
                return runtime;
            }
        }
        if (!moduleView || !sceneBuildMatches(moduleView)) return null;
        const exports = module.asm || module.wasmExports || null;
        const table = module.asm?.__indirect_function_table ||
            module.wasmExports?.__indirect_function_table ||
            module.__indirect_function_table || module.wasmTable;
        return table?.length > 41054 ? { table, module, exports } : null;
    }
    function sceneFunction(runtime, slot) {
        const fn = runtime?.table?.get(slot);
        if (typeof fn !== 'function') throw new Error(`Game function ${slot} is unavailable. Refresh Kour.`);
        return fn;
    }
    function sceneScratchPointer(runtime) {
        const key = runtime.table;
        let record = sceneScratch.get(key);
        const view = sceneView(runtime);
        if (record && infoValid(view, record.pointer, 32)) return record.pointer;
        // Some Unity loaders expose Module._malloc; others expose only the
        // WASM export named "malloc". Both belong to the same game memory.
        const sources = [
            ['Module', runtime.module],
            ['WASM exports', runtime.exports],
            ['Module.asm', runtime.module?.asm],
            ['Module.wasmExports', runtime.module?.wasmExports]
        ];
        for (const [label, owner] of sources) {
            const malloc = owner?._malloc || owner?.malloc;
            const free = owner?._free || owner?.free;
            if (typeof malloc !== 'function' || typeof free !== 'function') continue;
            const pointer = Reflect.apply(malloc, owner, [32]) >>> 0;
            if (!infoValid(sceneView(runtime), pointer, 32)) {
                if (pointer) Reflect.apply(free, owner, [pointer]);
                throw new Error('Game scratch allocation returned an invalid address.');
            }
            record = { pointer, free: () => Reflect.apply(free, owner, [pointer]) };
            sceneScratch.set(key, record);
            sceneDiagnostics.allocator = label;
            return pointer;
        }
        throw new Error('Game allocator unavailable. Save this script and refresh Kour.');
    }

    function registryShooters(runtime) {
        const view = sceneView(runtime);
        if (!sceneBuildMatches(view)) throw new Error('Player-data layout does not match this build.');
        const klass = infoUint(view, 0x696b74);
        if (!infoClassNamed(view, klass, 'PlayerController')) return null;
        const statics = infoUint(view, klass + 0x5c);
        const set = infoUint(view, statics + 4);
        if (!infoValid(view, set, 40) ||
            !infoClassNamed(view, infoUint(view, set), 'HashSet`1')) return null;
        // HashSet<Shooter>: slots +12; count +16; lastIndex +20; version +32.
        // Slot is { int hashCode; int next; Shooter* value; }, stride 12.
        // These offsets are also read by the game's instantiated HashSet code.
        const count = view.getInt32(set + 16, true);
        const lastIndex = view.getInt32(set + 20, true);
        const version = view.getInt32(set + 32, true);
        if (count < 0 || count > 256 || lastIndex < count || lastIndex > 1024) {
            throw new Error('Unexpected player registry size; no scan performed.');
        }
        if (!count) return [];
        const slots = infoUint(view, set + 12);
        if (!infoValid(view, slots, 16)) return null;
        const capacity = infoUint(view, slots + 12);
        if (capacity < lastIndex || capacity > 8192 ||
            !infoValid(view, slots, 16 + lastIndex * 12)) return null;
        const players = [];
        for (let i = 0; i < lastIndex; i++) {
            const slot = slots + 16 + i * 12;
            if (view.getInt32(slot, true) < 0) continue;
            const shooter = infoUint(view, slot + 8);
            if (infoObjectNamed(view, shooter, 'Shooter', 0x208)) players.push(shooter);
        }
        if (view.getInt32(set + 32, true) !== version) return null;
        return players;
    }
    function cleanWeaponName(text) {
        return String(text || '').replace(/\(Clone\)$/i, '')
            .replace(/^weapon[_\-\s]*/i, '').replace(/_/g, ' ').trim().slice(0, 28);
    }
    function readWeaponName(runtime, shooter) {
        let view = sceneView(runtime);
        const weapon = infoUint(view, shooter + 0xe4);
        if (!infoValid(view, weapon, 0xa0) || !infoUint(view, weapon + 8) ||
            !infoDerivesFrom(view, infoUint(view, weapon), 'Weapon')) return '';
        const native = infoUint(view, weapon + 8);
        const old = playerInfoWeaponNames.get(weapon);
        if (old?.native === native) return old.name;
        let name = '';
        try {
            const pointer = sceneFunction(runtime, 3300)(weapon, 0);
            // Native calls can grow WASM memory: never reuse a detached view.
            name = cleanWeaponName(readManagedString(sceneView(runtime), pointer, 96));
        } catch (_) {}
        if (!name) {
            view = sceneView(runtime);
            name = cleanWeaponName(readManagedString(view, infoUint(view, weapon + 0x18), 96));
        }
        if (name) {
            if (playerInfoWeaponNames.size > 256) playerInfoWeaponNames.clear();
            playerInfoWeaponNames.set(weapon, { native, name });
        }
        return name;
    }
    function readShooterInfo(runtime, shooter, now) {
        let view = sceneView(runtime);
        if (!infoObjectNamed(view, shooter, 'Shooter', 0x208)) return null;
        const native = infoUint(view, shooter + 8);
        const photon = infoUint(view, shooter + 0x10);
        if (!infoObjectNamed(view, photon, 'PhotonView', 0x5c)) return null;
        const isMine = view.getUint8(photon + 0x40);
        if (isMine !== 0 && !(isMine === 1 && hazyBadges && thirdPerson)) return null;
        const owner = infoUint(view, photon + 0x50);
        if (!infoValid(view, owner, 0x28) ||
            !infoClassNamed(view, infoUint(view, owner), 'Player')) return null;
        const actor = view.getInt32(owner + 0x0c, true);
        if (actor <= 0 || actor > 1000000) return null;
        const name = readManagedString(view, infoUint(view, owner + 0x14), 32);
        let health = null, maxHealth = null;
        const hp = infoUint(view, shooter + 0x6c);
        if (infoObjectNamed(view, hp, 'Health', 0xb8) && infoUint(view, hp + 0x14) === shooter) {
            if (view.getUint8(hp + 0xa8) === 1) return null; // globalDied
            const max = readObscuredInt(view, hp + 0x58);
            const current = readObscuredInt(view, hp + 0x6c);
            if (Number.isInteger(max) && max > 0 && max <= 10000 &&
                Number.isInteger(current) && current >= -1000 && current <= max + 1000) {
                maxHealth = max;
                health = Math.max(0, current);
            }
        }
        const old = playerInfoByActor.get(actor);
        let transform = old?.shooter === shooter && old?.shooterNative === native ? old.transform : 0;
        if (!infoObjectNamed(view, transform, 'Transform', 12) ||
            (old && infoUint(view, transform + 8) !== old.transformNative)) {
            transform = sceneFunction(runtime, 2964)(shooter, 0);
            view = sceneView(runtime);
        }
        if (!infoObjectNamed(view, transform, 'Transform', 12)) return null;
        const transformNative = infoUint(view, transform + 8);
        // Each label is optional. A missing weapon must not hide a valid name/HP.
        const weapon = weaponESP ? readWeaponName(runtime, shooter) : '';
        return { actor, name, health, maxHealth, weapon, isLocalBadge: isMine === 1, t: now,
            shooter, shooterNative: native, transform, transformNative, runtime };
    }

    // THIRD PERSON — CAMERA OFFSET + FULL-CHARACTER VISIBILITY

    const THIRD_PERSON_OFFSET_Y = 0.45;
    const THIRD_PERSON_OFFSET_Z = -5.8;
    const THIRD_PERSON_BUILD_CHECKS = [
        [0x3c67c4, 0x64], [0x6d8d7c, 2964],
        [0x6d96a4, 41053], [0x6d96a8, 41054],
        [0x6bf588, 18182], // CameraWalk.Update
        [0x3c7300, 0x20]  // CameraWalk.playerController

    let thirdPersonRecord = null;
    let thirdPersonErrorReported = false;
    const cameraUpdateHooks = new Map();

    // In the uploaded 4.24 build, Setup.Start assigns locally hidden character
    // graphics to layer 7. Shooter.SetWeapon assigns the local first-person
    // weapon hierarchy to layer 15. Moving the camera alone never changes this.
    // Change only what our camera renders; do not change object layers, player
    // ownership, character health, colliders, or the local weapon's game logic.
    const THIRD_PERSON_BODY_LAYER = 7;
    const THIRD_PERSON_VIEWMODEL_LAYER = 15;
    const THIRD_PERSON_VISIBILITY_BITS =
        (1 << THIRD_PERSON_BODY_LAYER) | (1 << THIRD_PERSON_VIEWMODEL_LAYER);
    const THIRD_PERSON_VISIBILITY_CHECKS = [
        [0x3c9e44, 0xbc],      // Shooter.weaponCamera
        [0x3c9cf4, 0x58],      // Setup.hideForMeGO
        [0x6d6b4c, 11420],     // Camera.get_cullingMask
        [0x6d6b50, 38499],     // Camera.set_cullingMask
        [0x6d8d40, 3167],      // Behaviour.get_enabled
        [0x6d8d44, 3045]       // Behaviour.set_enabled
    ];
    let thirdPersonVisualRecord = null;

    function thirdPersonVisibilityBuildMatches(view) {
        return Boolean(view) && THIRD_PERSON_VISIBILITY_CHECKS.every(([address, value]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === value);
    }
    function readThirdPersonMask(api, camera) {
        const value = api.getMask(camera, 0);
        if (!Number.isInteger(value) || value < -2147483648 || value > 4294967295) {
            throw new Error('Camera returned an invalid visibility mask.');
        }
        return value | 0;
    }
    function readThirdPersonCameraEnabled(api, camera) {
        const value = api.getEnabled(camera, 0);
        if (value !== 0 && value !== 1) {
            throw new Error('Weapon camera returned an invalid enabled state.');
        }
        return value;
    }
    function thirdPersonCameraAlive(view, pointer, native) {
        return infoObjectNamed(view, pointer, 'Camera', 12) &&
            infoUint(view, pointer + 8) === native;
    }
    function restoreThirdPersonVisuals() {
        const record = thirdPersonVisualRecord;
        if (!record) return;
        const view = sceneView(record.runtime);
        if (!thirdPersonVisibilityBuildMatches(view)) {
            thirdPersonVisualRecord = null;
            return; // Never call build-specific methods on a different layout.
        }
        let failure = null;
        try {
            if (thirdPersonCameraAlive(view, record.camera, record.cameraNative)) {
                const current = readThirdPersonMask(record, record.camera);
                // Restore ONLY our two visibility bits. Preserve any unrelated
                // culling-mask changes the game made while Third Person was on.
                if ((current & THIRD_PERSON_VISIBILITY_BITS) === record.appliedBits) {
                    const restored = (current & ~THIRD_PERSON_VISIBILITY_BITS) |
                        record.originalBits;
                    if (restored !== current) {
                        record.setMask(record.camera, restored, 0);
                        if (readThirdPersonMask(record, record.camera) !== restored) {
                            throw new Error('Could not restore the first-person visibility mask.');
                        }
                    }
                }
            }
        } catch (error) { failure = error; }
        try {
            if (record.weaponCamera && thirdPersonCameraAlive(
                sceneView(record.runtime), record.weaponCamera, record.weaponCameraNative)) {
                const current = readThirdPersonCameraEnabled(record, record.weaponCamera);
                if (current === 0 && record.originalWeaponEnabled !== current) {
                    record.setEnabled(record.weaponCamera, record.originalWeaponEnabled, 0);
                    if (readThirdPersonCameraEnabled(record, record.weaponCamera) !==
                        record.originalWeaponEnabled) {
                        throw new Error('Could not restore the first-person weapon camera.');
                    }
                }
            }
        } catch (error) { if (!failure) failure = error; }
        // Keep the record on failure so the error cleanup can retry restoration.
        if (failure) throw failure;
        thirdPersonVisualRecord = null;
    }
    function applyThirdPersonVisuals(ctx) {
        const view = sceneView(ctx.runtime);
        if (!thirdPersonVisibilityBuildMatches(view)) {
            throw new Error('Third-person visibility layout does not match this build.');
        }
        let weaponCamera = infoUint(view, ctx.shooter + 0xbc);
        // A variant that shares one camera must never disable the main camera.
        if (weaponCamera === ctx.camera) weaponCamera = 0;
        if (weaponCamera && !infoObjectNamed(view, weaponCamera, 'Camera', 12)) {
            throw new Error('The local first-person weapon camera is unavailable.');
        }
        const weaponCameraNative = weaponCamera ? infoUint(view, weaponCamera + 8) : 0;
        const previous = thirdPersonVisualRecord;
        if (!sameThirdPersonCamera(previous, ctx) ||
            previous.weaponCamera !== weaponCamera ||
            previous.weaponCameraNative !== weaponCameraNative) {
            restoreThirdPersonVisuals();
            const api = {
                ...ctx, weaponCamera, weaponCameraNative,
                getMask: sceneFunction(ctx.runtime, 11420),
                setMask: sceneFunction(ctx.runtime, 38499),
                getEnabled: sceneFunction(ctx.runtime, 3167),
                setEnabled: sceneFunction(ctx.runtime, 3045)
            };
            const originalMask = readThirdPersonMask(api, ctx.camera);
            const originalWeaponEnabled = weaponCamera
                ? readThirdPersonCameraEnabled(api, weaponCamera) : 0;
            // Capture before either write so a later error can undo both changes.
            thirdPersonVisualRecord = {
                ...api,
                originalBits: originalMask & THIRD_PERSON_VISIBILITY_BITS,
                appliedBits: 1 << THIRD_PERSON_BODY_LAYER,
                originalWeaponEnabled
            };
        }
        const record = thirdPersonVisualRecord;
        const current = readThirdPersonMask(record, ctx.camera);
        const desired = (current | (1 << THIRD_PERSON_BODY_LAYER)) &
            ~(1 << THIRD_PERSON_VIEWMODEL_LAYER);
        if (current !== desired) {
            record.setMask(ctx.camera, desired, 0);
            if (readThirdPersonMask(record, ctx.camera) !== desired) {
                throw new Error('Full-character visibility did not read back correctly.');
            }
        }
        // Suppress the separate first-person draw pass, not the weapon object.
        // Its shooting/reload code remains enabled and otherwise unchanged.
        if (record.weaponCamera && readThirdPersonCameraEnabled(record, record.weaponCamera)) {
            record.setEnabled(record.weaponCamera, 0, 0);
            if (readThirdPersonCameraEnabled(record, record.weaponCamera) !== 0) {
                throw new Error('The first-person weapon camera could not be hidden.');
            }
        }
    }

    // ============================================================
    // THIRD-PERSON BODY ANIMATION — LOCAL PLAYER ONLY
    // ============================================================
    // The uploaded 4.24 CharacterAnimatorManager.Update handles emotes on the
    // IsMine path, but calculates estimatedMoveVertical/Horizontal only on the
    // remote-player path. SlowUpdate feeds those estimates to Animator "Move".
    // Supply estimates for the visible local body, using actual body velocity.
    // Never change Photon ownership or call a Shooter RPC to animate locally.
    const THIRD_PERSON_ANIMATION_CHECKS = [
        [0x3c9dd8, 0x44],       // Shooter.characterAnimatorManager
        [0x3c67a8, 0x48],       // PlayerController.characterAnimatorManager
        [0x3c7460, 0x14],       // CharacterAnimatorManager.shooter
        [0x3c7480, 0x2e],       // localIsListeningForDanceCancel
        [0x3c7488, 0x34],       // velMoveTreshold
        [0x3c7490, 0x3c],       // parentPlayer (Transform)
        [0x3c7494, 0x40],       // animator
        [0x3c749c, 0x50],       // estimatedMoveVertical
        [0x3c74a0, 0x54],       // estimatedMoveHorizontal
        [0x3c9e94, 0x1a0],      // Shooter.isCrouching
        [0x3c6834, 0xe4],       // PlayerController.onGround
        [0x3c6820, 0xb0],       // PlayerController.rb
        [0x6bf68c, 18247],      // CharacterAnimatorManager.SetCrouch (local only)
        [0x6bf690, 18248],      // CharacterAnimatorManager.SetAirState (local only)
        [0x6bf698, 18250],      // CharacterAnimatorManager.SlowUpdate
        [0x6d5898, 37607],      // Animator.get_isInitialized
        [0x6d8d40, 3167],       // Behaviour.get_enabled
        [0x6d8d44, 3045],       // Behaviour.set_enabled
        [0x6d96bc, 41059],      // Transform.get_right
        [0x6d96c8, 4579],       // Transform.get_forward
        [0x6dec38, 45204]       // Rigidbody.get_linearVelocity
    ];
    let thirdPersonAnimationRecord = null;

    function thirdPersonAnimationBuildMatches(view) {
        return Boolean(view) && THIRD_PERSON_ANIMATION_CHECKS.every(([address, value]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === value);
    }
    function thirdPersonAnimationAlive(record, view) {
        if (!record || !view) return false;
        return infoObjectNamed(view, record.shooter, 'Shooter', 0x208) &&
            infoUint(view, record.shooter + 8) === record.shooterNative &&
            infoObjectNamed(view, record.controller, 'PlayerController', 0x12c) &&
            infoUint(view, record.controller + 8) === record.controllerNative &&
            infoObjectNamed(view, record.manager, 'CharacterAnimatorManager', 0x5c) &&
            infoUint(view, record.manager + 8) === record.managerNative &&
            infoUint(view, record.manager + 0x14) === record.shooter &&
            infoObjectNamed(view, record.animator, 'Animator', 12) &&
            infoUint(view, record.animator + 8) === record.animatorNative &&
            infoUint(view, record.manager + 0x40) === record.animator;
    }
    function sameThirdPersonAnimator(a, b) {
        return Boolean(a && b && a.runtime.table === b.runtime.table &&
            a.controller === b.controller && a.controllerNative === b.controllerNative &&
            a.shooter === b.shooter && a.shooterNative === b.shooterNative &&
            a.manager === b.manager && a.managerNative === b.managerNative &&
            a.animator === b.animator && a.animatorNative === b.animatorNative &&
            a.body === b.body && a.bodyNative === b.bodyNative &&
            a.parent === b.parent && a.parentNative === b.parentNative);
    }
    function resolveThirdPersonAnimator(ctx) {
        const view = sceneView(ctx.runtime);
        if (!thirdPersonAnimationBuildMatches(view)) {
            throw new Error('Body-animation layout does not match this Kour build.');
        }
        const photon = infoUint(view, ctx.shooter + 0x10);
        if (!infoObjectNamed(view, photon, 'PhotonView', 0x44) ||
            view.getUint8(photon + 0x40) !== 1) return null;
        const manager = infoUint(view, ctx.shooter + 0x44);
        if (!infoObjectNamed(view, manager, 'CharacterAnimatorManager', 0x5c) ||
            infoUint(view, ctx.controller + 0x48) !== manager ||
            infoUint(view, manager + 0x14) !== ctx.shooter) return null;
        const animator = infoUint(view, manager + 0x40);
        const parent = infoUint(view, manager + 0x3c);
        const body = infoUint(view, ctx.controller + 0xb0);
        if (!infoObjectNamed(view, animator, 'Animator', 12) ||
            !infoObjectNamed(view, parent, 'Transform', 12) ||
            !infoObjectNamed(view, body, 'Rigidbody', 12)) return null;
        return {
            ...ctx, manager, animator, parent, body,
            managerNative: infoUint(view, manager + 8),
            animatorNative: infoUint(view, animator + 8),
            parentNative: infoUint(view, parent + 8),
            bodyNative: infoUint(view, body + 8),
            getEnabled: sceneFunction(ctx.runtime, 3167),
            setEnabled: sceneFunction(ctx.runtime, 3045),
            isInitialized: sceneFunction(ctx.runtime, 37607),
            getVelocity: sceneFunction(ctx.runtime, 45204),
            getForward: sceneFunction(ctx.runtime, 4579),
            getRight: sceneFunction(ctx.runtime, 41059),
            slowUpdate: sceneFunction(ctx.runtime, 18250),
            setCrouch: sceneFunction(ctx.runtime, 18247),
            setAirState: sceneFunction(ctx.runtime, 18248)
        };
    }
    function readThirdPersonAnimationEnabled(record) {
        const value = record.getEnabled(record.animator, 0);
        if (value !== 0 && value !== 1) throw new Error('Invalid body Animator enabled state.');
        return value;
    }
    function readThirdPersonAnimationVector(record, fn, object, pointer) {
        // A dedicated scratch result, not the player's cached physics velocity.
        fn(pointer, object, 0);
        const view = sceneView(record.runtime);
        const value = { x: view.getFloat32(pointer, true),
            y: view.getFloat32(pointer + 4, true), z: view.getFloat32(pointer + 8, true) };
        if (!Object.values(value).every(n => Number.isFinite(n) && Math.abs(n) < 10000)) {
            throw new Error('Body animation received an invalid direction or velocity.');
        }
        return value;
    }
    function thirdPersonAnimationMovement(velocity, forward, right, threshold) {
        const speed = Math.hypot(velocity.x, velocity.z);
        if (speed <= Math.max(0.05, threshold)) return { vertical: 0, horizontal: 0 };
        const forwardLength = Math.hypot(forward.x, forward.z);
        const rightLength = Math.hypot(right.x, right.z);
        if (forwardLength < 0.0001 || rightLength < 0.0001) {
            return { vertical: 0, horizontal: 0 };
        }
        const vertical = (velocity.x * forward.x + velocity.z * forward.z) /
            (speed * forwardLength);
        const horizontal = (velocity.x * right.x + velocity.z * right.z) /
            (speed * rightLength);
        // The native Move parameter expects -1 / 0 / 1, not raw world speed.
        // Sideways motion uses the existing horizontal fallback in SlowUpdate.
        return {
            vertical: Math.abs(vertical) > 0.10 ? Math.sign(vertical) : 0,
            horizontal: Math.abs(horizontal) > 0.10 ? Math.sign(horizontal) : 0
        };
    }
    function restoreThirdPersonAnimation() {
        const record = thirdPersonAnimationRecord;
        if (!record) return;
        let view = sceneView(record.runtime);
        if (!thirdPersonAnimationBuildMatches(view) || !thirdPersonAnimationAlive(record, view)) {
            thirdPersonAnimationRecord = null;
            return; // Replaced/destroyed character: never write through stale pointers.
        }
        let failure = null;
        let restoredMovement = false;
        try {
            for (const [offset, original, applied] of [
                [0x50, record.originalVertical, record.appliedVertical],
                [0x54, record.originalHorizontal, record.appliedHorizontal]
            ]) {
                if (applied === null) continue;
                const current = view.getFloat32(record.manager + offset, true);
                // Preserve unrelated changes, for example an emote or a new script.
                if (Math.abs(current - applied) <= 0.00001) {
                    view.setFloat32(record.manager + offset, original, true);
                    restoredMovement = true;
                }
            }
            if (restoredMovement && record.isInitialized(record.animator, 0) === 1) {
                record.slowUpdate(record.manager, 0);
            }
        } catch (error) { failure = error; }
        try {
            // Only undo enabling that this feature itself performed.
            if (record.enabledByUs && readThirdPersonAnimationEnabled(record) === 1) {
                record.setEnabled(record.animator, record.originalEnabled, 0);
                if (readThirdPersonAnimationEnabled(record) !== record.originalEnabled) {
                    throw new Error('Could not restore body Animator enabled state.');
                }
            }
        } catch (error) { if (!failure) failure = error; }
        if (failure) throw failure;
        thirdPersonAnimationRecord = null;
    }
    function updateThirdPersonAnimation(ctx) {
        const current = resolveThirdPersonAnimator(ctx);
        if (!current) {
            if (thirdPersonAnimationRecord) restoreThirdPersonAnimation();
            return false; // Allow a new character/skin to finish creating its Animator.
        }
        if (!sameThirdPersonAnimator(thirdPersonAnimationRecord, current)) {
            restoreThirdPersonAnimation();
            const view = sceneView(current.runtime);
            const originalVertical = view.getFloat32(current.manager + 0x50, true);
            const originalHorizontal = view.getFloat32(current.manager + 0x54, true);
            if (![originalVertical, originalHorizontal].every(n => Number.isFinite(n) && Math.abs(n) <= 2)) {
                throw new Error('Unexpected local body-animation movement values.');
            }
            thirdPersonAnimationRecord = {
                ...current, originalVertical, originalHorizontal,
                originalEnabled: readThirdPersonAnimationEnabled(current),
                enabledByUs: false, appliedVertical: null, appliedHorizontal: null,
                lastGrounded: null, lastCrouched: null, wasDancing: false
            };
        }
        const record = thirdPersonAnimationRecord;
        if (readThirdPersonAnimationEnabled(record) === 0) {
            record.enabledByUs = true; // Capture before calling into native code.
            record.setEnabled(record.animator, 1, 0);
            if (readThirdPersonAnimationEnabled(record) !== 1) {
                throw new Error('The local body Animator could not be enabled.');
            }
        }
        const initialized = record.isInitialized(record.animator, 0);
        if (initialized !== 0 && initialized !== 1) throw new Error('Invalid body Animator readiness state.');
        if (!initialized) return false;

        let view = sceneView(record.runtime);
        const dancing = view.getUint8(record.manager + 0x2e);
        if (dancing > 1) throw new Error('Unexpected body animation/emote state.');
        if (dancing) {
            record.wasDancing = true;
            return true; // Keep Kour's dance/emote animation in control.
        }
        if (record.wasDancing) {
            record.wasDancing = false;
            record.lastGrounded = null;
            record.lastCrouched = null;
        }
        const threshold = view.getFloat32(record.manager + 0x34, true);
        if (!Number.isFinite(threshold) || threshold < 0 || threshold > 100) {
            throw new Error('Unexpected body walking-animation threshold.');
        }
        const pointer = sceneScratchPointer(record.runtime);
        const velocity = readThirdPersonAnimationVector(record, record.getVelocity, record.body, pointer);
        const forward = readThirdPersonAnimationVector(record, record.getForward, record.parent, pointer);
        const right = readThirdPersonAnimationVector(record, record.getRight, record.parent, pointer);
        const movement = thirdPersonAnimationMovement(velocity, forward, right, threshold);
        view = sceneView(record.runtime);
        let changed = record.appliedVertical === null || record.appliedHorizontal === null;
        for (const [offset, value] of [[0x50, movement.vertical], [0x54, movement.horizontal]]) {
            if (view.getFloat32(record.manager + offset, true) !== value) {
                view.setFloat32(record.manager + offset, value, true);
                changed = true;
            }
        }
        record.appliedVertical = movement.vertical;
        record.appliedHorizontal = movement.horizontal;
        // Push transitions immediately. The game's regular SlowUpdate continues
        // to use these cached estimates between changes; never restart a clip
        // or rebind the Animator every frame.
        if (changed) record.slowUpdate(record.manager, 0);

        view = sceneView(record.runtime);
        const grounded = view.getUint8(record.controller + 0xe4);
        const crouched = view.getUint8(record.shooter + 0x1a0);
        if (grounded > 1 || crouched > 1) throw new Error('Unexpected grounded/crouched animation state.');
        // Call the local manager, NOT Shooter.SetAirState/Photon RPC methods.
        if (record.lastCrouched !== crouched) {
            record.setCrouch(record.manager, crouched, 0);
            record.lastCrouched = crouched;
        }
        if (record.lastGrounded !== grounded) {
            record.setAirState(record.manager, grounded, 0);
            record.lastGrounded = grounded;
        }
        return true;
    }

    function thirdPersonBuildMatches(view) {
        return Boolean(view) && THIRD_PERSON_BUILD_CHECKS.every(([address, value]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === value);
    }
    function resolveThirdPersonCamera() {
        const player = speedEngine.getPlayer();
        const runtime = findSceneRuntime();
        if (!player || !runtime) return null;
        let view = sceneView(runtime);
        if (view.buffer !== player.view.buffer) return null;
        if (!thirdPersonBuildMatches(view)) {
            throw new Error('Camera layout does not match this Kour build.');
        }
        const camera = infoUint(view, player.controller + 0x64);
        if (!infoObjectNamed(view, camera, 'Camera', 12)) return null;
        const transform = sceneFunction(runtime, 2964)(camera, 0);
        view = sceneView(runtime);
        if (!infoObjectNamed(view, transform, 'Transform', 12)) return null;
        return { ...player, runtime, camera, transform,
            cameraNative: infoUint(view, camera + 8),
            transformNative: infoUint(view, transform + 8),
            getLocalPosition: sceneFunction(runtime, 41053),
            setLocalPosition: sceneFunction(runtime, 41054) };
    }
    function readThirdPersonPosition(ctx, pointer) {
        ctx.getLocalPosition(pointer, ctx.transform, 0);
        const view = sceneView(ctx.runtime);
        const p = { x: view.getFloat32(pointer, true), y: view.getFloat32(pointer + 4, true),
            z: view.getFloat32(pointer + 8, true) };
        if (!Object.values(p).every(v => Number.isFinite(v) && Math.abs(v) < 1000)) {
            throw new Error('Camera returned an invalid local position.');
        }
        return p;
    }
    function writeThirdPersonPosition(ctx, pointer, p) {
        const view = sceneView(ctx.runtime);
        view.setFloat32(pointer, p.x, true); view.setFloat32(pointer + 4, p.y, true);
        view.setFloat32(pointer + 8, p.z, true);
        ctx.setLocalPosition(ctx.transform, pointer, 0);
    }
    function sameThirdPersonCamera(a, b) {
        return Boolean(a && b && a.runtime.table === b.runtime.table &&
            a.controller === b.controller && a.controllerNative === b.controllerNative &&
            a.camera === b.camera && a.transform === b.transform &&
            a.cameraNative === b.cameraNative && a.transformNative === b.transformNative);
    }
    function closeCameraPosition(a, b) {
        return a && b && ['x', 'y', 'z'].every(k => Math.abs(a[k] - b[k]) <= 0.001);
    }
    function restoreThirdPerson(restoreVisuals = true) {
        const record = thirdPersonRecord;
        thirdPersonRecord = null;
        let failure = null;
        if (restoreVisuals) {
            try { restoreThirdPersonAnimation(); }
            catch (error) { failure = error; }
            try { restoreThirdPersonVisuals(); }
            catch (error) { if (!failure) failure = error; }
        }
        try {
            if (record) {
                const view = sceneView(record.runtime);
                if (thirdPersonBuildMatches(view) &&
                    infoObjectNamed(view, record.camera, 'Camera', 12) &&
                    infoObjectNamed(view, record.transform, 'Transform', 12) &&
                    infoUint(view, record.camera + 8) === record.cameraNative &&
                    infoUint(view, record.transform + 8) === record.transformNative) {
                    const pointer = sceneScratchPointer(record.runtime);
                    const current = readThirdPersonPosition(record, pointer);
                    // Do not overwrite a newer position set on death/respawn.
                    if (closeCameraPosition(current, record.applied)) {
                        writeThirdPersonPosition(record, pointer, record.original);
                    }
                }
            }
        } catch (error) { if (!failure) failure = error; }
        if (failure) throw failure;
    }
    function stopThirdPersonWithError(error) {
        thirdPerson = false;
        try { restoreThirdPerson(); } catch (_) {}
        sceneDiagnostics.camera = 'error';
        renderActiveMods();
        if (!thirdPersonErrorReported) {
            thirdPersonErrorReported = true;
            reportSceneError('Third Person', error);
        }
    }
    function updateThirdPerson() {
        if (!thirdPerson || !document.pointerLockElement || document.hidden ||
            (typeof document.hasFocus === 'function' && !document.hasFocus())) {
            try { if (thirdPersonRecord || thirdPersonVisualRecord || thirdPersonAnimationRecord) restoreThirdPerson(); }
            catch (error) { stopThirdPersonWithError(error); }
            sceneDiagnostics.camera = thirdPerson ? 'waiting for game focus' : 'off';
            return;
        }
        try {
            const ctx = resolveThirdPersonCamera();
            if (!ctx) {
                if (thirdPersonRecord || thirdPersonVisualRecord || thirdPersonAnimationRecord) restoreThirdPerson();
                sceneDiagnostics.camera = 'waiting for local camera';
                return;
            }
            const pointer = sceneScratchPointer(ctx.runtime);
            if (!sameThirdPersonCamera(thirdPersonRecord, ctx)) {
                // A camera-update hook clears only the position record. Keep
                // visibility stable across that update; its own code checks
                // player/camera identity and restores on a real replacement.
                restoreThirdPerson(false);
                thirdPersonRecord = { ...ctx, original: readThirdPersonPosition(ctx, pointer), applied: null };
            }
            const record = thirdPersonRecord;
            const current = readThirdPersonPosition(ctx, pointer);
            // A normal camera update may have replaced its bob/crouch offset.
            // Adopt that new base, but never add the offset to our own last write.
            if (record.applied && !closeCameraPosition(current, record.applied)) {
                record.original = current;
            }
            const next = { x: record.original.x,
                y: record.original.y + THIRD_PERSON_OFFSET_Y,
                z: record.original.z + THIRD_PERSON_OFFSET_Z };
            if (!closeCameraPosition(current, next)) writeThirdPersonPosition(ctx, pointer, next);
            const actual = readThirdPersonPosition(ctx, pointer);
            if (!closeCameraPosition(actual, next)) throw new Error('Camera move did not read back correctly.');
            record.applied = actual;
            applyThirdPersonVisuals(ctx);
            const animationReady = updateThirdPersonAnimation(ctx);
            sceneDiagnostics.camera = animationReady
                ? 'camera + character visibility + local animation applied'
                : 'camera + character visibility applied; waiting for body Animator';
            thirdPersonErrorReported = false;
        } catch (error) { stopThirdPersonWithError(error); }
    }
    function prepareThirdPersonRuntime(table, memory) {
        const view = memory?.buffer ? new DataView(memory.buffer) : null;
        if (!sceneBuildMatches(view) || !thirdPersonBuildMatches(view)) return false;
        if (cameraUpdateHooks.has(table)) return true;
        const original = table.get(18182);
        if (typeof original !== 'function') return false;
        const shimBytes = new Uint8Array([
            0,97,115,109,1,0,0,0,1,6,1,96,2,127,127,0,
            2,13,1,4,104,97,122,121,4,115,116,101,112,0,0,
            3,2,1,0,7,8,1,4,115,116,101,112,0,1,
            10,10,1,8,0,32,0,32,1,16,0,11
        ]);
        const step = (object, methodInfo) => {
            let local = false;
            if (thirdPerson || thirdPersonRecord) {
                try {
                    const v = new DataView(memory.buffer);
                    const controller = infoUint(v, object + 0x20);
                    const camera = infoUint(v, controller + 0x64);
                    const player = speedEngine.getPlayer();
                    local = Boolean(player && player.view.buffer === v.buffer &&
                        player.controller === controller &&
                        infoObjectNamed(v, object, 'CameraWalk', 0x44) &&
                        infoObjectNamed(v, camera, 'Camera', 12));
                    if (local && thirdPersonRecord) restoreThirdPerson(false);
                } catch (error) { stopThirdPersonWithError(error); }
            }
            // Restore the first-person base BEFORE the game's camera smoothing,
            // then apply the offset afterwards. Otherwise that smoothing can
            // repeatedly feed the third-person offset back into itself.
            const result = original(object, methodInfo);
            if (local) updateThirdPerson();
            return result;
        };
        const shim = new WebAssembly.Instance(new WebAssembly.Module(shimBytes), { hazy: { step } });
        table.set(18182, shim.exports.step);
        cameraUpdateHooks.set(table, { original, wrapper: shim.exports.step, shim });
        return true;
    }
    function setThirdPersonEnabled(enabled) {
        thirdPerson = Boolean(enabled);
        thirdPersonErrorReported = false;
        updateThirdPerson();
    }
    document.addEventListener('pointerlockchange', () => updateThirdPerson());
    window.addEventListener('blur', () => updateThirdPerson());
    document.addEventListener('visibilitychange', () => updateThirdPerson());
    window.addEventListener('pagehide', () => {
        thirdPerson = false;
        try { restoreThirdPerson(); } catch (_) {}
        for (const [table, hook] of cameraUpdateHooks) {
            try { if (table.get(18182) === hook.wrapper) table.set(18182, hook.original); } catch (_) {}
        }
        cameraUpdateHooks.clear();
        playerInfoByActor.clear(); playerInfoWeaponNames.clear();
        for (const allocation of sceneScratch.values()) { try { allocation.free(); } catch (_) {} }
        sceneScratch.clear();
    });

        [0x482ca0, 0x2a98f],   // moduleName -> "PhotonUnityNetworking.dll"

        [0x482ca8, 0x6c9b90], // methodPointers

    function photonBuildMatches(view) {
        return Boolean(view) && PHOTON_BUILD_CHECKS.every(([address, value]) =>
            address + 4 <= view.byteLength &&
            view.getUint32(address, true) === value
        );
    }

    function findPhotonRuntime() {
        const module = window.unityInstance?.Module || window.Module || null;
        const moduleView = currentMemoryView(module);

        const candidates = [];
        const seen = new Set();

        function add(table, view) {
            if (!table || typeof table.get !== 'function' || !view) return;
            if (table.length <= PHOTON_SEND_ALL_TABLE) return;
            if (seen.has(table)) return;
            seen.add(table);
            candidates.push({ table, view });
        }

        if (moduleView) {
            add(module?.asm?.__indirect_function_table, moduleView);
            add(module?.wasmExports?.__indirect_function_table, moduleView);
            add(module?.__indirect_function_table, moduleView);
            add(module?.wasmTable, moduleView);
        }

        for (const record of capturedRuntimes) {
            if (!record?.memory?.buffer) continue;
            add(record.table, new DataView(record.memory.buffer));
        }

        return candidates.find(item => photonBuildMatches(item.view)) || null;
    }

    function photonMethods() {
        const runtime = findPhotonRuntime();
        if (!runtime) return null;

        try {
            const getSendRate = runtime.table.get(PHOTON_GET_SEND_RATE_TABLE);
            const setSendRate = runtime.table.get(PHOTON_SET_SEND_RATE_TABLE);
            const getSerialRate = runtime.table.get(PHOTON_GET_SERIAL_RATE_TABLE);
            const setSerialRate = runtime.table.get(PHOTON_SET_SERIAL_RATE_TABLE);
            const getPing = runtime.table.get(PHOTON_GET_PING_TABLE);
            const sendAll = runtime.table.get(PHOTON_SEND_ALL_TABLE);

            if (![getSendRate, setSendRate, getSerialRate, setSerialRate, getPing, sendAll]
                .every(fn => typeof fn === 'function')) {
                return null;
            }

            return {
                runtime,
                getSendRate,
                setSendRate,
                getSerialRate,
                setSerialRate,
                getPing,
                sendAll
            };
        } catch (_) {
            return null;
        }
    }

    function applyPingBoost() {
        if (!pingBoostWanted) return false;

        const api = photonMethods();
        if (!api) return false;

        try {
            const currentSendRate = api.getSendRate(0) | 0;
            const currentSerialRate = api.getSerialRate(0) | 0;

            if (
                currentSendRate <= 0 || currentSendRate > 1000 ||
                currentSerialRate <= 0 || currentSerialRate > 1000
            ) {
                throw new Error('Ping Boost: unexpected Photon rate values.');
            }

            if (pingBoostOriginalSendRate === null) {
                pingBoostOriginalSendRate = currentSendRate;
            }
            if (pingBoostOriginalSerialRate === null) {
                pingBoostOriginalSerialRate = currentSerialRate;
            }

            // Higher PUN rates reduce client-side waiting/batching delay.
            // This cannot change the physical ISP/server round-trip distance.
            const targetSendRate = Math.min(
                120,
                Math.max(PING_BOOST_MIN_SEND_RATE, pingBoostOriginalSendRate)
            );
            const targetSerialRate = Math.min(
                targetSendRate,
                60,
                Math.max(PING_BOOST_MIN_SERIAL_RATE, pingBoostOriginalSerialRate)
            );

            if (currentSendRate !== targetSendRate) {
                api.setSendRate(targetSendRate, 0);
            }
            if (currentSerialRate !== targetSerialRate) {
                api.setSerialRate(targetSerialRate, 0);
            }

            const verifySend = api.getSendRate(0) | 0;
            const verifySerial = api.getSerialRate(0) | 0;

            if (verifySend !== targetSendRate || verifySerial !== targetSerialRate) {
                throw new Error('Ping Boost: Photon rates did not apply.');
            }

            pingBoost = true;
            pingBoostErrorReported = false;
            return true;
        } catch (error) {
            pingBoost = false;
            if (!pingBoostErrorReported) {
                pingBoostErrorReported = true;
                console.warn('[Hazy Client] Ping Boost:', error);
            }
            return false;
        }
    }

    function restorePingBoost() {
        const api = photonMethods();

        try {
            if (api) {
                if (Number.isInteger(pingBoostOriginalSendRate) &&
                    pingBoostOriginalSendRate > 0) {
                    api.setSendRate(pingBoostOriginalSendRate, 0);
                }
                if (Number.isInteger(pingBoostOriginalSerialRate) &&
                    pingBoostOriginalSerialRate > 0) {
                    api.setSerialRate(pingBoostOriginalSerialRate, 0);
                }
            }
        } catch (_) {}

        pingBoost = false;
        pingBoostOriginalSendRate = null;
        pingBoostOriginalSerialRate = null;
        pingBoostLastFlushAt = 0;
    }

    function setPingBoost(enabled) {
        pingBoostWanted = Boolean(enabled);

        if (!pingBoostWanted) {
            restorePingBoost();
        } else {
            applyPingBoost();
        }

        renderActiveMods();
        updateLiveTheme(performance.now());
    }

    function maintainPingBoost() {
        if (!pingBoostWanted) return;

        // Re-apply after reconnects / room transitions if Photon resets its rates.
        if (!applyPingBoost()) return;
    }

    function flushPingBoost(now) {
        if (!pingBoostWanted || !pingBoost) return;
        if (document.hidden || now - pingBoostLastFlushAt < PING_BOOST_FLUSH_MS) return;

        const api = photonMethods();
        if (!api) return;

        try {
            // Force any currently queued PUN commands out now instead of waiting
            // for the next normal send interval.
            api.sendAll(0);
            pingBoostLastFlushAt = now;
        } catch (error) {
            pingBoost = false;
            if (!pingBoostErrorReported) {
                pingBoostErrorReported = true;
                console.warn('[Hazy Client] Ping Boost flush:', error);
            }
        }
    }

    window.setInterval(maintainPingBoost, PING_BOOST_CHECK_MS);
    window.addEventListener('pagehide', () => {
        pingBoostWanted = false;
        restorePingBoost();
    });

    // ============================================================
    // CHICKEN ESP — EXACT CHICKENCONTROLLER + SKINNED MESH DISCOVERY
    // ============================================================
    // These entries/offsets come from the user's exported Kour 4.24 build.
    // ChickenController.Update is used only to remember live chicken instances.
    // The chicken's real SkinnedMeshRenderer -> shared Mesh -> submesh index
    // counts are then read through Unity's own methods. Those exact draw counts
    // feed the same through-wall shader path used by Player ESP.
    const CHICKEN_UPDATE_TABLE = 18336;
    const CHICKEN_RENDERER_OFFSET = 0x24;
    const CHICKEN_GET_SHARED_MESH_TABLE = 39371;
    const CHICKEN_GET_SUBMESH_COUNT_TABLE = 39397;
    const CHICKEN_GET_INDEX_COUNT_TABLE = 39460;
    const CHICKEN_COUNT_SLOTS = 12;
    const CHICKEN_STALE_MS = 2500;
    const CHICKEN_REFRESH_MS = 250;

    // Static build/layout guards. Fail closed on a different Kour build.
    const CHICKEN_BUILD_CHECKS = [
        [0x6bf7f0, CHICKEN_UPDATE_TABLE],      // ChickenController.Update
        [0x680834, 0x3c7590],                 // ChickenController field-offset table
        [0x3c75a4, CHICKEN_RENDERER_OFFSET],  // chickenMeshRenderer
        [0x6d79e8, CHICKEN_GET_SHARED_MESH_TABLE],
        [0x6d7a54, CHICKEN_GET_SUBMESH_COUNT_TABLE],
        [0x6d7b78, CHICKEN_GET_INDEX_COUNT_TABLE]
    ];

    const chickenControllers = new Map();
    const chickenDrawCounts = new Array(CHICKEN_COUNT_SLOTS).fill(0);
    let chickenHook = null;
    let chickenHookErrorReported = false;

    function validChickenPointer(view, pointer, bytes = 12) {
        return Number.isInteger(pointer) && pointer >= 1024 && pointer % 4 === 0 &&
            pointer + bytes <= view.byteLength;
    }

    function chickenBuildMatches(view) {
        return Boolean(view) && CHICKEN_BUILD_CHECKS.every(([address, value]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === value);
    }

    function restoreChickenHook() {
        if (!chickenHook) return;
        try {
            if (chickenHook.table.get(CHICKEN_UPDATE_TABLE) === chickenHook.wrapper) {
                chickenHook.table.set(CHICKEN_UPDATE_TABLE, chickenHook.original);
            }
        } catch (_) {}
        chickenHook = null;
    }

    function prepareChickenRuntime(table, memory) {
        if (!table || !memory || typeof table.get !== 'function' ||
            table.length <= CHICKEN_GET_INDEX_COUNT_TABLE) return false;

        const view = new DataView(memory.buffer);
        if (!chickenBuildMatches(view)) return false;

        if (chickenHook && chickenHook.table === table) {
            if (table.get(CHICKEN_UPDATE_TABLE) !== chickenHook.wrapper) {
                throw new Error('Chicken ESP: another script changed the chicken update hook.');
            }
            return true;
        }

        restoreChickenHook();

        const original = table.get(CHICKEN_UPDATE_TABLE);
        if (typeof original !== 'function') {
            throw new Error('Chicken ESP: ChickenController.Update is unavailable.');
        }

        // Typed (i32, i32) -> void WASM trampoline, the exact signature of Update.
        const shimBytes = new Uint8Array([
            0,97,115,109,1,0,0,0,
            1,6,1,96,2,127,127,0,
            2,13,1,4,104,97,122,121,4,115,116,101,112,0,0,
            3,2,1,0,
            7,8,1,4,115,116,101,112,0,1,
            10,10,1,8,0,32,0,32,1,16,0,11
        ]);

        const step = (controller, methodInfo) => {
            // Remember the live instance. We don't modify chicken movement/state.
            if (Number.isInteger(controller) && controller >= 1024) {
                chickenControllers.set(controller, performance.now());
            }
            return original(controller, methodInfo);
        };

        const shimModule = new WebAssembly.Module(shimBytes);
        const shim = new WebAssembly.Instance(shimModule, { hazy: { step } });
        const wrapper = shim.exports.step;

        table.set(CHICKEN_UPDATE_TABLE, wrapper);
        if (table.get(CHICKEN_UPDATE_TABLE) !== wrapper) {
            throw new Error('Chicken ESP: Update hook installation failed.');
        }

        chickenHook = { table, memory, original, wrapper, shim };
        chickenHookErrorReported = false;
        return true;
    }

    function findChickenRuntime() {
        if (chickenHook) return chickenHook;

        const module = window.unityInstance?.Module || window.Module || null;
        const view = currentMemoryView(module);
        const candidates = [
            module?.asm?.__indirect_function_table,
            module?.wasmExports?.__indirect_function_table,
            module?.__indirect_function_table,
            module?.wasmTable
        ];

        for (const record of capturedRuntimes) {
            candidates.push(record.table);
        }

        for (const table of candidates) {
            if (!table || typeof table.get !== 'function' ||
                table.length <= CHICKEN_GET_INDEX_COUNT_TABLE) continue;

            let memory = null;
            for (const record of capturedRuntimes) {
                if (record.table === table) {
                    memory = record.memory;
                    break;
                }
            }
            if (!memory && module?.wasmMemory) memory = module.wasmMemory;
            if (!memory?.buffer) continue;

            try {
                if (prepareChickenRuntime(table, memory)) return chickenHook;
            } catch (error) {
                if (!chickenHookErrorReported) {
                    chickenHookErrorReported = true;
                    console.warn('[Hazy Client] Chicken ESP hook:', error);
                }
            }
        }

        // If the runtime table was not captured yet, do nothing rather than guess.
        if (view && !chickenBuildMatches(view)) return null;
        return chickenHook;
    }

    function refreshChickenDrawCounts() {
        // Keep the hook warm even while the visual toggle is off, so Chicken ESP
        // is immediate when enabled later in the same match.
        const runtime = findChickenRuntime();
        if (!runtime || !chickenESP || document.hidden) return;

        try {
            const table = runtime.table;
            const view = new DataView(runtime.memory.buffer);
            if (!chickenBuildMatches(view)) return;

            const getSharedMesh = table.get(CHICKEN_GET_SHARED_MESH_TABLE);
            const getSubMeshCount = table.get(CHICKEN_GET_SUBMESH_COUNT_TABLE);
            const getIndexCount = table.get(CHICKEN_GET_INDEX_COUNT_TABLE);

            if (![getSharedMesh, getSubMeshCount, getIndexCount]
                .every(fn => typeof fn === 'function')) {
                throw new Error('Chicken ESP: mesh functions unavailable.');
            }

            const now = performance.now();
            const counts = new Set();

            for (const [controller, lastSeen] of chickenControllers) {
                if (now - lastSeen > CHICKEN_STALE_MS) {
                    chickenControllers.delete(controller);
                    continue;
                }

                if (!validChickenPointer(view, controller, CHICKEN_RENDERER_OFFSET + 4) ||
                    view.getUint32(controller + 8, true) === 0) {
                    chickenControllers.delete(controller);
                    continue;
                }

                const renderer = view.getUint32(controller + CHICKEN_RENDERER_OFFSET, true);
                if (!validChickenPointer(view, renderer, 12) ||
                    view.getUint32(renderer + 8, true) === 0) continue;

                const mesh = getSharedMesh(renderer, 0);
                if (!validChickenPointer(view, mesh, 12) ||
                    view.getUint32(mesh + 8, true) === 0) continue;

                const subMeshCount = getSubMeshCount(mesh, 0) | 0;
                if (subMeshCount <= 0 || subMeshCount > CHICKEN_COUNT_SLOTS) continue;

                for (let sub = 0; sub < subMeshCount; sub++) {
                    const count = getIndexCount(mesh, sub, 0) >>> 0;
                    // drawElements count is this exact mesh index count.
                    if (count >= 3 && count <= 2000000) counts.add(count);
                }
            }

            const sorted = [...counts].sort((a, b) => a - b).slice(0, CHICKEN_COUNT_SLOTS);
            chickenDrawCounts.fill(0);
            for (let i = 0; i < sorted.length; i++) chickenDrawCounts[i] = sorted[i];
        } catch (error) {
            if (!chickenHookErrorReported) {
                chickenHookErrorReported = true;
                console.warn('[Hazy Client] Chicken ESP mesh discovery:', error);
            }
        }
    }

    // Runtime discovery is cheap; mesh reads only occur while Chicken ESP is on.
    window.setInterval(() => {
        try { findChickenRuntime(); } catch (_) {}
        refreshChickenDrawCounts();
    }, CHICKEN_REFRESH_MS);

    window.addEventListener('pagehide', restoreChickenHook);

    function performAirJump() {
        const ctx = speedEngine.getPlayer();
        if (!ctx) return false;
        let view = ctx.view;
        // Confirm the jump method slot and relevant field layout too.
        const checks = [
            [0x6beaa4, AIR_JUMP_TABLE],
            [0x3c67fc, 0x9c], [0x3c6808, 0xa8],
            [0x3c6824, 0xb4], [0x3c6834, 0xe4], [0x3c6838, 0xe5]
        ];
        if (!checks.every(([address, value]) => view.getUint32(address, true) === value)) {
            throw new Error('Air Jump does not match this game build.');
        }
        const table = findJumpTable(ctx);
        if (!table) throw new Error('Jump functions unavailable; save the script and refresh Kour.');
        const jump = table.get(AIR_JUMP_TABLE);
        const getVelocity = table.get(GET_VELOCITY_TABLE);
        const setVelocity = table.get(SET_VELOCITY_TABLE);
        if (![jump, getVelocity, setVelocity].every(fn => typeof fn === 'function')) {
            throw new Error('The expected jump functions are unavailable.');
        }
        const controller = ctx.controller;
        const body = view.getUint32(controller + 0xb0, true);
        const velocity = controller + 0xb4;
        const height = view.getFloat32(controller + 0x9c, true);
        const additive = view.getUint8(controller + 0xa8);
        if (!Number.isFinite(height) || height <= 0 || height > 100 || additive > 1) {
            throw new Error('Unexpected jump settings; no jump applied.');
        }

        // FixedUpdate reads velocity from the actual physics body. Read it
        // first, run the game's Jump(force) routine, then write it back.
        // Changing only the cached controller velocity would be overwritten.
        getVelocity(velocity, body, 0);
        view = currentMemoryView(ctx.module);
        const components = [0, 4, 8].map(offset => view.getFloat32(velocity + offset, true));
        if (!components.every(value => Number.isFinite(value) && Math.abs(value) < 10000)) {
            throw new Error('Unexpected player velocity; no jump applied.');
        }
        // Use the current jump strength (including High Jump when enabled),
        // without stacking upward speed. Restore the additive flag afterward.
        view.setUint8(controller + 0xa8, 0);
        try {
            jump(controller, 1, 0);
        } finally {
            currentMemoryView(ctx.module).setUint8(controller + 0xa8, additive);
        }
        view = currentMemoryView(ctx.module);
        // Consume this jump request even when the game's autoJump is enabled.
        view.setUint8(controller + 0xe5, 0);
        setVelocity(body, velocity, 0);
        return true;
    }

    function editingText(event) {
        const nodes = [event.target, document.activeElement];
        return nodes.some(node => node && (
            node.isContentEditable ||
            /^(INPUT|TEXTAREA|SELECT)$/.test(node.tagName || '') ||
            node.closest?.('[role="textbox"]')
        ));
    }

    // ============================================================
    // AUTO WALK — HOLD KOUR'S NORMAL FORWARD INPUT
    // ============================================================
    // We feed a normal W key through the pointer-locked game target so Kour's
    // own movement/camera rotation continues to decide the world direction.
    const AUTO_WALK_CHECK_MS = 50;
    let autoWalkSyntheticDown = false;
    let physicalWDown = false;

    function dispatchAutoWalkKey(down) {
        const target = document.pointerLockElement || document;
        const event = new KeyboardEvent(down ? 'keydown' : 'keyup', {
            key: 'w',
            code: 'KeyW',
            bubbles: true,
            cancelable: true,
            composed: true
        });

        // Unity/Emscripten builds sometimes still read the legacy numeric keys.
        try {
            Object.defineProperty(event, 'keyCode', { value: 87 });
            Object.defineProperty(event, 'which', { value: 87 });
        } catch (_) {}

        target.dispatchEvent(event);
    }

    function updateAutoWalkInput() {
        const shouldWalk =
            autoWalk &&
            Boolean(document.pointerLockElement) &&
            !document.hidden &&
            (typeof document.hasFocus !== 'function' || document.hasFocus()) &&
            !editingText({ target: document.activeElement });

        if (shouldWalk) {
            if (!physicalWDown && !autoWalkSyntheticDown) {
                dispatchAutoWalkKey(true);
                autoWalkSyntheticDown = true;
            }
        } else if (autoWalkSyntheticDown) {
            // Do not synthesize W-up over a real W key the user is still holding.
            if (!physicalWDown) dispatchAutoWalkKey(false);
            autoWalkSyntheticDown = false;
        }
    }

    function setAutoWalkEnabled(enabled) {
        autoWalk = Boolean(enabled);
        updateAutoWalkInput();
    }

    window.addEventListener('keydown', event => {
        if (event.isTrusted && event.code === 'KeyW') {
            physicalWDown = true;
        }
    }, true);

    window.addEventListener('keyup', event => {
        if (!event.isTrusted || event.code !== 'KeyW') return;
        physicalWDown = false;

        // A real W-up also releases Unity's state. Reassert Auto Walk on the
        // next frame when it is still enabled.
        if (autoWalk) {
            autoWalkSyntheticDown = false;
            requestAnimationFrame(updateAutoWalkInput);
        }
    }, true);

    window.setInterval(updateAutoWalkInput, AUTO_WALK_CHECK_MS);
    window.addEventListener('blur', updateAutoWalkInput);
    document.addEventListener('pointerlockchange', updateAutoWalkInput);
    document.addEventListener('visibilitychange', updateAutoWalkInput);
    window.addEventListener('pagehide', () => setAutoWalkEnabled(false));

    // ============================================================
    // AUTO JUMP — JUMP ON EACH REAL LANDING
    // ============================================================
    const AUTO_JUMP_MIN_INTERVAL_MS = 90;
    let lastAutoJumpAt = -Infinity;
    let autoJumpErrorReported = false;

    function updateAutoJump(now) {
        if (!autoJump ||
            fly || noClipFly || jetpack ||
            !document.pointerLockElement ||
            document.hidden ||
            (typeof document.hasFocus === 'function' && !document.hasFocus()) ||
            now - lastAutoJumpAt < AUTO_JUMP_MIN_INTERVAL_MS) {
            return;
        }

        try {
            const ctx = speedEngine.getPlayer();
            if (!ctx) return;

            const view = currentMemoryView(ctx.module);
            if (!view) return;

            // Exact PlayerController.onGround field from Kour 4.24.
            if (view.getUint32(0x3c6834, true) !== 0xe4) {
                throw new Error('Auto Jump: grounded field does not match this Kour build.');
            }

            const grounded = view.getUint8(ctx.controller + 0xe4);
            if (grounded !== 0 && grounded !== 1) {
                throw new Error('Auto Jump: unexpected grounded state.');
            }
            if (!grounded) return;

            // Keep Auto Jump compatible with High Jump.
            if (highJump) highJumpEngine.tick();

            if (performAirJump()) {
                lastAutoJumpAt = now;
                autoJumpErrorReported = false;
            }
        } catch (error) {
            autoJump = false;
            renderActiveMods();
            if (!autoJumpErrorReported) {
                autoJumpErrorReported = true;
                console.warn('[Hazy Client] Auto Jump stopped:', error);
            }
        }
    }

    // ============================================================
    // BUNNY HOP — LANDING JUMPS WHILE MOVING
    // ============================================================
    // Auto Jump jumps even from a standstill. Bunny Hop only fires while the
    // player's real world-space movement input is nonzero, preserving the
    // horizontal Rigidbody velocity through the native Jump() call.
    const BUNNY_HOP_INPUT_MIN = 0.16;
    let bunnyHopErrorReported = false;

    function updateBunnyHop(now) {
        if (!bunnyHop ||
            fly || noClipFly || jetpack ||
            !document.pointerLockElement ||
            document.hidden ||
            (typeof document.hasFocus === 'function' && !document.hasFocus()) ||
            now - lastAutoJumpAt < AUTO_JUMP_MIN_INTERVAL_MS) {
            return;
        }

        try {
            const ctx = speedEngine.getPlayer();
            if (!ctx) return;

            const view = currentMemoryView(ctx.module);
            if (!view) return;

            const layout = [
                [0x3c6828, 0xc0], // inputDir
                [0x3c6834, 0xe4]  // onGround
            ];
            if (!layout.every(([address, value]) =>
                address + 4 <= view.byteLength &&
                view.getUint32(address, true) === value)) {
                throw new Error('Bunny Hop: movement layout does not match this Kour build.');
            }

            const grounded = view.getUint8(ctx.controller + 0xe4);
            if (grounded !== 0 && grounded !== 1) {
                throw new Error('Bunny Hop: unexpected grounded state.');
            }
            if (!grounded) return;

            const inputX = view.getFloat32(ctx.controller + 0xc0, true);
            const inputZ = view.getFloat32(ctx.controller + 0xc8, true);
            if (![inputX, inputZ].every(Number.isFinite) ||
                Math.hypot(inputX, inputZ) < BUNNY_HOP_INPUT_MIN) {
                return;
            }

            if (highJump) highJumpEngine.tick();

            if (performAirJump()) {
                // Share the landing cooldown with Auto Jump so enabling both can
                // never double-trigger the same landing.
                lastAutoJumpAt = now;
                bunnyHopErrorReported = false;
            }
        } catch (error) {
            bunnyHop = false;
            renderActiveMods();
            if (!bunnyHopErrorReported) {
                bunnyHopErrorReported = true;
                console.warn('[Hazy Client] Bunny Hop stopped:', error);
            }
        }
    }

    // ============================================================
    // DOUBLE JUMP + GLIDE — SHARED VERIFIED LOCAL MOVEMENT PATH
    // ============================================================
    // Double Jump allows exactly one extra native jump after leaving the ground.
    // Glide only caps downward velocity while Space is held; it never changes X/Z.
    const GLIDE_MAX_FALL_SPEED = -3.5;
    const GLIDE_UPDATE_MS = 24;
    const DOUBLE_JUMP_STATE_CHECK_MS = 40;
    let doubleJumpUsed = false;
    let doubleJumpErrorReported = false;
    let glideErrorReported = false;
    let lastGlideUpdateAt = -Infinity;
    let lastDoubleJumpGroundCheckAt = -Infinity;

    function readJumpMovementState(needVelocity = false) {
        const ctx = speedEngine.getPlayer();
        if (!ctx) return null;
        let view = currentMemoryView(ctx.module);
        if (!view) return null;
        const checks = [
            [0x3c6820, 0xb0], [0x3c6824, 0xb4], [0x3c6834, 0xe4],
            [0x6dec38, GET_VELOCITY_TABLE], [0x6dec3c, SET_VELOCITY_TABLE]
        ];
        if (!checks.every(([address, value]) =>
            address + 4 <= view.byteLength && view.getUint32(address, true) === value)) {
            throw new Error('Double Jump / Glide do not match this Kour build.');
        }
        const groundedByte = view.getUint8(ctx.controller + 0xe4);
        if (groundedByte !== 0 && groundedByte !== 1) {
            throw new Error('Unexpected grounded state.');
        }
        const result = { ctx, grounded: groundedByte === 1 };
        if (!needVelocity) return result;

        const table = findJumpTable(ctx);
        if (!table) throw new Error('Movement functions unavailable; save and refresh Kour.');
        const getVelocity = table.get(GET_VELOCITY_TABLE);
        const setVelocity = table.get(SET_VELOCITY_TABLE);
        if (typeof getVelocity !== 'function' || typeof setVelocity !== 'function') {
            throw new Error('Expected movement functions are unavailable.');
        }
        const body = view.getUint32(ctx.controller + 0xb0, true);
        const velocityPointer = ctx.controller + 0xb4;
        if (!body) throw new Error('Local Rigidbody is unavailable.');
        getVelocity(velocityPointer, body, 0);
        view = currentMemoryView(ctx.module);
        const velocity = {
            x: view.getFloat32(velocityPointer, true),
            y: view.getFloat32(velocityPointer + 4, true),
            z: view.getFloat32(velocityPointer + 8, true)
        };
        if (!Object.values(velocity).every(value => Number.isFinite(value) && Math.abs(value) <= 200)) {
            throw new Error('Unexpected local velocity.');
        }
        return { ...result, view, table, body, velocityPointer, velocity, setVelocity };
    }

    function updateDoubleJumpGroundState(now) {
        if (!doubleJump || now - lastDoubleJumpGroundCheckAt < DOUBLE_JUMP_STATE_CHECK_MS) return;
        lastDoubleJumpGroundCheckAt = now;
        try {
            const state = readJumpMovementState(false);
            if (state?.grounded) doubleJumpUsed = false;
            doubleJumpErrorReported = false;
        } catch (error) {
            doubleJump = false;
            doubleJumpUsed = false;
            renderActiveMods();
            if (!doubleJumpErrorReported) {
                doubleJumpErrorReported = true;
                console.warn('[Hazy Client] Double Jump stopped:', error);
            }
        }
    }

    function updateGlide(now) {
        if (!glide || !spaceHeld || fly || noClipFly || jetpack ||
            !document.pointerLockElement || document.hidden ||
            (typeof document.hasFocus === 'function' && !document.hasFocus()) ||
            now - lastGlideUpdateAt < GLIDE_UPDATE_MS) return;
        lastGlideUpdateAt = now;
        try {
            const state = readJumpMovementState(true);
            if (!state || state.grounded || state.velocity.y >= GLIDE_MAX_FALL_SPEED) return;
            const view = currentMemoryView(state.ctx.module);
            view.setFloat32(state.velocityPointer, Math.fround(state.velocity.x), true);
            view.setFloat32(state.velocityPointer + 4, Math.fround(GLIDE_MAX_FALL_SPEED), true);
            view.setFloat32(state.velocityPointer + 8, Math.fround(state.velocity.z), true);
            state.setVelocity(state.body, state.velocityPointer, 0);
            glideErrorReported = false;
        } catch (error) {
            glide = false;
            renderActiveMods();
            if (!glideErrorReported) {
                glideErrorReported = true;

    // ============================================================
    // ONE SHOT — CURRENT OWN-PLAYER DAMAGE, BUILD 4.24
    // ============================================================

    // Damage uses a signed 32-bit integer. The requested
    // 99,999,999,999 cannot be represented by that type.
    // Leave ample room for the game's hit multipliers.
    const ONE_SHOT_DAMAGE = 999999;
    const ONE_SHOT_CHECK_MS = 250;

    function createOneShotController(getPlayer, getTable, notify) {
        const CURRENT_WEAPON = 0xe0;
        const CURRENT_DAMAGE = 0x148;
        const SET_DAMAGE_SLOT = 15310;
        const GET_DAMAGE_SLOT = 16074;

        const checks = [
            [0x3c9e68, CURRENT_WEAPON],
            [0x3c9e80, CURRENT_DAMAGE],
            [0x3caa40, 0x6c],
            [0x3caa44, 0x70],
            [0x3caa70, 0x9c],
            [0x6bccec, SET_DAMAGE_SLOT],
            [0x6bccdc, GET_DAMAGE_SLOT],
            [0x63c5b8, 0x80080006]
        ];

        let wanted = false;
        let saved = null;
        let errorReported = false;

        function valid(view, pointer, length = 4) {
            return Number.isInteger(pointer) && pointer >= 1024 &&
                pointer % 4 === 0 && pointer + length <= view.byteLength;
        }

        function uint(view, pointer) {
            return valid(view, pointer)
                ? view.getUint32(pointer, true)
                : 0;
        }

        function isWeaponClass(view, klass) {
            for (let depth = 0; depth < 12; depth++) {
                if (!valid(view, klass, 0x60) ||
                    uint(view, klass + 0x3c) !== klass) return false;

                const name = uint(view, klass + 0x08);
                if (name >= 1024 && name + 7 <= view.byteLength) {
                    const expected = 'Weapon';
                    let match = true;
                    for (let i = 0; i < expected.length; i++) {
                        if (view.getUint8(name + i) !== expected.charCodeAt(i)) {
                            match = false;
                            break;
                        }
                    }
                    if (match && view.getUint8(name + 6) === 0) return true;
                }

                const parent = uint(view, klass + 0x2c);
                if (parent === klass) return false;
                klass = parent;
            }
            return false;
        }

        function resolve() {
            // The shared resolver already checks the exact build,
            // Shooter.current, player ownership and native objects.
            const player = getPlayer();
            if (!player) return null;

            const view = currentMemoryView(player.module);
            if (!view) return null;

            if (!checks.every(([address, value]) =>
                uint(view, address) === value)) {
                throw new Error('One Shot: different damage layout; not applied.');
            }

            const weapon = uint(view, player.shooter + CURRENT_WEAPON);
            if (!valid(view, weapon, 0xa0) ||
                !uint(view, weapon + 0x08) ||
                !isWeaponClass(view, uint(view, weapon)) ||
                view.getUint8(weapon + 0x6c) !== 1 ||
                uint(view, weapon + 0x70) !== player.shooter) return null;

            const damagePointer = player.shooter + CURRENT_DAMAGE;
            if (!valid(view, damagePointer, 20) ||
                view.getUint8(damagePointer + 8) !== 1) return null;

            const table = getTable(player);
            if (!table) {
                throw new Error('One Shot: save the script and refresh Kour.');
            }

            const setDamage = table.get(SET_DAMAGE_SLOT);
            const getDamage = table.get(GET_DAMAGE_SLOT);
            if (typeof setDamage !== 'function' ||
                typeof getDamage !== 'function') {
                throw new Error('One Shot: expected damage functions unavailable.');
            }

            return {
                ...player,
                weapon,
                weaponNative: uint(view, weapon + 0x08),
                damagePointer,
                setDamage,
                getDamage
            };
        }

        function sameWeapon(a, b) {
            return Boolean(a && b && a.module === b.module &&
                a.shooter === b.shooter && a.shooterNative === b.shooterNative &&
                a.weapon === b.weapon && a.weaponNative === b.weaponNative);
        }

        function readDamage(ctx) {
            const value = ctx.getDamage(ctx.damagePointer, 0);
            if (!Number.isInteger(value) || value < 0 || value > 2147483647) {
                throw new Error('One Shot: unexpected damage value.');
            }
            return value;
        }

        function writeDamage(ctx, value) {
            // Use the game's own integer conversion routine, not a raw
            // integer overwrite of its structured cached damage value.
            ctx.setDamage(ctx.damagePointer, value, 0);
            if (readDamage(ctx) !== value) {
                throw new Error('One Shot: damage did not read back correctly.');
            }
        }

        function restore() {
            if (!saved) return;
            const ctx = resolve();
            if (sameWeapon(saved, ctx) && readDamage(ctx) === ONE_SHOT_DAMAGE) {
                writeDamage(ctx, saved.original);
            }
        }

        function tick() {
            if (!wanted) return;
            try {
                const ctx = resolve();
                if (!ctx) return;

                const current = readDamage(ctx);
                if (!sameWeapon(saved, ctx)) {
                    // Never capture our own previous boost as normal damage.
                    if (current === ONE_SHOT_DAMAGE) return;
                    saved = { ...ctx, original: current };
                } else if (current !== ONE_SHOT_DAMAGE) {
                    // A weapon/game setting has refreshed the normal damage.
                    saved.original = current;
                }

                // Zero-damage weapons stay zero-damage.
                if (saved.original <= 0) return;
                if (current !== ONE_SHOT_DAMAGE) {
                    writeDamage(ctx, ONE_SHOT_DAMAGE);
                }
            } catch (error) {
                try { restore(); } catch (_) {}
                saved = null;
                wanted = false;
                notify(false);
                if (!errorReported) {
                    errorReported = true;
                    console.warn('[Hazy Client] One Shot stopped:', error);
                }
            }
        }

        function setEnabled(enabled) {
            enabled = Boolean(enabled);
            if (enabled === wanted) return;
            wanted = enabled;
            notify(wanted);

            if (wanted) {
                errorReported = false;
                tick();
            } else {
                try { restore(); }
                catch (error) {
                    console.warn('[Hazy Client] One Shot restore failed:', error);
                }
                saved = null;
            }
        }

        return { tick, setEnabled, get wanted() { return wanted; } };
    }

    const oneShotEngine = createOneShotController(
        () => speedEngine.getPlayer(),
        ctx => findJumpTable(ctx),
        enabled => {
            oneShot = enabled;
            renderActiveMods();
        }
    );

    window.setInterval(() => oneShotEngine.tick(), ONE_SHOT_CHECK_MS);
    window.addEventListener('pagehide', () => oneShotEngine.setEnabled(false));

    const PLAYER_VERTEX_RANGES = [
        { min:1481, max:1483 },
        { min:1553, max:1555 },
        { min:5615, max:5617 },
        { min:3875, max:3877 }
    ];

    // ============================================================
    // MECHANICAL WEAPON SETTINGS — LOCAL PLAYER, KOUR 4.24 ONLY
    // ============================================================
    // The source archive's Shooter.Shoot uses 1/currentFireRate as the
    // interval. No Spread removes the instability used to randomize the ray;
    // it does not suppress camera recoil. The native reload coroutine still
    // performs ammo/UI/animation completion — only its remaining wait is cut.
    const RAPID_FIRE_MULTIPLIER = 3;
    const RAPID_FIRE_SOFT_CAP = 30; // Never lower an already faster native rate.
    const MECHANICS_CHECKS = [
        [0x3c9e68, 0xe0], [0x3c9e74, 0x100], [0x3c9e7c, 0x130],
        [0x3caa04, 0x30], [0x3caa5c, 0x88],
        [0x3caa78, 0xa4], [0x3caa7c, 0xa8], [0x3c6840, 0xe8],
        [0x3caa40, 0x6c], [0x3caa44, 0x70],
        [0x6bcc4c, 15314], [0x6bcc50, 15313],
        [0x6bea70, 17482], [0x6c1d90, 20702], [0x6c1da0, 20706],
        [0x6c1eb0, 20774],
        [0x3c9d60, 8], [0x3c9d68, 0x10], [0x3c9d6c, 0x14]
    ];
    const mechanicsHooks = new Map();
    let mechanicsRecord = null;
    let mechanicsScratchRecord = null;
    let mechanicsErrorReported = false;
    let mechanicsBusy = false;

    function anyWeaponSettingsOn() {
        return noSpread || rapidFire || instantReload;
    }
    function mechanicsBuildMatches(view) {
        return Boolean(view) && MECHANICS_CHECKS.every(([p, n]) =>
            p + 4 <= view.byteLength && view.getUint32(p, true) === n);
    }
    function mechanicsClose(a, b) {
        return Number.isFinite(a) && Number.isFinite(b) &&
            Math.abs(a - b) <= Math.max(0.000001, Math.abs(b) * 0.000001);
    }
    function releaseMechanicsScratch() {
        if (!mechanicsScratchRecord) return;
        try { mechanicsScratchRecord.free(); } catch (_) {}
        mechanicsScratchRecord = null;
    }
    function mechanicsScratch(runtime) {
        if (mechanicsScratchRecord?.table === runtime.table &&
            infoValid(sceneView(runtime), mechanicsScratchRecord.pointer, 32)) {
            return mechanicsScratchRecord.pointer;
        }
        releaseMechanicsScratch();
        for (const owner of [runtime.module, runtime.exports,
            runtime.module?.asm, runtime.module?.wasmExports]) {
            const malloc = owner?._malloc || owner?.malloc;
            const free = owner?._free || owner?.free;
            if (typeof malloc !== 'function' || typeof free !== 'function') continue;
            const pointer = Reflect.apply(malloc, owner, [32]) >>> 0;
            if (!infoValid(sceneView(runtime), pointer, 32)) {
                if (pointer) Reflect.apply(free, owner, [pointer]);
                throw new Error('Weapon-settings scratch allocation failed.');
            }
            mechanicsScratchRecord = { table: runtime.table, pointer,
                free: () => Reflect.apply(free, owner, [pointer]) };
            return pointer;
        }
        throw new Error('Weapon-settings allocator unavailable. Save and refresh Kour.');
    }
    function mechanicsRead(ctx, field) {
        const view = sceneView(ctx.runtime);
        if (!infoValid(view, field.address, field.protected ? 24 : 4)) {
            throw new Error('Weapon-settings field is unavailable.');
        }
        if (!field.protected) return view.getFloat32(field.address, true);
        // Embedded ObscuredFloat has no boxed-object header.
        if (view.getUint8(field.address + 12) !== 1) return null;
        const scratch = mechanicsScratch(ctx.runtime);
        const fresh = sceneView(ctx.runtime);
        new Uint8Array(fresh.buffer, fresh.byteOffset, fresh.byteLength)
            .copyWithin(scratch, field.address, field.address + 24);
        const value = sceneFunction(ctx.runtime, 15313)(scratch, 0);
        return Number.isFinite(value) ? value : null;
    }
    function mechanicsWrite(ctx, field, value) {
        if (!Number.isFinite(value) || value < 0 || value > 100000) {
            throw new Error('Weapon setting outside its supported range.');
        }
        value = Math.fround(value);
        if (field.protected) {
            // Native float -> ObscuredFloat conversion initializes the whole struct.
            sceneFunction(ctx.runtime, 15314)(field.address, value, 0);
        } else {
            sceneView(ctx.runtime).setFloat32(field.address, value, true);
        }
        if (!mechanicsClose(mechanicsRead(ctx, field), value)) {
            throw new Error('Weapon setting did not read back correctly.');
        }
    }
    function resolveMechanicsPlayer() {
        const local = speedEngine.getPlayer();
        const runtime = findSceneRuntime();
        if (!local || !runtime) return null;
        const view = sceneView(runtime);
        if (view.buffer !== local.view.buffer) return null;
        if (!mechanicsBuildMatches(view)) {
            throw new Error('Weapon settings do not match this Kour build.');
        }
        const weapon = infoUint(view, local.shooter + 0xe0);
        if (!infoValid(view, weapon, 0x100) || !infoUint(view, weapon + 8) ||
            !infoDerivesFrom(view, infoUint(view, weapon), 'Weapon') ||
            view.getUint8(weapon + 0x6c) !== 1 ||
            infoUint(view, weapon + 0x70) !== local.shooter) return null;
        return { ...local, runtime, weapon, weaponClass: infoUint(view, weapon),
            weaponNative: infoUint(view, weapon + 8) };
    }
    function sameMechanicsWeapon(a, b) {
        return Boolean(a && b && a.runtime.table === b.runtime.table &&
            a.shooter === b.shooter && a.shooterNative === b.shooterNative &&
            a.controller === b.controller && a.controllerNative === b.controllerNative &&
            a.weapon === b.weapon && a.weaponNative === b.weaponNative &&
            a.weaponClass === b.weaponClass);
    }
    function mechanicsFieldStillOwned(record, field) {
        const view = sceneView(record.runtime);
        if (!mechanicsBuildMatches(view) ||
            !infoObjectNamed(view, record.shooter, 'Shooter', 0x208) ||
            infoUint(view, record.shooter + 8) !== record.shooterNative) return false;
        if (field.owner === 'weapon') {
            return infoValid(view, record.weapon, 0x100) &&
                infoUint(view, record.weapon) === record.weaponClass &&
                infoUint(view, record.weapon + 8) === record.weaponNative &&
                infoUint(view, record.weapon + 0x70) === record.shooter &&
                view.getUint8(record.weapon + 0x6c) === 1;
        }
        // Cached stats belong to the current weapon only. Never restore a
        // previous gun's fire/reload settings over the gun just equipped.
        if (infoUint(view, record.shooter + 0xe0) !== record.weapon ||
            infoUint(view, record.weapon + 8) !== record.weaponNative) return false;
        if (field.owner === 'controller') {
            return infoObjectNamed(view, record.controller, 'PlayerController', 0x12c) &&
                infoUint(view, record.controller + 8) === record.controllerNative &&
                infoUint(view, record.controller + 0x4c) === record.shooter;
        }
        return true;
    }
    function restoreMechanicsFields(record, group = null) {
        if (!record) return;
        let firstError = null;
        for (const [key, field] of record.fields) {
            if (group && field.group !== group) continue;
            try {
                if (mechanicsFieldStillOwned(record, field) &&
                    mechanicsClose(mechanicsRead(record, field), field.applied)) {
                    mechanicsWrite(record, field, field.original);
                }
                record.fields.delete(key);
            } catch (error) { if (!firstError) firstError = error; }
        }
        if (firstError) throw firstError;
    }
    function applyMechanicsField(ctx, key, group, owner, offset, protectedValue, target) {
        const address = ctx[owner] + offset;
        const definition = { address, owner, protected: protectedValue, group };
        const current = mechanicsRead(ctx, definition);
        // A weapon still being initialized is not a usable settings target yet.
        if (current === null) return;
        if (!Number.isFinite(current) || current < 0 || current > 100000) {
            throw new Error(`Unexpected ${key} value; no guessed write applied.`);
        }
        let field = mechanicsRecord.fields.get(key);
        if (!field) {
            field = { ...definition, original: current, applied: null };
            mechanicsRecord.fields.set(key, field);
        } else if (!mechanicsClose(current, field.applied)) {
            // Keep a legitimate in-game refresh as the new baseline. Do not
            // multiply our previously boosted number again on every update.
            field.original = current;
        }
        const desired = Math.fround(target(field.original));
        field.applied = desired; // Recorded before write, so cleanup can retry.
        if (!mechanicsClose(current, desired)) mechanicsWrite(ctx, field, desired);
    }
    function failWeaponSettings(error) {
        noSpread = false;
        rapidFire = false;
        instantReload = false;
        try {
            restoreMechanicsFields(mechanicsRecord);
            mechanicsRecord = null;
        } catch (_) {}
        renderActiveMods();
        if (!mechanicsErrorReported) {
            mechanicsErrorReported = true;
            reportSceneError('Weapon settings', error);
        }
    }
    function updateWeaponSettings(object = 0, kind = '') {
        if (mechanicsBusy || (!anyWeaponSettingsOn() && !mechanicsRecord)) return;
        mechanicsBusy = true;
        try {
            const ctx = resolveMechanicsPlayer();
            if (object && (!ctx || (kind === 'controller'
                ? object !== ctx.controller : object !== ctx.shooter))) return;
            if (!ctx) {
                // Keep only verifiably live old values eligible for restoration.
                if (mechanicsRecord) {
                    restoreMechanicsFields(mechanicsRecord);
                    mechanicsRecord = null;
                }
                return;
            }
            if (!sameMechanicsWeapon(mechanicsRecord, ctx)) {
                restoreMechanicsFields(mechanicsRecord);
                mechanicsRecord = { ...ctx, fields: new Map() };
            }
            prepareWeaponSettingsRuntime(ctx.runtime.table,
                ctx.runtime.memory || { buffer: sceneView(ctx.runtime).buffer });
            if (noSpread) {
                applyMechanicsField(ctx, 'spread scale', 'spread', 'weapon', 0xa4, false, () => 0);
                applyMechanicsField(ctx, 'base spread scale', 'spread', 'weapon', 0xa8, false, () => 0);
                applyMechanicsField(ctx, 'static spread', 'spread', 'weapon', 0x30, false, () => 0);
                applyMechanicsField(ctx, 'current spread', 'spread', 'controller', 0xe8, false, () => 0);
            } else restoreMechanicsFields(mechanicsRecord, 'spread');
            if (rapidFire) {
                const faster = base => Math.max(base,
                    Math.min(base * RAPID_FIRE_MULTIPLIER, RAPID_FIRE_SOFT_CAP));
                applyMechanicsField(ctx, 'weapon fire rate', 'fire', 'weapon', 0x88, false, faster);
                applyMechanicsField(ctx, 'cached fire rate', 'fire', 'shooter', 0x100, true, faster);
            } else restoreMechanicsFields(mechanicsRecord, 'fire');
            if (instantReload) {
                applyMechanicsField(ctx, 'reload wait', 'reload', 'shooter', 0x130, true, () => 0);
            } else restoreMechanicsFields(mechanicsRecord, 'reload');
            if (!anyWeaponSettingsOn() && !mechanicsRecord.fields.size) mechanicsRecord = null;
        } catch (error) { failWeaponSettings(error); }
        finally { mechanicsBusy = false; }
    }
    function setWeaponSetting(name, enabled) {
        enabled = Boolean(enabled);
        if (name === 'spread') noSpread = enabled;
        else if (name === 'fire') rapidFire = enabled;
        else if (name === 'reload') instantReload = enabled;
        else throw new RangeError('Unknown weapon setting.');
        mechanicsErrorReported = false;
        updateWeaponSettings();
        renderActiveMods();
    }

    function mechanicsTypedWrapper(callback, parameterCount, returnsInt) {
        // Tiny typed wrapper, valid as an entry in Unity's WASM function table.
        const section = (id, data) => [id, data.length, ...data];
        const type = [1, 0x60, parameterCount,
            ...Array(parameterCount).fill(0x7f), ...(returnsInt ? [1, 0x7f] : [0])];
        const body = [0];
        for (let i = 0; i < parameterCount; i++) body.push(0x20, i);
        body.push(0x10, 0, 0x0b);
        const bytes = new Uint8Array([
            0,97,115,109,1,0,0,0,
            ...section(1, type),
            ...section(2, [1,4,104,97,122,121,4,115,116,101,112,0,0]),
            ...section(3, [1,0]),
            ...section(7, [1,4,115,116,101,112,0,1]),
            ...section(10, [1,body.length,...body])
        ]);
        const instance = new WebAssembly.Instance(new WebAssembly.Module(bytes),
            { hazy: { step: callback } });
        return { wrapper: instance.exports.step, instance };
    }
    function prepareWeaponSettingsRuntime(table, memory) {
        if (!table || typeof table.get !== 'function' || !memory?.buffer ||
            !mechanicsBuildMatches(new DataView(memory.buffer))) return false;
        const existing = mechanicsHooks.get(table);
        if (existing) {
            if (existing.some(h => table.get(h.slot) !== h.wrapper)) {
                throw new Error('Another script replaced a weapon-settings callback.');
            }
            return true;
        }
        const installed = [];
        try {
            for (const [slot, count, kind] of [[17482, 2, 'controller'],
                [20702, 2, 'shooter'], [20706, 3, 'shooter']]) {
                const original = table.get(slot);
                if (typeof original !== 'function') throw new Error('Weapon callback unavailable.');
                const callback = (...args) => {
                    if (!anyWeaponSettingsOn()) return original(...args);
                    updateWeaponSettings(args[0], kind);
                    try { return original(...args); }
                    finally { updateWeaponSettings(args[0], kind); }
                };
                const shim = mechanicsTypedWrapper(callback, count, false);
                table.set(slot, shim.wrapper);
                installed.push({ slot, original, ...shim });
            }
            const slot = 20774; // Shooter.<IenuReload>d__121.MoveNext
            const original = table.get(slot);
            if (typeof original !== 'function') throw new Error('Reload callback unavailable.');
            const callback = (iterator, methodInfo) => {
                if (instantReload) {
                    try {
                        const ctx = resolveMechanicsPlayer();
                        const view = ctx ? sceneView(ctx.runtime) : null;
                        if (ctx && ctx.runtime.table === table &&
                            infoValid(view, iterator, 0x1c) &&
                            infoClassNamed(view, infoUint(view, iterator), '<IenuReload>d__121') &&
                            infoUint(view, iterator + 0x10) === ctx.shooter) {
                            const state = view.getInt32(iterator + 8, true);
                            // State 0 initializes; state 1 is the native wait loop.
                            // Setting only the wait retains ordinary completion,
                            // including per-shell reloads, weapon flags and ammo UI.
                            if (state === 0 || state === 1) {
                                view.setFloat32(iterator + 0x14, 0, true);
                            }
                        }
                    } catch (error) { failWeaponSettings(error); }
                }
                return original(iterator, methodInfo);
            };
            const shim = mechanicsTypedWrapper(callback, 2, true);
            table.set(slot, shim.wrapper);
            installed.push({ slot, original, ...shim });
            if (installed.some(h => table.get(h.slot) !== h.wrapper)) {
                throw new Error('Weapon-settings callback installation failed.');
            }
            mechanicsHooks.set(table, installed);
            return true;
        } catch (error) {
            for (const h of installed) {
                try { if (table.get(h.slot) === h.wrapper) table.set(h.slot, h.original); } catch (_) {}
            }
            throw error;
        }
    }
    // Startup wrappers cover the local controller and normal firing path.
    // The low-rate maintenance pass covers lobby/weapon replacement transitions.
    window.setInterval(() => updateWeaponSettings(), 500);
    window.addEventListener('pagehide', () => {
        noSpread = false; rapidFire = false; instantReload = false; autoReload = false;
        try { restoreMechanicsFields(mechanicsRecord); } catch (_) {}
        mechanicsRecord = null;
        releaseMechanicsScratch();
        // Keep the wrappers as inert forwarders for back/forward cache restores.
    });

    // ============================================================
    // AUTO RELOAD — ZERO-AMMO NORMAL R-KEY REQUEST
    // ============================================================
    // This does not write ammo or bypass reload completion. When the local gun
    // reaches exactly zero ammo, it sends the same R-key press used by the game.
    const AUTO_RELOAD_CHECK_MS = 100;
    const AUTO_RELOAD_RETRY_MS = 850;
    const AUTO_RELOAD_CHECKS = Object.freeze([
        [0x6c1dec, 20725] // Shooter.GetCurrentAmmo
    ]);
    let autoReloadLastCheck = -Infinity;
    let autoReloadLastRequest = -Infinity;
    let autoReloadWeaponIdentity = 0;
    let autoReloadErrorReported = false;

    function dispatchAutoReloadKey() {
        const target = document.pointerLockElement || document;
        const init = {
            key: 'r', code: 'KeyR', bubbles: true, cancelable: true, composed: true
        };
        const down = new KeyboardEvent('keydown', init);
        const up = new KeyboardEvent('keyup', init);
        try {
            Object.defineProperty(down, 'keyCode', { value: 82 });
            Object.defineProperty(down, 'which', { value: 82 });
            Object.defineProperty(up, 'keyCode', { value: 82 });
            Object.defineProperty(up, 'which', { value: 82 });
        } catch (_) {}
        target.dispatchEvent(down);
        target.dispatchEvent(up);
    }

    function setAutoReloadEnabled(enabled) {
        autoReload = Boolean(enabled);
        autoReloadLastCheck = -Infinity;
        autoReloadLastRequest = -Infinity;
        autoReloadWeaponIdentity = 0;
        autoReloadErrorReported = false;
    }

    function updateAutoReload(now) {
        if (!autoReload || now - autoReloadLastCheck < AUTO_RELOAD_CHECK_MS) return;
        autoReloadLastCheck = now;
        if (!document.pointerLockElement || document.hidden ||
            (typeof document.hasFocus === 'function' && !document.hasFocus()) ||
            editingText({ target: document.activeElement })) return;
        try {
            const ctx = resolveMechanicsPlayer();
            if (!ctx) return;
            const view = sceneView(ctx.runtime);
            if (!AUTO_RELOAD_CHECKS.every(([address, value]) =>
                address + 4 <= view.byteLength && view.getUint32(address, true) === value)) {
                throw new Error('Auto Reload does not match this Kour build.');
            }
            if (ctx.weapon !== autoReloadWeaponIdentity) {
                autoReloadWeaponIdentity = ctx.weapon;
                autoReloadLastRequest = -Infinity;
            }
            const getAmmo = sceneFunction(ctx.runtime, 20725);
            const ammo = getAmmo(ctx.shooter, 0);
            if (!Number.isInteger(ammo)) return;
            if (ammo > 0 || ammo === -1) {
                autoReloadLastRequest = -Infinity;
                return;
            }
            if (ammo !== 0 || now - autoReloadLastRequest < AUTO_RELOAD_RETRY_MS) return;
            autoReloadLastRequest = now;
            dispatchAutoReloadKey();
            autoReloadErrorReported = false;
        } catch (error) {
            setAutoReloadEnabled(false);
            renderActiveMods();
            if (!autoReloadErrorReported) {
                autoReloadErrorReported = true;
                reportSceneError('Auto Reload', error);
            }
        }
    }

    // TRIGGERBOT — REAL CENTER-RAY HIT + NORMAL SHOOTER.SHOOT PATH
    // ============================================================

    // No screen-space ESP proximity, fabricated hit, ammo/rate rewrite, or RPC.
    const TRIGGER_CHECK_MS = 50;
    const TRIGGER_CHECKS = [
        [0x6dea24, 45074],   // Physics.Raycast(origin, direction, out hit, range, mask)
        [0x6c1db4, 20711],   // Shooter.GetMultiFromRaycastHit(hit, out PhotonView)
        [0x6c1da0, 20706],   // Shooter.Shoot(recursiveBurst)
        [0x6c1dec, 20725],   // Shooter.GetCurrentAmmo
        [0x6bcc50, 15313],   // ObscuredFloat -> float (read a copy)
        [0x6d94f0, 12372],   // Time.time
        [0x6d969c, 2965], [0x6d96c8, 4579], [0x6d8d7c, 2964],
        [0x6c253c, 21187],   // TeamMember.thisTeam
        [0x6c0458, 19115],   // Health.IsSpawnProtected
        [0x3e5190, 0x0c],   // PhotonTeam.Code
        [0x3c9dcc, 0x38], [0x3c9dd4, 0x40], [0x3c9e0c, 0x6c],
        [0x3c9e3c, 0xb4], [0x3c9e40, 0xb8], [0x3c9e50, 0xc8],
        [0x3c9e68, 0xe0], [0x3c9e70, 0xe8], [0x3c9e78, 0x118],
        [0x3c9e84, 0x15c], [0x3c9ed4, 0x1d8],
        [0x3ca9f0, 0x20], [0x3caa40, 0x6c], [0x3caa44, 0x70],
        [0x3caaa4, 0xcd], [0x3c8254, 0x80], [0x3c8274, 0xa8],
        [0x3c930c, 0x50], [0x3c9314, 0x58], [0x3c92f8, 0x3c],
        [0x3c9328, 8], [0x3c9330, 10], [0x3c9340, 11]
    ];
    let triggerLastCheck = -Infinity;
    let triggerErrorReported = false;
    let triggerAllocation = null;
    let triggerPendingCandidate = null;
    const triggerDiagnostics = { state: 'off', shots: 0, lastError: '' };
    Object.defineProperty(window, '__hazyTriggerStatus', {
        configurable: true, get: () => ({ ...triggerDiagnostics })
    });
    function releaseTriggerAllocation() {
        if (triggerAllocation) {
            try { triggerAllocation.free(); } catch (_) {}
            triggerAllocation = null;
        }
    }
    function triggerScratch(runtime) {
        if (triggerAllocation?.table === runtime.table &&
            infoValid(sceneView(runtime), triggerAllocation.pointer, 128)) {
            return triggerAllocation.pointer;
        }
        releaseTriggerAllocation();
        const sources = [runtime.module, runtime.exports, runtime.module?.asm, runtime.module?.wasmExports];
        for (const owner of sources) {
            const malloc = owner?._malloc || owner?.malloc;
            const free = owner?._free || owner?.free;
            if (typeof malloc !== 'function' || typeof free !== 'function') continue;
            const pointer = Reflect.apply(malloc, owner, [128]) >>> 0;
            if (!infoValid(sceneView(runtime), pointer, 128)) {
                if (pointer) Reflect.apply(free, owner, [pointer]);
                throw new Error('Triggerbot scratch allocation failed.');
            }
            triggerAllocation = { table: runtime.table, pointer,
                free: () => Reflect.apply(free, owner, [pointer]) };
            return pointer;
        }
        throw new Error('Triggerbot allocator unavailable. Save and refresh Kour.');
    }
    function setTriggerbotEnabled(enabled) {
        triggerbot = Boolean(enabled);
        triggerErrorReported = false;
        triggerLastCheck = -Infinity;
        triggerPendingCandidate = null;
        triggerDiagnostics.state = triggerbot ? 'waiting for game' : 'off';
        if (!triggerbot) releaseTriggerAllocation();
    }
    function triggerInputAllowed() {
        return extraMovementInputActive() && !leftMouseHeld;
    }
    function triggerReadFloat(runtime, address, scratch) {
        const view = sceneView(runtime);
        if (!infoValid(view, address, 24) || view.getUint8(address + 12) !== 1) return null;
        // Use a copy, as the native value-conversion API accepts a value-type pointer.
        const bytes = new Uint8Array(view.buffer, view.byteOffset, view.byteLength);
        bytes.copyWithin(scratch + 96, address, address + 24);
        const n = sceneFunction(runtime, 15313)(scratch + 96, 0);
        return Number.isFinite(n) ? n : null;
    }
    function triggerHealth(runtime, shooter, local = false) {
        const view = sceneView(runtime);
        const hp = infoUint(view, shooter + 0x6c);
        if (!infoObjectNamed(view, hp, 'Health', 0xb8) ||
            infoUint(view, hp + 0x14) !== shooter ||
            view.getUint8(hp + (local ? 0x80 : 0xa8)) !== 0) return 0;
        const health = readObscuredInt(view, hp + 0x6c);
        return Number.isInteger(health) && health > 0 ? hp : 0;
    }
    function triggerTeamCode(runtime, shooter) {
        const view = sceneView(runtime);
        const member = infoUint(view, shooter + 0x40);
        if (!infoObjectNamed(view, member, 'TeamMember', 0xa0)) return undefined;
        const team = sceneFunction(runtime, 21187)(member, 0);
        if (!team) return null; // The native shot path also accepts no-team play.
        const fresh = sceneView(runtime);
        if (!infoValid(fresh, team, 16) || !infoClassNamed(fresh, infoUint(fresh, team), 'PhotonTeam')) {
            return undefined;
        }
        return fresh.getUint8(team + 12);
    }
    function findTriggerCandidate() {
        const local = speedEngine.getPlayer();
        const runtime = findSceneRuntime();
        if (!local || !runtime) return null;
        let view = sceneView(runtime);
        if (view.buffer !== local.view.buffer) return null;
        if (!TRIGGER_CHECKS.every(([p, n]) => p + 4 <= view.byteLength && view.getUint32(p, true) === n)) {
            throw new Error('Triggerbot does not match this game build.');
        }
        if (view.getUint8(local.shooter + 0x38) !== 1 ||
            view.getUint8(local.shooter + 0x1d8) !== 0 ||
            !triggerHealth(runtime, local.shooter, true)) return null;
        const input = infoUint(view, local.shooter + 0xc8);
        if (!infoObjectNamed(view, input, 'PlayerInputHandler', 0x74) ||
            infoUint(view, input + 0x3c) !== local.shooter ||
            view.getUint8(input + 0x50) || view.getUint8(input + 0x58)) return null;
        const statics = infoUint(view, infoUint(view, input) + 0x5c);
        if (!infoValid(view, statics, 12) || view.getUint8(statics + 8) ||
            view.getUint8(statics + 10) || view.getUint8(statics + 11)) return null;
        const weapon = infoUint(view, local.shooter + 0xe0);
        if (!infoValid(view, weapon, 0x100) || !infoUint(view, weapon + 8) ||
            !infoDerivesFrom(view, infoUint(view, weapon), 'Weapon') ||
            view.getUint8(weapon + 0x6c) !== 1 ||
            infoUint(view, weapon + 0x70) !== local.shooter ||
            view.getUint8(weapon + 0xcd) || view.getUint8(weapon + 0x20)) return null;
        // Charge/release weapons are not driven by this normal-shot helper.
        const scratch = triggerScratch(runtime);
        const now = sceneFunction(runtime, 12372)(0);
        const nextShot = triggerReadFloat(runtime, local.shooter + 0xe8, scratch);
        const nextSwitch = triggerReadFloat(runtime, local.shooter + 0x15c, scratch);
        const range = triggerReadFloat(runtime, local.shooter + 0x118, scratch);
        if (!Number.isFinite(now) || nextShot === null || nextSwitch === null ||
            now < nextShot || now < nextSwitch || range === null || range <= 0 || range > 10000) return null;
        const ammo = sceneFunction(runtime, 20725)(local.shooter, 0);
        if (!Number.isInteger(ammo) || (ammo <= 0 && ammo !== -1)) return null;
        view = sceneView(runtime);
        const camera = infoUint(view, local.shooter + 0xb8);
        if (!infoObjectNamed(view, camera, 'Camera', 12)) return null;
        const cameraTransform = sceneFunction(runtime, 2964)(camera, 0);
        if (!infoObjectNamed(sceneView(runtime), cameraTransform, 'Transform', 12)) return null;
        sceneFunction(runtime, 2965)(scratch, cameraTransform, 0);
        sceneFunction(runtime, 4579)(scratch + 16, cameraTransform, 0);
        view = sceneView(runtime);
        const vector = [0,4,8,16,20,24].map(o => view.getFloat32(scratch + o, true));
        const dirLength = Math.hypot(...vector.slice(3));
        if (!vector.every(Number.isFinite) || dirLength < 0.9 || dirLength > 1.1) return null;
        const mask = view.getInt32(local.shooter + 0xb4, true);
        view.setUint32(scratch + 80, 0, true);
        // The nearest real collider stops the ray. A wall does not become a
        // target even if a player happens to be behind it or highlighted by ESP.
        const hit = sceneFunction(runtime, 45074)(scratch, scratch + 16, scratch + 32, range, mask, 0);
        if (hit !== 1) return null;
        const multiplier = sceneFunction(runtime, 20711)(local.shooter, scratch + 32, scratch + 80, 0);
        view = sceneView(runtime);
        const photon = infoUint(view, scratch + 80);
        if (!Number.isFinite(multiplier) || multiplier <= 0 ||
            !infoObjectNamed(view, photon, 'PhotonView', 0x5c) || view.getUint8(photon + 0x40) !== 0) return null;
        const players = registryShooters(runtime);
        if (!players) return null;
        const target = players.find(shooter => shooter !== local.shooter &&
            infoUint(view, shooter + 0x10) === photon);
        if (!target) return null;
        const hp = triggerHealth(runtime, target);
        if (!hp || sceneFunction(runtime, 19115)(hp, 0) !== 0) return null;
        const ownTeam = triggerTeamCode(runtime, local.shooter);
        const otherTeam = triggerTeamCode(runtime, target);
        if (ownTeam === undefined || otherTeam === undefined ||
            (ownTeam !== null && otherTeam !== null && ownTeam === otherTeam)) return null;
        return { local, runtime, target, weapon };
    }
    function updateTriggerbot(now) {
        if (!triggerbot || !triggerInputAllowed()) {
            triggerPendingCandidate = null;
            triggerDiagnostics.state = triggerbot ? 'paused / manual fire' : 'off';
            return;
        }
        if (now - triggerLastCheck < TRIGGER_CHECK_MS) return;
        triggerLastCheck = now;
        try {
            const candidate = findTriggerCandidate();
            if (!candidate) {
                triggerPendingCandidate = null;
                triggerDiagnostics.state = 'no ready valid target';
                return;
            }
            // Confirm acquisition across two checks, rather than firing at a
            // one-frame edge while the camera is being changed or resumed.
            const previous = triggerPendingCandidate;
            triggerPendingCandidate = candidate;
            if (!previous || previous.target !== candidate.target ||
                previous.local.controllerNative !== candidate.local.controllerNative ||
                previous.weapon !== candidate.weapon || previous.runtime.table !== candidate.runtime.table) {
                triggerDiagnostics.state = 'confirming target';
                return;
            }
            if (!triggerbot || !triggerInputAllowed()) return;
            if (oneShot) oneShotEngine.tick();
            // Use the game's normal firing implementation, not a fake hit RPC.
            // Respect its reload, ammo, weapon switch, and next-shot timestamp.
            sceneFunction(candidate.runtime, 20706)(candidate.local.shooter, 0, 0);
            triggerDiagnostics.shots++;
            triggerDiagnostics.state = 'normal shot requested';
        } catch (error) {
            setTriggerbotEnabled(false);
            triggerDiagnostics.state = 'error';
            triggerDiagnostics.lastError = error.message || String(error);
            renderActiveMods();
            if (!triggerErrorReported) {
                triggerErrorReported = true;
                reportSceneError('Triggerbot', error);
            }
        }
    }
    window.addEventListener('pagehide', () => setTriggerbotEnabled(false));

    // ============================================================
    // PLAYER NETWORK POSITIONS FOR AIMBOT
    // ============================================================
    const positions = new Map();

    // Hitbox ESP renders every animation frame, but Kour's remote transform
    // snapshots can arrive much less frequently than the display refresh rate.
    // Keep a separate visual state so boxes can move smoothly between packets
    // without changing the raw positions used by Aimbot.
    const hitboxRenderStates = new Map();
    const HITBOX_MAX_PREDICTION_MS = 600;
    const HITBOX_SMOOTH_RATE = 30;
    const HITBOX_MAX_TRACKED_SPEED = 80;
    const HITBOX_TELEPORT_DISTANCE = 22;

    const PARAM_ACTOR_NR = 254;
    const TYPE_BYTE = 0x0b;
    const EVENT_PLAYER_TRANSFORM = 0xc9;
    const EVENT_PLAYER_LEAVE = 0xfe;
    const POSITION_MARKER = [0xd6,0x0c];

    function trailerActorNr(bytes) {
        const l = bytes.length;
        if (l < 3 || bytes[l-3] !== PARAM_ACTOR_NR || bytes[l-2] !== TYPE_BYTE) return null;
        return bytes[l-1];
    }

    function decodeTransformPosition(bytes) {
        let index = -1;
        for (let i=0; i<=bytes.length-2; i++) {
            if (bytes[i] === POSITION_MARKER[0] && bytes[i+1] === POSITION_MARKER[1]) {
                index = i;
                break;
            }
        }
        if (index === -1 || index + 14 > bytes.length) return null;
        try {
            const view = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
            const x = view.getFloat32(index+2, false);
            const y = view.getFloat32(index+6, false);
            const z = view.getFloat32(index+10, false);
            if (![x,y,z].every(Number.isFinite)) return null;
            return {x,y,z};
        } catch (_) { return null; }
    }

    function handleGameFrame(bytes) {
        if (bytes.length < 3 || bytes[0] !== 0xf3 || bytes[1] !== 4) return;
        if (bytes[2] === EVENT_PLAYER_LEAVE) {
            const actor = trailerActorNr(bytes);
            if (actor !== null) {
                positions.delete(actor);
                hitboxRenderStates.delete(actor);
                if (lockedActor === actor) lockedActor = null;
            }
            return;
        }
        if (bytes[2] !== EVENT_PLAYER_TRANSFORM) return;
        const actor = trailerActorNr(bytes);
        const pos = decodeTransformPosition(bytes);
        if (actor === null || !pos) return;

        const receivedAt = performance.now();
        const previous = positions.get(actor);
        let vx = 0;
        let vy = 0;
        let vz = 0;
        let sampleMs = 0;

        if (previous && Number.isFinite(previous.pt)) {
            sampleMs = receivedAt - previous.pt;

            if (sampleMs >= 12 && sampleMs <= 2000) {
                const dt = sampleMs / 1000;
                const dx = pos.x - previous.x;
                const dy = pos.y - previous.y;
                const dz = pos.z - previous.z;
                const distance = Math.hypot(dx, dy, dz);

                if (distance <= HITBOX_TELEPORT_DISTANCE) {
                    const rawVX = dx / dt;
                    const rawVY = dy / dt;
                    const rawVZ = dz / dt;
                    const rawSpeed = Math.hypot(rawVX, rawVY, rawVZ);

                    if (Number.isFinite(rawSpeed) && rawSpeed <= HITBOX_MAX_TRACKED_SPEED) {
                        // A little velocity blending prevents packet jitter from becoming
                        // visible as shaking boxes while still responding quickly to turns.
                        const blend = Number.isFinite(previous.vx) ? 0.72 : 1;
                        vx = (Number(previous.vx) || 0) * (1 - blend) + rawVX * blend;
                        vy = (Number(previous.vy) || 0) * (1 - blend) + rawVY * blend;
                        vz = (Number(previous.vz) || 0) * (1 - blend) + rawVZ * blend;
                    }
                } else {
                    // Respawn/teleport: do not animate across the whole map.
                    hitboxRenderStates.delete(actor);
                }
            }
        }

        positions.set(actor, {
            ...pos,
            t: Date.now(),
            pt: receivedAt,
            vx,
            vy,
            vz,
            sampleMs
        });
    }

    function espRoom(runtime) {
        if (sceneFunction(runtime, 27689)(0) !== 1) return 0;
        const room = sceneFunction(runtime, 27666)(0);
        const view = sceneView(runtime);
        return infoValid(view, room, 12) && infoClassNamed(view, infoUint(view, room), 'Room') ? room : 0;

    function releaseESPScratch() {
        const old = espAllocation; espAllocation = null;
        if (old) { try { old.free(); } catch (_) {} }
    }
    function espScratch(runtime) {
        if (espAllocation?.table === runtime.table && infoValid(sceneView(runtime), espAllocation.pointer, 256)) {
            return espAllocation.pointer;
        }
        releaseESPScratch();
        for (const owner of [runtime.module, runtime.exports, runtime.module?.asm, runtime.module?.wasmExports]) {
            const alloc = owner?._malloc || owner?.malloc;
            const free = owner?._free || owner?.free;
            if (typeof alloc !== 'function' || typeof free !== 'function') continue;
            const pointer = Reflect.apply(alloc, owner, [256]) >>> 0;
            if (!infoValid(sceneView(runtime), pointer, 256)) {
                if (pointer) Reflect.apply(free, owner, [pointer]);
                throw new Error('ESP scratch allocation failed.');
            }
            espAllocation = { table: runtime.table, pointer, free: () => Reflect.apply(free, owner, [pointer]) };
            return pointer;
        }
        throw new Error('ESP allocator unavailable. Save and refresh Kour.');
    }
    function espAlive(view, pointer, name, native) {
        return infoObjectNamed(view, pointer, name, 12) &&
            (native === undefined || infoUint(view, pointer + 8) === native);
    }
    function espVector(runtime, transform, scratch) {
        if (!espAlive(sceneView(runtime), transform, 'Transform')) return null;
        sceneFunction(runtime, 2965)(scratch, transform, 0);
        const view = sceneView(runtime);
        const p = { x: view.getFloat32(scratch, true), y: view.getFloat32(scratch + 4, true),
            z: view.getFloat32(scratch + 8, true) };
        return Object.values(p).every(n => Number.isFinite(n) && Math.abs(n) < 100000) ? p : null;
    }
    function multiplyESPMatrix(a, b) {
        const result = new Float64Array(16);
        for (let c = 0; c < 4; c++) for (let r = 0; r < 4; r++) {
            result[c * 4 + r] = a[r] * b[c * 4] + a[4 + r] * b[c * 4 + 1] +
                a[8 + r] * b[c * 4 + 2] + a[12 + r] * b[c * 4 + 3];
        }
        return result;
    }

        espGeneration++;

    function resolveVPMatrixForFrame() {
        // Invalidate first: a failed/no-camera frame must never reuse a matrix.
        espFrame = null; vpMatrix = null;
        if ((!aimbot && !anyProjectedVisualsOn()) || !espInputActive()) return null;
        const local = speedEngine.getPlayer(), runtime = findSceneRuntime(), canvas = espGameCanvas();
        if (!local || !runtime || !canvas || sceneView(runtime)?.buffer !== local.view.buffer) return null;
        let view = sceneView(runtime);
        if (!espBuildMatches(view)) throw new Error('ESP camera/rig layout differs from the supplied Kour build.');
        const room = espRoom(runtime);
        view = sceneView(runtime);
        if (!room || view.getUint8(local.shooter + 0x38) !== 1) return null;
        const localHealth = infoUint(view, local.shooter + 0x6c);
        if (infoObjectNamed(view, localHealth, 'Health', 0xb8) && view.getUint8(localHealth + 0x80)) return null;
        const camera = infoUint(view, local.controller + 0x64);
        if (!espAlive(view, camera, 'Camera') || sceneFunction(runtime, 3167)(camera, 0) !== 1) return null;
        view = sceneView(runtime);
        // Do not select the first-person weapon camera, a lobby camera, or Camera.main by name.
        const cameraNative = infoUint(view, camera + 8);
        const identity = { table: runtime.table, room, controller: local.controller,
            native: local.controllerNative, camera, cameraNative };
        if (!espSession || Object.keys(identity).some(k => identity[k] !== espSession[k])) {
            espRigCache.clear(); espRendererCache.clear(); playerInfoByActor.clear();
            playerInfoWeaponNames.clear(); hitboxRenderStates.clear(); espSession = identity;
            refreshRemotePlayerInfo();
        }
        const scratch = espScratch(runtime);
        sceneFunction(runtime, 38521)(scratch, camera, 0);
        view = sceneView(runtime);
        const v = new Float64Array(16);
        for (let i = 0; i < 16; i++) v[i] = view.getFloat32(scratch + 4 * i, true);
        sceneFunction(runtime, 38522)(scratch, camera, 0);
        view = sceneView(runtime);
        const p = new Float64Array(16);
        for (let i = 0; i < 16; i++) p[i] = view.getFloat32(scratch + 4 * i, true);
        if (![...v, ...p].every(Number.isFinite) || Math.abs(p[0]) < 1e-6 || Math.abs(p[5]) < 1e-6) return null;
        sceneFunction(runtime, 38512)(scratch, camera, 0);
        view = sceneView(runtime);
        const viewport = [0, 4, 8, 12].map(o => view.getFloat32(scratch + o, true));
        if (!viewport.every(Number.isFinite) || viewport[0] < 0 || viewport[1] < 0 ||
            viewport[2] <= 0 || viewport[3] <= 0 || viewport[0] + viewport[2] > 1.001 ||
            viewport[1] + viewport[3] > 1.001) return null;
        const near = sceneFunction(runtime, 38480)(camera, 0);
        const far = sceneFunction(runtime, 4850)(camera, 0);
        if (!Number.isFinite(near) || !Number.isFinite(far) || near <= 0 || far <= near) return null;
        const transform = sceneFunction(runtime, 2964)(camera, 0);
        const eye = espVector(runtime, transform, scratch);
        if (!eye) return null;
        const cr = canvas.getBoundingClientRect();
        if (cr.width < 10 || cr.height < 10) return null;
        const rect = { left: cr.left + viewport[0] * cr.width,
            top: cr.top + (1 - viewport[1] - viewport[3]) * cr.height,
            width: cr.width * viewport[2], height: cr.height * viewport[3] };
        rect.right = rect.left + rect.width; rect.bottom = rect.top + rect.height;
        vpMatrix = multiplyESPMatrix(p, v);
        espFrame = { runtime, local, room, canvas, camera, cameraNative, matrix: vpMatrix,
            viewMatrix: v, near, far, eye, rect, scratch, at: performance.now() };
        espStatus.state = 'live camera'; espStatus.lastError = '';
        return espFrame;
    }
    function projectESPPoint(position, loose = false, frame = espFrame) {
        if (!frame || !position || !Object.values(position).every(Number.isFinite)) return null;
        const { x, y, z } = position, m = frame.matrix, v = frame.viewMatrix;
        const depth = -(v[2] * x + v[6] * y + v[10] * z + v[14]);
        const cx = m[0]*x + m[4]*y + m[8]*z + m[12];
        const cy = m[1]*x + m[5]*y + m[9]*z + m[13];
        const cw = m[3]*x + m[7]*y + m[11]*z + m[15];
        if (![depth, cx, cy, cw].every(Number.isFinite)) return null;
        if (!loose && (depth < frame.near || depth > frame.far || cw <= 1e-8)) return null;
        const denominator = loose ? Math.max(1e-6, Math.abs(cw)) : cw;
        return { x: (cx / denominator * 0.5 + 0.5) * frame.rect.width,
            y: (0.5 - cy / denominator * 0.5) * frame.rect.height,
            depth, behind: depth <= frame.near };
    }
    function projectToScreen(position, width, height) {
        const p = projectESPPoint(position);
        if (!p) return null;
        // Aim uses CSS canvas dimensions. Account for a non-fullscreen camera viewport.
        const cr = espFrame.canvas.getBoundingClientRect(), vr = espFrame.rect;
        return { x: (vr.left - cr.left + p.x) * width / cr.width,
            y: (vr.top - cr.top + p.y) * height / cr.height };
    }
    function projectToScreenLoose(position, width, height) {
        return projectESPPoint(position, true);
    }
    function projectESPSegment(a, b, frame = espFrame) {
        if (!frame || !a || !b) return null;
        const v = frame.viewMatrix;
        const depth = p => -(v[2]*p.x + v[6]*p.y + v[10]*p.z + v[14]);
        const da = depth(a), db = depth(b);
        if (!Number.isFinite(da) || !Number.isFinite(db)) return null;
        let lo = 0, hi = 1;
        const delta = db - da;
        if (Math.abs(delta) < 1e-9) {
            if (da < frame.near || da > frame.far) return null;
        } else {
            const t1 = (frame.near * 1.0001 - da) / delta;
            const t2 = (frame.far - da) / delta;
            lo = Math.max(lo, Math.min(t1, t2)); hi = Math.min(hi, Math.max(t1, t2));
            if (hi < lo) return null;
        }
        const lerp = t => ({ x: a.x+(b.x-a.x)*t, y: a.y+(b.y-a.y)*t, z: a.z+(b.z-a.z)*t });
        const pa = projectESPPoint(lerp(lo), false, frame), pb = projectESPPoint(lerp(hi), false, frame);
        return pa && pb ? [pa, pb] : null;
    }
    function onDrawCall(gl) {
        if (!aimbot && !anyProjectedVisualsOn()) return;
        if (espQueued && gl.canvas === activeGL?.canvas) return;
        if (!espInputActive()) return;
        const canvas = espGameCanvas();
        if (!canvas || gl.canvas !== canvas) return;
        activeGL = gl; espLastDrawAt = performance.now();
        if (espQueued) return;
        espQueued = true;
        const generation = espGeneration;
        queueMicrotask(() => {
            espQueued = false;
            if (generation !== espGeneration) return;
            try {
                if (!resolveVPMatrixForFrame()) { resetESPTracking('waiting for live player/camera'); return; }
                updateVisualESP(performance.now());
            } catch (error) {
                resetESPTracking('unavailable');
                espStatus.lastError = error.message || String(error);
                if (performance.now() - espLastErrorAt > 10000) {
                    espLastErrorAt = performance.now();
                    reportSceneError('ESP', error);
                }
            }
        });
    }
    function maintainVisualESP(now) {

            if (espFrame || espSession) resetESPTracking('paused / no fresh render');
            else clearVisualESP(false);
        }
    }

    function installWebGLHooks() {
        if (typeof WebGL2RenderingContext === 'undefined') return;
        const GL = WebGL2RenderingContext.prototype;

        // ========================================================
        // FPS BOOST — SKIP REAL-TIME SHADOW MAP RENDER PASSES
        // ========================================================
        // Unity renders real-time shadows into off-screen depth-only framebuffers.
        // When FPS Boost is on, those draw calls are skipped. Normal scene/color
        // rendering, ESP shaders and the default screen framebuffer are untouched.
        const drawFramebuffers = new WeakMap();
        const shadowFramebufferCache = new WeakMap();

        function invalidateShadowFramebuffer(gl) {
            const framebuffer = drawFramebuffers.get(gl);
            if (framebuffer) shadowFramebufferCache.delete(framebuffer);
        }

        function isShadowFramebuffer(gl) {
            const currentDrawFramebuffer = drawFramebuffers.get(gl);
            if (!fpsBoost || !currentDrawFramebuffer) return false;

            let shadowLike = false;
            try {
                const target = gl.DRAW_FRAMEBUFFER || gl.FRAMEBUFFER;
                const none = gl.NONE;

                const colorType = gl.getFramebufferAttachmentParameter(
                    target,
                    gl.COLOR_ATTACHMENT0,
                    gl.FRAMEBUFFER_ATTACHMENT_OBJECT_TYPE
                );

                const depthType = gl.getFramebufferAttachmentParameter(
                    target,
                    gl.DEPTH_ATTACHMENT,
                    gl.FRAMEBUFFER_ATTACHMENT_OBJECT_TYPE
                );

                let depthStencilType = none;
                try {
                    depthStencilType = gl.getFramebufferAttachmentParameter(
                        target,
                        gl.DEPTH_STENCIL_ATTACHMENT,
                        gl.FRAMEBUFFER_ATTACHMENT_OBJECT_TYPE
                    );
                } catch (_) {}

                const hasDepth = depthType !== none || depthStencilType !== none;
                const noColorAttachment = colorType === none;

                const colorMask = gl.getParameter(gl.COLOR_WRITEMASK);
                const noColorWrites = Array.isArray(colorMask)
                    ? colorMask.every(value => value === false)
                    : colorMask && typeof colorMask.length === 'number'
                        ? Array.from(colorMask).every(value => value === false)
                        : false;

                const depthWrites = gl.getParameter(gl.DEPTH_WRITEMASK) === true;

                // True depth-only targets are the normal Unity shadow-map path.
                // The no-color-write fallback catches packed shadow passes that still
                // have a color attachment but deliberately write depth only.
                shadowLike = hasDepth && depthWrites && (noColorAttachment || noColorWrites);
            } catch (_) {
                shadowLike = false;
            }

            shadowFramebufferCache.set(currentDrawFramebuffer, shadowLike);
            return shadowLike;
        }

        const originalBindFramebuffer = GL.bindFramebuffer;
        GL.bindFramebuffer = function(target, framebuffer) {
            const result = originalBindFramebuffer.apply(this, arguments);
            if (target === this.FRAMEBUFFER || target === this.DRAW_FRAMEBUFFER) {
                drawFramebuffers.set(this, framebuffer || null);
            }
            return result;
        };

        const originalFramebufferTexture2D = GL.framebufferTexture2D;
        GL.framebufferTexture2D = function(...args) {
            const result = originalFramebufferTexture2D.apply(this, args);
            invalidateShadowFramebuffer(this);
            return result;
        };

        const originalFramebufferRenderbuffer = GL.framebufferRenderbuffer;
        GL.framebufferRenderbuffer = function(...args) {
            const result = originalFramebufferRenderbuffer.apply(this, args);
            invalidateShadowFramebuffer(this);
            return result;
        };

        const originalLinkProgram = GL.linkProgram;
        GL.linkProgram = function(program) {
            // Re-linked programs need fresh uniform locations.
            if (program) delete program.__hazyUniforms;
            return originalLinkProgram.apply(this, arguments);
        };

        const originalShaderSource = GL.shaderSource;
        GL.shaderSource = function(shader,source) {
            try {
                if (typeof source === 'string') {
                    if (source.includes('gl_Position') && !source.includes('hazyVertexCount')) {
                        const playerConditions = PLAYER_VERTEX_RANGES
                            .map(({min,max}) => `(hazyVertexCount >= ${min}.0 && hazyVertexCount <= ${max}.0)`)
                            .join(' || ');
                        const chickenUniforms = Array.from(
                            { length: CHICKEN_COUNT_SLOTS },
                            (_, i) => `uniform float hazyChickenCount${i};`
                        ).join('\n');
                        const chickenConditions = Array.from(
                            { length: CHICKEN_COUNT_SLOTS },
                            (_, i) => `(abs(hazyVertexCount - hazyChickenCount${i}) < 0.5)`
                        ).join(' || ');

                        source = source
                            .replace(/void\s+main\s*\(\s*\)\s*\{/, `
uniform float hazyVertexCount;
uniform float hazyESPEnabled;
uniform float hazyChickenESPEnabled;
${chickenUniforms}
out float hazyPlayerModel;
out float hazyChickenModel;
void main() {
    hazyPlayerModel = hazyVertexCount;
    hazyChickenModel = (${chickenConditions}) ? 1.0 : 0.0;
`)
                            .replace(/(gl_Position\s*=.+?;)/, `
$1
if (
    (hazyESPEnabled > 0.5 && (${playerConditions})) ||
    (hazyChickenESPEnabled > 0.5 && hazyChickenModel > 0.5)
) {
    gl_Position.z = 0.01 + gl_Position.z * 0.1;
}
`);
                    }

                    if (source.includes('SV_Target0') && !source.includes('hazyESPColor')) {
                        const playerConditions = PLAYER_VERTEX_RANGES
                            .map(({min,max}) => `(hazyPlayerModel >= ${min}.0 && hazyPlayerModel <= ${max}.0)`)
                            .join(' || ');

                        source = source
                            .replace(/void\s+main\s*\(\s*\)\s*\{/, `
uniform float hazyESPEnabled;
uniform float hazyChickenESPEnabled;
uniform vec3 hazyESPColor;
in float hazyPlayerModel;
in float hazyChickenModel;
void main() {
`)
                            .replace(/return;/, `
if (
    (
        (hazyESPEnabled > 0.5 && (${playerConditions})) ||
        (hazyChickenESPEnabled > 0.5 && hazyChickenModel > 0.5)
    ) &&
    SV_Target0.a > 0.5
) {
    SV_Target0 = vec4(hazyESPColor, 1.0);
}
return;
`);
                    }
                }
            } catch (_) {}
            return originalShaderSource.call(this,shader,source);
        };

        function updateESPUniforms(gl, vertexCount) {
            try {
                const program = gl.getParameter(gl.CURRENT_PROGRAM);
                if (!program) return;
                if (!program.__hazyUniforms) {
                    program.__hazyUniforms = {
                        vertexCount: gl.getUniformLocation(program,'hazyVertexCount'),
                        enabled: gl.getUniformLocation(program,'hazyESPEnabled'),
                        chickenEnabled: gl.getUniformLocation(program,'hazyChickenESPEnabled'),
                        chickenCounts: Array.from(
                            { length: CHICKEN_COUNT_SLOTS },
                            (_, i) => gl.getUniformLocation(program, `hazyChickenCount${i}`)
                        ),
                        color: gl.getUniformLocation(program,'hazyESPColor')
                    };
                }
                const u = program.__hazyUniforms;
                if (u.vertexCount !== null) gl.uniform1f(u.vertexCount,vertexCount);
                if (u.enabled !== null) gl.uniform1f(u.enabled,playerESP ? 1 : 0);
                if (u.chickenEnabled !== null) gl.uniform1f(u.chickenEnabled,chickenESP ? 1 : 0);
                if (u.chickenCounts) {
                    for (let i = 0; i < u.chickenCounts.length; i++) {
                        if (u.chickenCounts[i] !== null) {
                            gl.uniform1f(u.chickenCounts[i], chickenDrawCounts[i] || 0);
                        }
                    }
                }
                if (u.color !== null) {
                    gl.uniform3f(u.color,currentESPTheme.r/255,currentESPTheme.g/255,currentESPTheme.b/255);
                }
            } catch (_) {}
        }

        const originalDrawElements = GL.drawElements;
        GL.drawElements = function(...args) {
            if (isShadowFramebuffer(this)) return;
            onDrawCall(this);
            updateESPUniforms(this,args[1]);
            return originalDrawElements.apply(this,args);
        };

        if (typeof GL.drawElementsInstanced === 'function') {
            const originalDrawElementsInstanced = GL.drawElementsInstanced;
            GL.drawElementsInstanced = function(...args) {
                if (isShadowFramebuffer(this)) return;
                onDrawCall(this);
                updateESPUniforms(this,args[1]);
                return originalDrawElementsInstanced.apply(this,args);
            };
        }

        // Some Unity renderers use non-indexed draws for shadow casters too.
        if (typeof GL.drawArrays === 'function') {
            const originalDrawArrays = GL.drawArrays;
            GL.drawArrays = function(...args) {
                if (isShadowFramebuffer(this)) return;
                onDrawCall(this);
                updateESPUniforms(this,args[2]);
                return originalDrawArrays.apply(this,args);
            };
        }

        if (typeof GL.drawArraysInstanced === 'function') {
            const originalDrawArraysInstanced = GL.drawArraysInstanced;
            GL.drawArraysInstanced = function(...args) {
                if (isShadowFramebuffer(this)) return;
                onDrawCall(this);
                updateESPUniforms(this,args[2]);
                return originalDrawArraysInstanced.apply(this,args);
            };
        }
    }

    try { installWebGLHooks(); } catch (_) {}

    function anyProjectedVisualsOn() {
        return hitboxESP || linesESP || distanceESP || skeletonESP || offscreenArrows ||
            nameESP || healthESP || weaponESP || hazyBadges;
    }

    function drawOffscreenArrow(ctx, rect, width, height, screen) {
        if (!screen) return;
        const margin = Math.max(22, Math.min(42, Math.min(width, height) * 0.045));
        const centerX = width * 0.5;
        const centerY = height * 0.5;
        let dx = screen.x - centerX;
        let dy = screen.y - centerY;
        if (![dx, dy].every(Number.isFinite)) return;
        if (Math.hypot(dx, dy) < 0.001) dy = -1;

        const halfW = Math.max(1, centerX - margin);
        const halfH = Math.max(1, centerY - margin);
        const scale = Math.min(
            halfW / Math.max(0.001, Math.abs(dx)),
            halfH / Math.max(0.001, Math.abs(dy))
        );
        const px = rect.left + centerX + dx * scale;
        const py = rect.top + centerY + dy * scale;
        const angle = Math.atan2(dy, dx);
        const size = 10;

        ctx.save();
        ctx.translate(px, py);
        ctx.rotate(angle);
        ctx.beginPath();
        ctx.moveTo(size, 0);
        ctx.lineTo(-size * 0.72, size * 0.62);
        ctx.lineTo(-size * 0.45, 0);
        ctx.lineTo(-size * 0.72, -size * 0.62);
        ctx.closePath();
        ctx.fillStyle = rgb(currentMenuTheme);
        ctx.strokeStyle = 'rgba(0,0,0,0.82)';
        ctx.lineWidth = 2;
        ctx.shadowColor = rgba(currentMenuTheme, 0.70);
        ctx.shadowBlur = 6;
        ctx.fill();
        ctx.shadowBlur = 0;
        ctx.stroke();
        ctx.restore();
    }

    // Joint names are present in the supplied data.unity3d character assets.
    // This build strips Animator.GetBoneTransform/get_bones; traverse only the
    // local scene's character Animator hierarchy, then cache the real joints.
    const ESP_JOINT_NAMES = new Set([
        'Hips', 'Spine_01', 'Spine_02', 'Spine_03', 'Neck', 'Head',
        'Clavicle_L', 'Shoulder_L', 'Elbow_L', 'Hand_L',
        'Clavicle_R', 'Shoulder_R', 'Elbow_R', 'Hand_R',
        'UpperLeg_L', 'LowerLeg_L', 'Ankle_L', 'Ball_L', 'Toes_L',
        'UpperLeg_R', 'LowerLeg_R', 'Ankle_R', 'Ball_R', 'Toes_R'
    ]);
    const ESP_BOX_EDGES = [[0,1],[0,2],[0,4],[1,3],[1,5],[2,3],[2,6],
        [3,7],[4,5],[4,6],[5,7],[6,7]];
    function espObjectName(runtime, object) {
        const text = sceneFunction(runtime, 3300)(object, 0);
        return readManagedString(sceneView(runtime), text, 80);
    }
    function espRigRoot(info) {
        const runtime = info.runtime;
        let view = sceneView(runtime);
        const manager = infoUint(view, info.shooter + 0x44);
        if (!infoObjectNamed(view, manager, 'CharacterAnimatorManager', 0x5c) ||
            infoUint(view, manager + 0x14) !== info.shooter) return null;
        const animator = infoUint(view, manager + 0x40);
        if (!espAlive(view, animator, 'Animator')) return null;
        const root = sceneFunction(runtime, 2964)(animator, 0);
        view = sceneView(runtime);
        if (!espAlive(view, root, 'Transform')) return null;
        return { manager, animator, root, managerNative: infoUint(view, manager + 8),
            animatorNative: infoUint(view, animator + 8), rootNative: infoUint(view, root + 8) };
    }
    function espSameRig(cache, info, root) {
        return cache && root && cache.table === info.runtime.table &&
            cache.shooter === info.shooter && cache.shooterNative === info.shooterNative &&
            ['root', 'rootNative', 'manager', 'managerNative', 'animator', 'animatorNative']
                .every(k => cache[k] === root[k]);
    }
    function discoverESPRig(info, root, now) {
        const runtime = info.runtime, records = new Map(), children = new Map();
        const queue = [{ pointer: root.root, parent: 0, depth: 0 }];
        const visited = new Set();
        // Bounded traversal. No whole-heap scans, bone-index guesses or scene-wide searches.
        for (let i = 0; i < queue.length && i < 256; i++) {
            const node = queue[i];
            if (visited.has(node.pointer) || node.depth > 16) continue;
            visited.add(node.pointer);
            let view = sceneView(runtime);
            if (!espAlive(view, node.pointer, 'Transform')) continue;
            const native = infoUint(view, node.pointer + 8);
            const name = espObjectName(runtime, node.pointer);
            const record = { pointer: node.pointer, native, name, parent: node.parent };
            records.set(node.pointer, record);
            const siblings = children.get(node.parent) || [];
            siblings.push(node.pointer); children.set(node.parent, siblings);
            // Equipment/camera branches are not part of the body rig.
            if (/camera|weapon|viewmodel|canvas|particle|trail/i.test(name)) continue;
            const count = sceneFunction(runtime, 41095)(node.pointer, 0);
            if (!Number.isInteger(count) || count < 0 || count > 128) continue;
            for (let j = 0; j < count && queue.length < 256; j++) {
                const child = sceneFunction(runtime, 12214)(node.pointer, j, 0);
                if (child) queue.push({ pointer: child, parent: node.pointer, depth: node.depth + 1 });
            }
        }
        // Restrict to one coherent Hips subtree: never join body bones to a
        // similarly named head/hand on the first-person weapon or another rig.
        let best = [];
        for (const candidate of records.values()) {
            if (candidate.name !== 'Hips') continue;
            const branch = [candidate.pointer], selected = [];
            for (let i = 0; i < branch.length && i < 256; i++) {
                const record = records.get(branch[i]);
                if (record && ESP_JOINT_NAMES.has(record.name)) selected.push(record);
                branch.push(...(children.get(branch[i]) || []));
            }
            if (selected.length > best.length) best = selected;
        }
        const chosen = new Set(best.map(b => b.pointer)), edges = [];
        for (const bone of best) {
            let parent = bone.parent;
            for (let i = 0; parent && i < 16; i++) {
                if (chosen.has(parent)) { edges.push([parent, bone.pointer]); break; }
                parent = records.get(parent)?.parent || 0;
            }
        }
        const cache = { ...root, table: runtime.table, shooter: info.shooter,
            shooterNative: info.shooterNative, bones: best, edges, checked: now };
        espRigCache.set(info.actor, cache);
        return cache;
    }
    function readESPPose(info, rootPosition, frame, budget, now) {
        if (!skeletonESP || info.isLocalBadge) return null;
        const root = espRigRoot(info);
        if (!root) { espRigCache.delete(info.actor); return null; }
        let rig = espRigCache.get(info.actor);
        if (!espSameRig(rig, info, root)) { espRigCache.delete(info.actor); rig = null; }
        if ((!rig || now - rig.checked > 2000) && budget.remaining > 0) {
            budget.remaining--; rig = discoverESPRig(info, root, now);
        }
        if (!rig || rig.bones.length < 5 || rig.edges.length < 4) return null;
        const points = new Map();
        for (const bone of rig.bones) {
            let view = sceneView(info.runtime);
            if (!espAlive(view, bone.pointer, 'Transform', bone.native) ||
                sceneFunction(info.runtime, 41067)(bone.pointer, 0) !== bone.parent) {
                espRigCache.delete(info.actor); return null;
            }
            const p = espVector(info.runtime, bone.pointer, frame.scratch);
            if (!p || Math.hypot(p.x-rootPosition.x, p.y-rootPosition.y, p.z-rootPosition.z) > 20) {
                espRigCache.delete(info.actor); return null;
            }
            points.set(bone.pointer, p);
        }
        return { points, edges: rig.edges };
    }
    function espRendererAlive(view, object, native) {
        return infoValid(view, object, 12) && infoUint(view, object + 8) !== 0 &&
            (native === undefined || infoUint(view, object + 8) === native) &&
            infoDerivesFrom(view, infoUint(view, object), 'Renderer');
    }
    function readESPBounds(info, rootPosition, frame) {
        const runtime = info.runtime;
        let view = sceneView(runtime);
        const skin = infoUint(view, info.shooter + 0x3c);
        if (!infoObjectNamed(view, skin, 'CharacterSkinManager', 0x38)) return null;
        const head = infoUint(view, skin + 0x1c), array = infoUint(view, skin + 0x28);
        const skinNative = infoUint(view, skin + 8);
        let cache = espRendererCache.get(info.actor);
        const refs = head ? [head] : [];
        // bodyMeshRenderers is a managed SkinnedMeshRenderer[] in this supplied build.
        if (infoValid(view, array, 16)) {
            const klass = infoUint(view, array), namePtr = infoUint(view, klass + 8);
            if (infoAsciiEquals(view, namePtr, 'MeshRenderer[]') ||
                infoAsciiEquals(view, namePtr, 'Renderer[]') ||
                infoAsciiEquals(view, namePtr, 'SkinnedMeshRenderer[]')) {
                const count = view.getInt32(array + 12, true);
                if (count >= 0 && count <= 32 && infoValid(view, array, 16 + count * 4)) {
                    for (let i = 0; i < count; i++) refs.push(infoUint(view, array + 16 + i * 4));
                }
            }
        }
        const valid = [...new Set(refs)].filter(p => espRendererAlive(view, p));
        // Require some body geometry, not just an isolated hat/head.
        if (!valid.some(p => p !== head)) return null;
        const signature = valid.map(p => `${p}:${infoUint(view, p + 8)}`).join('|');
        if (!cache || cache.shooter !== info.shooter || cache.shooterNative !== info.shooterNative ||
            cache.skin !== skin || cache.skinNative !== skinNative || cache.signature !== signature) {
            cache = { shooter: info.shooter, shooterNative: info.shooterNative, skin, skinNative,
                signature, refs: valid.map(pointer => ({ pointer, native: infoUint(view, pointer + 8) })) };
            espRendererCache.set(info.actor, cache);
        }
        const min = { x: Infinity, y: Infinity, z: Infinity }, max = { x: -Infinity, y: -Infinity, z: -Infinity };
        let used = 0;
        for (const renderer of cache.refs) {
            view = sceneView(runtime);
            if (!espRendererAlive(view, renderer.pointer, renderer.native)) continue;
            sceneFunction(runtime, 2963)(frame.scratch, renderer.pointer, 0);
            view = sceneView(runtime);
            const n = [0,4,8,12,16,20].map(o => view.getFloat32(frame.scratch + o, true));
            if (!n.every(Number.isFinite) || n.slice(3).some(v => v < 0 || v > 12) ||
                Math.max(...n.slice(3)) < 0.005 ||
                Math.hypot(n[0]-rootPosition.x,n[1]-rootPosition.y,n[2]-rootPosition.z) > 20) continue;
            ['x','y','z'].forEach((axis, i) => {
                min[axis] = Math.min(min[axis], n[i]-n[i+3]);
                max[axis] = Math.max(max[axis], n[i]+n[i+3]);
            });
            used++;
        }
        if (!used || max.y-min.y < 0.05 || ['x','y','z'].some(k => max[k]-min[k] > 25)) return null;
        return { min, max };
    }
    function boundsFromESPPose(pose) {
        if (!pose?.points.size) return null;
        const min = { x: Infinity, y: Infinity, z: Infinity }, max = { x: -Infinity, y: -Infinity, z: -Infinity };
        for (const p of pose.points.values()) for (const k of ['x','y','z']) {
            min[k] = Math.min(min[k], p[k] - 0.15); max[k] = Math.max(max[k], p[k] + 0.15);
        }
        return { min, max };
    }
    function screenESPBounds(bounds, frame = espFrame) {
        if (!bounds || !frame) return null;
        const corners = Array.from({length:8}, (_, i) => ({
            x: (i&1) ? bounds.max.x : bounds.min.x, y: (i&2) ? bounds.max.y : bounds.min.y,
            z: (i&4) ? bounds.max.z : bounds.min.z }));
        const points = [];
        for (const [i,j] of ESP_BOX_EDGES) {
            const edge = projectESPSegment(corners[i], corners[j], frame);
            if (edge) points.push(...edge);
        }
        if (!points.length) return null;
        const x1 = Math.min(...points.map(p=>p.x)), x2 = Math.max(...points.map(p=>p.x));
        const y1 = Math.min(...points.map(p=>p.y)), y2 = Math.max(...points.map(p=>p.y));
        if (![x1,x2,y1,y2].every(Number.isFinite) || x2<0 || y2<0 || x1>frame.rect.width || y1>frame.rect.height) return null;
        // Avoid a giant enclosing rectangle when the camera is inside the body.
        if (x2-x1>frame.rect.width*4 || y2-y1>frame.rect.height*4) return null;
        return { x:frame.rect.left+x1, y:frame.rect.top+y1, width:x2-x1, height:y2-y1 };
    }
    // Classic Hitbox ESP: one upright full-body rectangle, independent of
    // animated renderer bounds, hats, head meshes, and Skeleton ESP state.
    // Keep the live root/camera from v2.63; only restore the old display shape.
    function uprightESPBodyBox(root, frame = espFrame) {
        if (!root || !frame || !['x', 'y', 'z'].every(k => Number.isFinite(root[k]))) return null;
        const rect = frame.rect;
        if (!rect || ![rect.left, rect.top, rect.width, rect.height].every(Number.isFinite) ||
            rect.width < 10 || rect.height < 10) return null;

        // Same world-height endpoints as the earlier tall-rectangle overlay.
        // These are visual proportions, not changes to any collision hitbox.
        const feet = projectESPPoint({ x: root.x, y: root.y - 0.15, z: root.z }, false, frame);
        const topOfBody = projectESPPoint({ x: root.x, y: root.y + 3.05, z: root.z }, false, frame);
        if (!feet || !topOfBody) return null;

        const top = Math.min(feet.y, topOfBody.y);
        const height = Math.abs(feet.y - topOfBody.y);
        // Reject tiny/near-camera projections rather than stretching a box
        // across the display. Normal viewport clipping still happens at draw time.
        if (!Number.isFinite(height) || height < 8 || height > rect.height * 1.5) return null;
        const width = height * 0.42;
        const left = (feet.x + topOfBody.x) * 0.5 - width * 0.5;
        if (!Number.isFinite(left) || left + width < 0 || left > rect.width ||
            top + height < 0 || top > rect.height) return null;

        return { x: rect.left + left, y: rect.top + top, width, height };
    }

    function sampleESPPlayer(info, frame, budget, now) {
        if (now-info.t > PLAYER_INFO_STALE_MS || info.runtime.table !== frame.runtime.table) return null;
        let view = sceneView(info.runtime);
        if (!espAlive(view,info.shooter,'Shooter',info.shooterNative) ||
            !espAlive(view,info.transform,'Transform',info.transformNative)) return null;
        const photon = infoUint(view,info.shooter+0x10);
        if (!infoObjectNamed(view,photon,'PhotonView',0x5c)) return null;
        const mine = view.getUint8(photon+0x40) === 1;
        if (mine !== info.isLocalBadge || (mine && (!thirdPerson || !badgeRoleForActor(info.actor)))) return null;
        const owner = infoUint(view,photon+0x50);
        if (!infoValid(view,owner,0x28) || view.getInt32(owner+0x0c,true)!==info.actor || view.getUint8(owner+0x1c)) return null;
        const hp = infoUint(view,info.shooter+0x6c);
        if (infoObjectNamed(view,hp,'Health',0xb8) &&
            (view.getUint8(hp+(mine?0x80:0xa8)) || readObscuredInt(view,hp+0x6c) === 0)) return null;
        const root = espVector(info.runtime,info.transform,frame.scratch);
        if (!root) return null;
        const pose = readESPPose(info,root,frame,budget,now);
        const bounds = readESPBounds(info,root,frame) || boundsFromESPPose(pose);
        const center = bounds ? { x:(bounds.min.x+bounds.max.x)/2, y:(bounds.min.y+bounds.max.y)/2,
            z:(bounds.min.z+bounds.max.z)/2 } : {x:root.x,y:root.y+1.45,z:root.z};
        const head = bounds ? {x:center.x,y:bounds.max.y+0.08,z:center.z} : {x:root.x,y:root.y+3.05,z:root.z};
        return { info, root, pose, bounds, center, head, box:uprightESPBodyBox(root,frame),
            distance:Math.hypot(center.x-frame.eye.x,center.y-frame.eye.y,center.z-frame.eye.z) };
    }
    function drawSkeletonESP(ctx, sample, frame = espFrame) {
        if (!sample.pose || !frame) return false;
        ctx.save();
        try {
            ctx.strokeStyle = rgb(currentESPTheme); ctx.lineWidth = 1.75;
            ctx.shadowColor = rgba(currentESPTheme,0.4); ctx.shadowBlur = 2;
            ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.beginPath();
            let count = 0;
            for (const [a,b] of sample.pose.edges) {
                const p=sample.pose.points.get(a), q=sample.pose.points.get(b);
                if (!p || !q || Math.hypot(p.x-q.x,p.y-q.y,p.z-q.z)>8) continue;
                const segment = projectESPSegment(p,q,frame);
                if (!segment) continue;
                ctx.moveTo(frame.rect.left+segment[0].x, frame.rect.top+segment[0].y);
                ctx.lineTo(frame.rect.left+segment[1].x, frame.rect.top+segment[1].y); count++;
            }
            if (count) ctx.stroke();
            return count>0;
        } finally { ctx.restore(); }
    }

    // Uses live scene objects and the local camera, never guessed network positions.
    // These are visual overlays only; they do not change player colliders/hitboxes.
    function ensureHitboxCanvas() {
        if (hitboxCanvas && hitboxCanvas.isConnected && hitboxContext) return true;
        if (!document.body) return false;

        const canvas = document.createElement('canvas');
        canvas.id = 'hazyHitboxESP';
        Object.assign(canvas.style, {
            position: 'fixed',
            left: '0',
            top: '0',
            width: '100vw',
            height: '100vh',
            pointerEvents: 'none',
            userSelect: 'none',
            zIndex: '2147483000'
        });

        document.body.appendChild(canvas);
        hitboxCanvas = canvas;
        hitboxContext = canvas.getContext('2d');
        return Boolean(hitboxContext);
    }

    function sizeHitboxCanvas() {
        if (!hitboxCanvas || !hitboxContext) return false;
        const dpr=Math.max(1,Math.min(2,window.devicePixelRatio||1));
        const width=Math.max(1,Math.round(window.innerWidth*dpr));
        const height=Math.max(1,Math.round(window.innerHeight*dpr));
        if (hitboxCanvas.width!==width) hitboxCanvas.width=width;
        if (hitboxCanvas.height!==height) hitboxCanvas.height=height;
        hitboxContext.setTransform(dpr,0,0,dpr,0,0);
        return true;
    }
    function clearVisualESP(resetTracking=false) {
        if (hitboxContext && hitboxCanvas) {
            hitboxContext.save();
            try {
                hitboxContext.setTransform(1,0,0,1,0,0);
                hitboxContext.clearRect(0,0,hitboxCanvas.width,hitboxCanvas.height);
            } finally { hitboxContext.restore(); }
        }
        if (resetTracking) hitboxRenderStates.clear();
    }

    // Backward-compatible name used by the existing Hitbox toggle.
    function clearHitboxESP() {
        clearVisualESP(!anyProjectedVisualsOn());
    }

    function updateVisualESP(nowPerf = performance.now()) {
        if (!ensureHitboxCanvas() || !sizeHitboxCanvas()) return;
        clearVisualESP(false);
        const frame = espFrame;
        if (!frame || !espInputActive() || !anyProjectedVisualsOn() || nowPerf-frame.at>250) return;
        const ctx = hitboxContext, rect = frame.rect;
        const samples = [], budget = { remaining: 1 }, seen = new Set();
        espStatus.players = 0; espStatus.animatedSkeletons = 0;
        ctx.save();
        try {
            // All UI positions are CSS pixels, clipped to the actual camera viewport.
            ctx.beginPath(); ctx.rect(rect.left,rect.top,rect.width,rect.height); ctx.clip();
            ctx.setLineDash([]); ctx.lineCap='round'; ctx.lineJoin='round';
            for (const info of playerInfoByActor.values()) {
                let sample;
                try { sample=sampleESPPlayer(info,frame,budget,nowPerf); }
                catch (error) { espStatus.lastError=error.message||String(error); continue; }
                if (!sample) continue;
                seen.add(info.actor); samples.push(sample);
                if (info.isLocalBadge) continue; // Only the own badge is drawn in Third Person.
                espStatus.players++;
                const torso=projectESPPoint(sample.center), loose=projectESPPoint(sample.center,true);
                if (offscreenArrows && loose && (loose.behind || loose.x<0 || loose.y<0 ||
                    loose.x>rect.width || loose.y>rect.height)) {
                    drawOffscreenArrow(ctx,rect,rect.width,rect.height,loose);
                }
                if (linesESP && torso) {
                    ctx.beginPath(); ctx.moveTo(rect.left+rect.width/2,rect.top+rect.height/2);
                    ctx.lineTo(rect.left+torso.x,rect.top+torso.y);
                    ctx.strokeStyle=rgb(currentMenuTheme); ctx.lineWidth=1.75;
                    ctx.shadowColor=rgba(currentMenuTheme,0.5); ctx.shadowBlur=3; ctx.stroke();
                }
                const box=sample.box;
                if (hitboxESP && box && box.width>=1 && box.height>=1) {
                    ctx.strokeStyle=rgb(currentESPTheme); ctx.lineWidth=2;
                    ctx.shadowColor=rgba(currentESPTheme,0.45); ctx.shadowBlur=2;
                    ctx.strokeRect(box.x,box.y,box.width,box.height);
                }
                if (skeletonESP && drawSkeletonESP(ctx,sample,frame)) espStatus.animatedSkeletons++;
            }
            drawPlayerInfoLabels(ctx,rect,nowPerf,samples);
            for (const actor of espRigCache.keys()) if (!seen.has(actor)) espRigCache.delete(actor);
            for (const actor of espRendererCache.keys()) if (!seen.has(actor)) espRendererCache.delete(actor);
        } finally { ctx.restore(); }
    }

    function getAimPoint(position) {
        return { x:position.x, y:position.y + AIM_POINT_CHOICES[aimPointIndex].height, z:position.z };
    }

    function targetForActor(actor, position, width, height, maxRadius) {
        if (!position || Date.now() - position.t > STALE_POSITION_MS) return null;
        const screen = projectToScreen(getAimPoint(position), width, height);
        if (!screen || screen.x < 0 || screen.x > width ||
            screen.y < 0 || screen.y > height) return null;
        const dx = screen.x - width / 2;
        const dy = screen.y - height / 2;
        const distance = Math.hypot(dx, dy);
        return distance <= maxRadius ? { actor, dx, dy, distance } : null;
    }

    function pickClosestTarget(width, height) {
        const now = Date.now();
        for (const [actor, position] of positions) {
            if (now - position.t > STALE_POSITION_MS) {
                positions.delete(actor);
                if (lockedActor === actor) lockedActor = null;
            }
        }
        if (lockedActor !== null) {
            const target = targetForActor(lockedActor, positions.get(lockedActor),
                width, height, AIM_FOV_CHOICES[aimFovIndex] + 50);
            if (target) return target;
            lockedActor = null;
        }
        let best = null;
        for (const [actor, position] of positions) {
            const target = targetForActor(actor, position, width, height, AIM_FOV_CHOICES[aimFovIndex]);
            if (target && (!best || target.distance < best.distance)) best = target;
        }
        if (best) lockedActor = best.actor;
        return best;
    }

    function roundAwayFromZero(value) {
        if (value === 0) return 0;
        const rounded = Math.round(value);
        if (rounded !== 0) return rounded;
        return value > 0 ? 1 : -1;
    }

    function sendMouseMovement(canvas,moveX,moveY) {
        try {
            const rect = canvas.getBoundingClientRect();
            const event = new MouseEvent('mousemove', {
                bubbles:true,cancelable:true,composed:true,view:window,
                clientX:rect.left+rect.width/2,
                clientY:rect.top+rect.height/2
            });
            Object.defineProperty(event,'movementX',{value:moveX,configurable:true});
            Object.defineProperty(event,'movementY',{value:moveY,configurable:true});
            injectingMouseMove = true;
            try { canvas.dispatchEvent(event); }
            finally { injectingMouseMove = false; }
        } catch (_) { injectingMouseMove = false; }
    }

    function updateAimbot() {
        if (!aimbot || !aimButtonHeld || !gameAimInputAllowed()) { lockedActor = null; return; }
        if (performance.now() < manualAimOverrideUntil) return;
        if (!aimbot || !activeGL || !activeGL.canvas || !vpMatrix || !document.pointerLockElement) return;
        const canvas = activeGL.canvas;
        const width = canvas.clientWidth || canvas.width;
        const height = canvas.clientHeight || canvas.height;
        if (width < 10 || height < 10) return;
        const target = pickClosestTarget(width,height);
        if (!target || target.distance < AIM_DEADZONE) return;

        const strength = AIM_STRENGTH_CHOICES[aimStrengthIndex];
        const moveX = roundAwayFromZero(Math.max(-strength.maxTurn, Math.min(strength.maxTurn, target.dx * strength.gain)));
        const moveY = roundAwayFromZero(Math.max(-strength.maxTurn, Math.min(strength.maxTurn, target.dy * strength.gain)));
        sendMouseMovement(canvas,moveX,moveY);
    }

    function updateNoRecoil() {
        if (!noRecoil || !leftMouseHeld || !activeGL || !activeGL.canvas || !document.pointerLockElement) return;
        sendMouseMovement(activeGL.canvas,0,NO_RECOIL_STRENGTH);
    }

    // ============================================================
    // DISPLAY FPS + DISPLAY PING
    // ============================================================
    // FPS comes from requestAnimationFrame. Ping comes directly from
    // PhotonNetwork.GetPing() in Kour's own networking assembly.
    let fpsSampleStart = performance.now();
    let fpsSampleFrames = 0;
    let displayedFPSValue = 0;

    let lastPingReadAt = 0;
    let displayedPingValue = null;

    function updateFPSCounter(now) {
        fpsSampleFrames++;

        const elapsed = now - fpsSampleStart;
        if (elapsed < 250) return;

        const measured = fpsSampleFrames * 1000 / Math.max(1, elapsed);
        displayedFPSValue =
            displayedFPSValue > 0
                ? displayedFPSValue * 0.55 + measured * 0.45
                : measured;

        fpsSampleFrames = 0;
        fpsSampleStart = now;

        if (!fpsCounterElement) return;

        if (!displayFPS) {
            fpsCounterElement.style.display = 'none';
            return;
        }

        fpsCounterElement.style.display = 'block';
        fpsCounterElement.textContent =
            `𝙁𝙋𝙎: ${Math.max(0, Math.round(displayedFPSValue))}`;
    }

    function updatePingCounter(now) {
        if (!pingCounterElement) return;

        if (!displayPing) {
            pingCounterElement.style.display = 'none';
            return;
        }

        pingCounterElement.style.display = 'block';

        if (now - lastPingReadAt >= 250) {
            lastPingReadAt = now;

            try {
                const api = photonMethods();
                if (api) {
                    const ping = api.getPing(0) | 0;
                    if (ping > 0 && ping <= 10000) {
                        displayedPingValue = ping;
                    } else {
                        displayedPingValue = null;
                    }
                }
            } catch (_) {}
        }

        pingCounterElement.textContent =
            displayedPingValue === null
                ? '𝙋𝙄𝙉𝙂: --'
                : `𝙋𝙄𝙉𝙂: ${displayedPingValue}ms`;
    }

    function positionStatsCounters() {
        if (!activeModsPanel) return;

        const rect = activeModsPanel.getBoundingClientRect();
        let nextTop = rect.bottom + 8;

        if (fpsCounterElement) {
            if (displayFPS) {
                fpsCounterElement.style.left = `${Math.round(rect.left)}px`;
                fpsCounterElement.style.top = `${Math.round(nextTop)}px`;
                fpsCounterElement.style.width = `${Math.round(rect.width)}px`;

                const fpsHeight =
                    fpsCounterElement.getBoundingClientRect().height || 37;
                nextTop += fpsHeight + 8;
            }
        }

        if (pingCounterElement) {
            if (displayPing) {
                pingCounterElement.style.left = `${Math.round(rect.left)}px`;
                pingCounterElement.style.top = `${Math.round(nextTop)}px`;
                pingCounterElement.style.width = `${Math.round(rect.width)}px`;
            }
        }
    }

    function refreshStatsCounterVisibility() {
        if (fpsCounterElement) {
            if (displayFPS) {
                fpsCounterElement.style.display = 'block';
                fpsCounterElement.textContent =
                    `𝙁𝙋𝙎: ${Math.max(0, Math.round(displayedFPSValue))}`;
            } else {
                fpsCounterElement.style.display = 'none';
            }
        }

        if (pingCounterElement) {
            if (displayPing) {
                pingCounterElement.style.display = 'block';
                pingCounterElement.textContent =
                    displayedPingValue === null
                        ? '𝙋𝙄𝙉𝙂: --'
                        : `𝙋𝙄𝙉𝙂: ${displayedPingValue}ms`;
            } else {
                pingCounterElement.style.display = 'none';
            }
        }

        requestAnimationFrame(positionStatsCounters);
    }

    window.addEventListener('resize', positionStatsCounters);

    function frame(now) {
        requestAnimationFrame(frame);
        updateFPSCounter(now);
        updatePingCounter(now);
        flushPingBoost(now);
        updateBunnyHop(now);
        updateAutoJump(now);
        updateDoubleJumpGroundState(now);
        updateGlide(now);
        updateAutoReload(now);
        updateThirdPerson();
        updateLiveTheme(now);
        maintainVisualESP(now);
        updateAimbot();
        updateTriggerbot(now);
        updateNoRecoil();
        // Speed maintenance runs separately, at most twice per second.
    }
    requestAnimationFrame(frame);

        Object.assign(categoryHeading.style, {
            height: '22px',
            lineHeight: '22px',
            margin: '0 20px 8px',
            color: 'rgba(255,255,255,0.88)',
            fontSize: '14px',
            fontWeight: '700',
            textAlign: 'center',
            textShadow: '0 1px 4px rgba(0,0,0,0.28)',
            pointerEvents: 'none'
        });
        panel.appendChild(categoryHeading);

        const pageArea = document.createElement('div');
        pageArea.id = 'hazyCategoryPages';
        Object.assign(pageArea.style, {
            width: '100%',
            height: '278px',
            boxSizing: 'border-box',
            overflowY: 'auto'
        });
        panel.appendChild(pageArea);

        function createMenuButton(parent, label, onClick, getState = null) {
            const button = document.createElement('button');
            button.type = 'button';
            const refreshLabel = () => {
                const value = typeof label === 'function' ? label() : label;
                if (button.textContent !== value) {
                    button.textContent = value;
                    button.setAttribute('aria-label', value.normalize('NFKC'));
                }
            };
            refreshLabel();
            Object.assign(button.style, {
                display: 'block', width: '260px', height: '46px',
                flexShrink: '0', margin: '0 auto', boxSizing: 'border-box',
                border: '2px solid transparent', borderRadius: '8px', color: 'white',
                fontSize: '18px', fontWeight: '700', cursor: 'pointer',
                transition: 'background-color 80ms linear, border-color 80ms linear, transform 120ms ease',
                textShadow: '0 1px 4px rgba(0,0,0,0.28)'
            });

            // The regular theme loop reads this getter even for hidden pages.
            // Category links have no on/off state; mod buttons retain theirs.
            const readButtonState = () => {
                refreshLabel();
                const enabled = getState ? Boolean(getState()) : false;
                if (getState && button.getAttribute('aria-pressed') !== String(enabled)) {
                    button.setAttribute('aria-pressed', String(enabled));
                }
                return enabled;
            };
            readButtonState();
            modButtonElements.push({ button, getState: readButtonState });

            button.addEventListener('pointerdown', e => {
                e.stopPropagation();
                button.style.transform = 'scale(0.97)';
            });
            for (const eventName of ['pointerup', 'pointercancel', 'pointerleave']) {
                button.addEventListener(eventName, () => {
                    button.style.transform = 'scale(1)';
                });
            }
            button.addEventListener('click', e => {
                e.stopPropagation();
                onClick();
                renderActiveMods();
                updateLiveTheme(performance.now());
            });
            parent.appendChild(button);
            return button;
        }

        // Create buttons once, then switch page visibility. Going Back or
        // changing category never disables a mod or duplicates its listeners.
        const allPages = [];
        function makePages(items, addButton) {
            const result = [];
            const count = Math.max(1, Math.ceil(items.length / MODS_PER_PAGE));
            for (let index = 0; index < count; index++) {
                const page = document.createElement('div');
                page.hidden = true;
                Object.assign(page.style, {
                    display: 'none', flexDirection: 'column', gap: '6px',
                    padding: '0', boxSizing: 'border-box'
                });
                const start = index * MODS_PER_PAGE;
                for (const item of items.slice(start, start + MODS_PER_PAGE)) {
                    addButton(page, item);
                }
                pageArea.appendChild(page);
                allPages.push(page);
                result.push(page);
            }
            return result;
        }

        const homePages = makePages(categories, (page, category) => {
            const button = createMenuButton(page, category.label, () => {
                showCategory(category.id);
            });
            button.dataset.categoryTarget = category.id;
        });

        for (const category of categories) {
            category.pages = makePages(category.mods, (page, mod) => {
                const button = createMenuButton(page, mod.label, mod.toggle, mod.getState);
                button.dataset.modId = mod.id;
                if (mod.description) button.title = mod.description;
            });
        }

        // ========================================================
        // PAGE CONTROLS — PREV / BACK / NEXT
        // ========================================================

        let currentCategory = null; // null = the category start screen
        let currentPage = 0;

        // Keep the page count above the buttons to leave the center for Back.
        const pageIndicator = document.createElement('div');
        pageIndicator.id = 'hazyPageIndicator';
        pageIndicator.setAttribute('aria-live', 'polite');
        Object.assign(pageIndicator.style, {
            position: 'absolute', left: '20px', right: '20px', bottom: '66px',
            height: '18px', lineHeight: '18px', color: 'rgba(255,255,255,0.88)',
            textAlign: 'center', fontSize: '13px', fontWeight: '700',
            textShadow: '0 1px 4px rgba(0,0,0,0.30)',
            userSelect: 'none', pointerEvents: 'none'
        });
        panel.appendChild(pageIndicator);

        const pageNav = document.createElement('div');
        pageNav.id = 'hazyPageNav';
        pageNav.setAttribute('role', 'navigation');
        pageNav.setAttribute('aria-label', 'Hazy Client menu navigation');
        Object.assign(pageNav.style, {
            position: 'absolute', left: '20px', right: '20px', bottom: '18px',
            height: '42px', display: 'flex', alignItems: 'center',
            justifyContent: 'space-between', gap: '8px'
        });

        function createNavButton(label, id, ariaLabel, onClick) {
            const button = document.createElement('button');
            button.id = id;
            button.type = 'button';
            button.textContent = label;
            button.setAttribute('aria-label', ariaLabel);
            Object.assign(button.style, {
                flex: '1 1 0', minWidth: '0', height: '36px', padding: '0 5px',
                border: '2px solid rgba(255,255,255,0.72)', borderRadius: '8px',
                background: 'rgba(0,0,0,0.28)', color: 'white',
                fontSize: '14px', fontWeight: '700', cursor: 'pointer',
                transition: 'transform 120ms ease, opacity 120ms ease, background-color 120ms ease',
                textShadow: '0 1px 4px rgba(0,0,0,0.30)'
            });
            button.addEventListener('pointerdown', e => {
                e.stopPropagation();
                if (!button.disabled) button.style.transform = 'scale(0.96)';
            });
            for (const eventName of ['pointerup', 'pointercancel', 'pointerleave']) {
                button.addEventListener(eventName, () => {
                    button.style.transform = 'scale(1)';
                });
            }
            button.addEventListener('click', e => {
                e.stopPropagation();
                if (!button.disabled) onClick();
            });
            pageNav.appendChild(button);
            return button;
        }

        const prevPage = createNavButton('𝙋𝙧𝙚𝙫', 'hazyPrevPage', 'Previous page', () => {
            showPage(currentPage - 1);
        });
        const backButton = createNavButton('𝘽𝙖𝙘𝙠', 'hazyBack', 'Back to categories', () => {
            showCategories();
        });
        const nextPage = createNavButton('𝙉𝙚𝙭𝙩', 'hazyNextPage', 'Next page', () => {
            showPage(currentPage + 1);
        });
        panel.appendChild(pageNav);

        function setNavDisabled(button, disabled) {
            button.disabled = disabled;
            button.style.opacity = disabled ? '0.38' : '1';
            button.style.cursor = disabled ? 'default' : 'pointer';
            button.style.transform = 'scale(1)';
        }

        function showPage(index) {
            const pages = currentCategory ? currentCategory.pages : homePages;
            currentPage = Math.max(0, Math.min(index, pages.length - 1));
            const selectedPage = pages[currentPage];
            const focusedElement = document.activeElement;
            const focusWasInPage = pageArea.contains(focusedElement);

            for (const page of allPages) {
                const visible = page === selectedPage;
                page.hidden = !visible;
                page.style.display = visible ? 'flex' : 'none';
            }
            pageArea.scrollTop = 0;
            panel.dataset.category = currentCategory ? currentCategory.id : 'home';
            categoryHeading.textContent = currentCategory
                ? currentCategory.label : '𝘾𝙖𝙩𝙚𝙜𝙤𝙧𝙞𝙚𝙨';
            categoryHeading.setAttribute('aria-label', currentCategory
                ? currentCategory.name : 'Categories');
            pageIndicator.textContent = `${currentPage + 1} / ${pages.length}`;
            pageIndicator.setAttribute('aria-label',
                `${currentCategory ? currentCategory.name : 'Categories'}, page ${currentPage + 1} of ${pages.length}`);

            setNavDisabled(prevPage, currentPage === 0);
            setNavDisabled(nextPage, currentPage === pages.length - 1);
            setNavDisabled(backButton, currentCategory === null);

            // Avoid leaving keyboard focus in a page that has just been hidden.
            if (focusWasInPage || focusedElement?.disabled) {
                selectedPage.querySelector('button')?.focus({ preventScroll: true });
            }
        }

        function showCategory(id) {
            const category = categories.find(item => item.id === id);
            if (!category) return;
            currentCategory = category;
            showPage(0);
        }

        function showCategories() {
            currentCategory = null;
            showPage(0); // Back always returns to the START of categories.
        }

        showCategories();

        document.body.appendChild(panel);
        updateLiveTheme(performance.now());
        renderActiveMods();

        let closeTimer = null;
        function openMenu(playSound = true) {
            const wasClosed = !menuOpen;
            clearTimeout(closeTimer);
            menuOpen = true;
            panel.style.display = 'block';
            panel.style.pointerEvents = 'auto';
            panel.classList.remove('hazy-menu-close');
            void panel.offsetWidth;
            panel.classList.add('hazy-menu-open');
            if (playSound && wasClosed) void menuDing.play();
        }

        function closeMenu() {
            menuOpen = false;
            menuDing.cancelPending();
            panel.classList.remove('hazy-menu-open');
            void panel.offsetWidth;
            panel.classList.add('hazy-menu-close');
            panel.style.pointerEvents = 'none';
            clearTimeout(closeTimer);
            closeTimer = setTimeout(() => {
                if (!menuOpen) {
                    panel.style.display = 'none';
                    panel.classList.remove('hazy-menu-close');
                }
            },190);
        }

        window.addEventListener('keydown', e => {
            if (e.code === 'ShiftRight' && !e.repeat) {
                menuOpen ? closeMenu() : openMenu();
            }
        }, true);

        let dragging = false;
        let offsetX = 0;
        let offsetY = 0;

        panel.addEventListener('pointerdown', e => {
            if (e.target.closest('button')) return;
            dragging = true;
            const rect = panel.getBoundingClientRect();
            offsetX = e.clientX - rect.left;
            offsetY = e.clientY - rect.top;
            panel.style.left = `${rect.left}px`;
            panel.style.top = `${rect.top}px`;
            panel.style.right = 'auto';
            panel.style.transformOrigin = 'center';
            title.style.cursor = 'grabbing';
            try { panel.setPointerCapture(e.pointerId); } catch (_) {}
        });

        panel.addEventListener('pointermove', e => {
            if (!dragging) return;
            let x = e.clientX-offsetX;
            let y = e.clientY-offsetY;
            x = Math.max(0,Math.min(x,window.innerWidth-panel.offsetWidth));
            y = Math.max(0,Math.min(y,window.innerHeight-panel.offsetHeight));
            panel.style.left = `${x}px`;
            panel.style.top = `${y}px`;
        });

        function stopDragging(e) {
            dragging = false;
            title.style.cursor = 'grab';
            try {
                if (panel.hasPointerCapture(e.pointerId)) panel.releasePointerCapture(e.pointerId);
            } catch (_) {}
        }
        panel.addEventListener('pointerup',stopDragging);
        panel.addEventListener('pointercancel',stopDragging);

        openMenu(false); // Initial automatic display stays silent.
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded',createMenu,{once:true});
    } else {
        createMenu();
    }
})();
